//callsOf, 自一列操作紀錄(JSON 事件)取出工具呼叫 [{ tool(小寫之工具名), input }](可能多個); 兩種格式:
//  ① opencode --format json: part.tool 與 part.state.input  ② claude --output-format stream-json: message.content[*](type 為 tool_use)之 name 與 input
let callsOf = (j) => {
    let out = []
    let a = j && j.part && j.part.state && j.part.state.input
    if (a && typeof a === 'object') {
        //status / error: 該次工具呼叫之結果(opencode 之紀錄才有; 供分辨「被權限拒絕而未執行」者)
        out.push({ tool: String((j.part && j.part.tool) || '').toLowerCase(), input: a, status: typeof j.part.state.status === 'string' ? j.part.state.status : null, error: typeof j.part.state.error === 'string' ? j.part.state.error : null })
    }
    let cs = j && j.message && Array.isArray(j.message.content) ? j.message.content : []
    for (let c of cs) {
        if (c && c.type === 'tool_use' && c.input && typeof c.input === 'object') {
            out.push({ tool: String(c.name || '').toLowerCase(), input: c.input })
        }
    }
    return out
}

//deniedHalt, 操作紀錄是否結束於「被權限拒絕」之工具呼叫(規則 R-RUN-7): opencode 於 ask 被自動拒絕後可能結束整個工作階段(離開碼 0),
//  節點因此未交付成品; 派工設定已帶 continue_loop_on_deny, 本函式供失敗訊息標明原因(該設定失效時之警示)
//  判法: 最後一步(最後一個 step_start 之後)內有被拒之工具呼叫 —— 同一步並行發出數個呼叫時, 被拒者不一定是最後一筆;
//    紀錄無 step_start 者(其他格式)只看最後一筆工具呼叫。被拒後續做而以文字收尾者, 最後一步無工具呼叫 ⇒ null
//  回 null | { tool, target }(target = 該呼叫之路徑或指令, 前 120 字元)
let deniedHalt = (logFile) => {
    let cur = []
    let sawStep = false
    try {
        for (let line of fs.readFileSync(logFile, 'utf8').split('\n')) {
            if (!line.startsWith('{')) {
                continue
            }
            let j
            try {
                j = JSON.parse(line)
            }
            catch (err) {
                continue
            }
            if (j.type === 'step_start' || (j.part && j.part.type === 'step-start')) {
                sawStep = true
                cur = []
            }
            for (let c of callsOf(j)) {
                cur.push(c)
            }
        }
    }
    catch (err) {
        return null
    }
    let last = (sawStep ? cur : cur.slice(-1)).find((c) => c.status === 'error' && typeof c.error === 'string' && /rejected permission|permission (was |is )?denied|not permitted/i.test(c.error))
    if (!last) {
        return null
    }
    let inp = last.input || {}
    return { tool: last.tool, target: String(inp.filePath || inp.file_path || inp.path || inp.command || '').replace(/\s+/g, ' ').slice(0, 120) }
}

let escRe = (s) => String(s).replace(/[.*+?^${}()|[\]\\]/g, '\\$&')
//extraItemsOf, 專案另加之禁區字樣(陣列; 各項 { pattern, label })之檢測與轉換: 回 [[RegExp, 標籤]]; 未給(undefined 或 null)回 []; 非陣列、非一般物件之項、pattern 非 RegExp 或非空字串(字串者須為合法之正規式)、label 非空字串(去頭尾空白後)⇒ 拋 invalid <where>[i]: must be …(不帶輸入值)
let extraItemsOf = (items, where) => {
    if (items === undefined || items === null) {
        return []
    }
    if (!Array.isArray(items)) {
        throw badInput(where, 'an array of { pattern, label }')
    }
    return items.map((x, i) => {
        let ok = isPlain(x) && typeof x.label === 'string' && x.label.trim() !== '' && (x.pattern instanceof RegExp || (typeof x.pattern === 'string' && x.pattern !== ''))
        let re = null
        if (ok) {
            try {
                re = x.pattern instanceof RegExp ? x.pattern : new RegExp(x.pattern, 'i')
            }
            catch (err) {
                ok = false
            }
        }
        if (!ok) {
            throw badInput(`${where}[${i}]`, 'an object { pattern, label } with a non-empty string label and a pattern that is a valid regular expression (non-empty string or RegExp)')
        }
        return [re, x.label.trim()]
    })
}
/**
 * 設定專案另加之禁區字樣(模組層狀態,供 forbiddenPats 使用);未給(undefined 或 null)= 清空;給了而無效(非陣列、項非一般物件、無非空 label、pattern 非 RegExp 或非空字串、非法之正規式)⇒ 拋 invalid items[i]: must be …(固定訊息、不帶輸入值;模組層狀態不變)
 *
 * @param {Array<Object>} items 輸入禁區項陣列,每項 { pattern, label },pattern 為 RegExp(沿用其本身之旗標)或正則之原始字串(以旗標 i 建立,不分大小寫)
 * @returns {Number} 回傳生效之禁區項數
 * @example
 * let n = setGuardExtra([{ pattern: 'secret', label: '機密字樣' }])
 */
let setGuardExtra = (items) => {
    //禁區字樣為安全閘(R-EV-54 ①): 給了而無效 ⇒ 拒絕(先於任何改變; 固定訊息、不帶輸入值); 未給(undefined 或 null)= 清空
    //  原: 非陣列視為空、缺 pattern 或 label 之項靜默略過 ⇒ 該禁區字樣失效而不報錯; 非法正則於 `new RegExp` 拋原生 SyntaxError
    let next = extraItemsOf(items, 'items')
    GUARD_EXTRA = next
    return GUARD_EXTRA.length
}
/**
 * 由設定導出專案另加之禁區項:策略庫位於專案夾之外時,其所在之資料夾名列為「策略庫專案」;另加 settings.guard.forbidden
 *
 * source.libraryRoot 給了而不是非空字串、guard 給了而不是一般物件、guard.forbidden 給了而無效(同 setGuardExtra 之項)者拋 invalid …(固定訊息、不帶輸入值)
 *
 * @param {Object} st 輸入設定物件,讀取 st.source.libraryRoot、st.FD_ROOT、st.guard.forbidden
 * @returns {Array<Object>} 回傳禁區項陣列,每項 { pattern(正則字串), label },可交給 setGuardExtra
 * @example
 * setGuardExtra(guardExtraOf(st))
 */
let guardExtraOf = (st) => {
    let out = []
    //禁區字樣之來源為安全閘(R-EV-54 ①): source.libraryRoot 給了而不是非空字串、guard 給了而不是一般物件、guard.forbidden 給了而不是 [{ pattern(合法之正規式字串), label(非空字串) }] ⇒ 拋 invalid …(固定訊息、不帶輸入值; 入口以 inputMsgOf 辨識而印 ERROR、離開碼 2)
    //  原: 靜默略過 ⇒ 該禁區失效而不報錯(缺 label、非陣列、非字串之 pattern 略過; 非法正則於其後之 `new RegExp` 拋原生 SyntaxError); 未給(undefined 或 null)= 無
    let srcLib = st && st.source !== undefined && st.source !== null ? st.source.libraryRoot : undefined
    if (srcLib !== undefined && srcLib !== null && (typeof srcLib !== 'string' || srcLib === '')) {
        throw badInput('source.libraryRoot', 'a non-empty string')
    }
    let lib = typeof srcLib === 'string' ? srcLib : ''
    let gd = st ? st.guard : undefined
    if (gd !== undefined && gd !== null && !isPlain(gd)) {
        throw badInput('guard', 'an object')
    }
    let more = []
    if (gd !== undefined && gd !== null && gd.forbidden !== undefined && gd.forbidden !== null) {
        extraItemsOf(gd.forbidden, 'guard.forbidden') //檢測(不合者拋); 輸出仍為原項之 { pattern, label }(setGuardExtra 再轉為正規式)
        more = gd.forbidden
    }
    let root = st && typeof st.FD_ROOT === 'string' ? st.FD_ROOT : null
    if (lib) {
        let abs = path.resolve(root || '.', lib)
        let outside = !root || path.relative(root, abs).startsWith('..') || path.isAbsolute(path.relative(root, abs))
        if (outside) {
            let segs = abs.split(/[\\/]+/).filter((s) => s && !/^[A-Za-z]:$/.test(s))
            let generic = /^(data|library|lib|strategies|strategy|db|store|out|output)$/i
            let name = segs.reverse().find((s) => !generic.test(s))
            if (name && name.length >= 4) {
                out.push({ pattern: escRe(name), label: '策略庫專案' })
            }
        }
    }
    for (let x of more) {
        out.push({ pattern: x.pattern, label: x.label.trim() })
    }
    return out
}

 * @param {Boolean} [nodeRules=false] 輸入是否套用節點專用規則,預設false
 * @returns {Array<Array>} 回傳陣列,每項 [RegExp, 名稱]
 * @example
 * let pats = forbiddenPats('./private', true)
 */
let forbiddenPats = (privateRoot, nodeRules = false) => [
    [/g_backtest/i, '回測期工具 g_backtest'],
    //迴圈專用之環境變數: ARSI_HARNESS(授權迴圈之子行程)與 ARSI_AGENT_ROLE(派工器交給 agent 之角色; 規則 R-RUN-11)—— 同一條、同一標籤; 搜尋樣式內者照 stripSearchPatterns(做夢)不算存取
    [/ARSI_HARNESS|ARSI_AGENT_ROLE/, '迴圈專用之環境變數'],
    [new RegExp(escRe(path.basename(privateRoot)), 'i'), '私有根'],
    [/ledger[\\/]+(train|full|eval|latest)/i, '帳本檔'],
    [/(^|[\s"'=(,])\.?[\\/]?data[\\/]/i, '專案之 data 夾'],
    ...GUARD_EXTRA,

    [/archive_\d{8}_regime\d*/i, '舊制之備份'],
    //  同一條、同一標籤: 迴圈驗證之程式(src/domain/verify*、backtest*)與庫況體檢之程式(src/domain/libStats.mjs、src/tools/libCheck.mjs; 規則 R-EV-48、R-BT-10 ④)
    //  —— 後者之輸出含評估期內全庫之統計(事後資訊), 節點不得讀取或匯入其程式; 執行 tools/libCheck.mjs 另由「執行迴圈之寫檔入口」(R-BT-7 ①)與入口之拒跑(R-RUN-11)擋
    //  另涵蓋套件裝於 node_modules/<套件>/ 下之建置產物與文件(dist、docs、sgud、test、README: 內含判定之程式與門檻之值; 同一個標籤; 規則 R-BT-10 ⑦)與專案內之 dist/<套件名>*(建置產物之檔名)
    ...(nodeRules ? [[new RegExp(`(?:lib|src)[\\\\/]+(?:domain[\\\\/]+(?:backtest|verify|libStats)|tools[\\\\/]+libCheck)|node_modules[\\\\/]+${escRe(PKG_NAME)}[\\\\/]+(?:dist|docs|sgud|test|README)|dist[\\\\/]+${escRe(PKG_NAME)}`, 'i'), ZONE_LABEL]] : []),
]
//節點專用、只套用於工具呼叫之指令與路徑(不套用於檔案內容與共用工具區之逐檔檢查 —— 程式內容中之「lib」「src」可能只是變數名):
//  ①以目錄或萬用字元涵蓋迴圈驗證之程式者(…/domain/*.mjs、rg … <程式夾>/domain、ls <程式夾>/domain/、grep -r … <程式夾>、列出套件夾或其 src): 其輸出會含 verify.mjs 之內容

let NODE_CMD_KEYS = ['command', 'filePath', 'file_path', 'path', 'pattern', 'glob', 'include']
let NODE_CMD_PATS = [
    [new RegExp([
        //<程式夾>/domain/*
        '(?:lib|src)[\\\\/]+domain[\\\\/]*\\*',
        //<程式夾>/domain 整個目錄
        '(^|[\\s"\'=(,\\\\/])\\.?[\\\\/]?(?:lib|src)[\\\\/]+domain[\\\\/]?(?=$|[\\s"\'|;&)])',
        //以列出 / 搜尋類指令涵蓋專案根之程式夾本身
        '\\b(ls|dir|find|tree|du|grep|egrep|rg|findstr)\\b[^|;&\\n]*[\\s"\']\\.?[\\\\/]?(?:lib|src)[\\\\/]?(?=$|[\\s"\'|;&)])',
        //套件夾或其 src 本身(含 …/src/*)
        `${escRe(PKG_NAME)}(?:[\\\\/]+src)?[\\\\/]*\\*?(?=$|[\\s"'|;&)])`,
    ].join('|'), 'i'), '以目錄或萬用字元涵蓋迴圈驗證之程式(程式夾本身、其 domain 夾)'],
    [/rsi[\\/]+logs/i, '迴圈之日誌 rsi/logs'],
]
let nodeCmdHits = (k, v) => NODE_CMD_PATS.filter(([re], i) => re.test(v) || (i === 0 && k !== 'command' && /^\.?[\\/]?(?:lib|src)[\\/]?$/.test(v))).map(([, label]) => label)
//contentHits, 一段程式內容(先去註解)所含之禁區字樣之名稱清單(共用工具區之逐檔檢查用; 與操作掃描同一組字樣)
let contentHits = (src, privateRoot, nodeRules = true) => {

}
//inlinePartsOf, 把指令內 node -e / --eval / -p / --print 之內嵌腳本中「比對內容用」之字樣分出來: 回 { text(去掉這些字樣後之指令), removed(被去掉之片段) }
//  做夢 agent 會以內嵌腳本掃描日誌自檢(例: node -e "... { k1: /g_backtest/g } ..."): 比對式內之字樣不是存取
//  只去三種: ①正規式字面值 ②比對函數(includes、indexOf、lastIndexOf、startsWith、endsWith、match、matchAll、search、split、replace、replaceAll、RegExp)之字串引數 ③腳本內之註解
//  **其他字串照留**(fs.readFileSync('…')、spawn('…') 之路徑是實際之存取; 陣列或變數內之字樣無從判定用途, 寧可多報); 內嵌腳本以外之引號內容照查(被引號包住之路徑仍是路徑)
//  **被去掉之片段內含禁區字樣者不得不列**: scanTrace 改列為待核對(inlineNotes; 不計為觸及禁區)
//  **不變式**: 原指令(只去搜尋樣式)命中之每個禁區字樣, 必在「觸及禁區」或「待核對」之一 —— 內嵌腳本之語法無從完全判定, 故 scanTrace 以原指令之比對結果為上界, 不依賴片段之切法
//  只認 node 之選項位置(腳本路徑之前)之 -e / --eval / -p / --print: node x.mjs -e "…" 之 -e 是腳本自己之選項, 其引號內容不是內嵌腳本
//  帶引數之選項, 其引數不得以 - 起首(解法唯一; 否則畸形之指令會使判定式指數回溯)
let INLINE_NODE = /(\bnode(?:\.exe)?["']?(?:\s+(?:(?:-r|--require|--import|--loader|--experimental-loader)\s+(?!-)\S+|(?!-e\s|-p\s|-pe\s|--eval\s|--print\s)-\S*))*\s+(?:-e|--eval|-p|--print|-pe)\s+)("(?:[^"\\]|\\[\s\S])*"|'[^']*')/gi
let MATCH_CALL = /(?:\.(?:includes|indexOf|lastIndexOf|startsWith|endsWith|match|matchAll|search|split|replace|replaceAll)|\bRegExp)\s*\(\s*$/
let REGEX_PREV = /(?:^|[(,=:[!&|?{};\n]|=>|\b(?:return|typeof|case|in|of|delete|void|instanceof|new|yield|await|else|do))\s*$/
let maskInlineScript = (s) => {
    let n = s.length
    let out = ''
    let removed = []
    let i = 0
    while (i < n) {
        let c = s[i]
        //字串以外之反斜線: 連同下一字元原樣保留(\/\/、\/* 不是註解之開頭)
        if (c === '\\') {
            out += s.slice(i, i + 2)
            i += 2
            continue
        }
        if (c === '/' && s[i + 1] === '/') {
            let e = i
            while (e < n && s[e] !== '\n') {
                e++
            }
            removed.push(s.slice(i, e))
            i = e
            continue
        }
        if (c === '/' && s[i + 1] === '*') {
            let e = s.indexOf('*/', i + 2)
            let end = e < 0 ? n : e + 2
            removed.push(s.slice(i, end))
            i = end
            continue
        }
        if (c === '\'' || c === '"' || c === '`') {
            let j = i + 1
            while (j < n && s[j] !== c) {
                j += s[j] === '\\' ? 2 : 1
            }
            if (MATCH_CALL.test(out.slice(-32))) {
                removed.push(s.slice(i, j + 1))
                out += `${c}${c}`
            }
            else {
                out += s.slice(i, j + 1)
            }
            i = j + 1
            continue
        }
        if (c === '/' && REGEX_PREV.test(out.slice(-12))) {
            let j = i + 1
            let inClass = false
            let closed = false
            while (j < n && s[j] !== '\n') {
                if (s[j] === '\\') {
                    j += 2
                    continue
                }
                if (s[j] === '[') {
                    inClass = true
                }
                else if (s[j] === ']') {
                    inClass = false
                }
                else if (s[j] === '/' && !inClass) {
                    closed = true
                    break
                }
                j++
            }
            if (closed) {
                j++
                while (j < n && /[a-z]/i.test(s[j])) {
                    j++
                }
                removed.push(s.slice(i, j))
                out += '/ /'
                i = j
                continue
            }
        }
        out += c
        i++
    }
    return { text: out, removed }
}
let inlinePartsOf = (cmd) => {
    let removed = []
    let text = String(cmd).replace(INLINE_NODE, (m, head, q) => {
        let body = q.slice(1, -1)
        //外層為雙引號者先還原殼層之跳脫(\" \\ \$ \`)
        let r = maskInlineScript(q[0] === '"' ? body.replace(/\\(["\\$`])/g, '$1') : body)
        removed.push(...r.removed)
        //外層為雙引號者: 分析時還原之 \" 於重組時跳脫回去(原本文內不可能有裸 "; 否則重組後之引號配對錯位, 其後之 stripSearchPatterns 〔去 grep 之樣式等〕失準 ⇒ 多報)
        return head + q[0] + (q[0] === '"' ? r.text.replace(/"/g, '\\"') : r.text) + q[0]
    })
    return { text, removed }
}
//stripInlinePatterns, 去掉指令內內嵌腳本之比對字樣後之指令(inlinePartsOf 之 text)
let stripInlinePatterns = (cmd) => inlinePartsOf(cmd).text
/**

let scanTrace = (opt) => {

    let o = objReq(opt, 'opt')

    }
    let planOn = planRule === null ? nodeRules : planRule === true

    let rlOn = nodeRules && typeof stateRoot === 'string' && stateRoot !== ''

    //  listBase = 專案根(相對路徑之解析與工作目錄之核對用; 規則 R-BT-6); cwdMoved = 此前之指令曾切換目錄(其後未帶工作目錄之欄者, 工作目錄不確定)
    let reListRoot = null
    let listBase = null
    let cwdMoved = false
    //zone, 節點之「先切換目錄再以相對路徑」之工作目錄(相對專案根; '' = 專案根; null = 不確定)與 pushd 之堆疊, 跨工具呼叫沿用(規則 R-BT-10 ⑦)
    let zone = { dir: '', tail: '', stack: [] }

        }
    }
    //cwdKnownOf, 該次工具呼叫之工作目錄是否確定為專案根(規則 R-BT-6): 帶工作目錄之欄(workdir / cwd)者以之為準; 未帶者須此前之指令未切換目錄
    let cwdKnownOf = (inp) => {
        let wd = inp.workdir !== undefined ? inp.workdir : inp.cwd
        return wd === undefined ? !cwdMoved : typeof wd === 'string' && path.relative(path.resolve(listBase), absOf(listBase, wd)) === ''
    }
    let hits = new Map()
    let fileMap = new Map()
    let inlineMap = new Map()

    let hasPat = (re, s) => {
        re.lastIndex = 0
        return re.test(s)
    }
    let test = (s, where, to = hits) => {
        for (let [re, label] of pats) {
            if (hasPat(re, s)) {
                let k = `${label}(${where})`
                if (!to.has(k)) {
                    to.set(k, String(s).replace(/\s+/g, ' ').slice(0, 160))
                }
            }
        }
    }
    let note = null
    let nEvals = null
    //規劃先於評估(nodeRules): 第幾個工具呼叫首次寫入節點文件、第幾個首次以完整模式評估自己的實驗檔(起點之複本 base_pick.mjs、參考節點成品之複本 ref_<id 前 8 碼>.mjs 與範例不算: 重現既有成品不是選版)
    let firstDoc = null
    let firstFull = null

    if (fs.existsSync(fpLog)) {

        nEvals = 0
        for (let line of fs.readFileSync(fpLog, 'utf8').split('\n')) {
            if (!line.startsWith('{')) {

            }

            }
            catch (err) {
                continue
            }

                nEv++

                let denied = status === 'error' && typeof error === 'string' && /rejected permission|permission (was |is )?denied|not permitted/i.test(error)

                for (let [k, v] of Object.entries(inp)) {
                    //寫檔工具之 content 是檔案內容(其程式檔另由下方掃), 不當成指令或路徑

                        continue

                    //做夢(非節點規則): 搜尋樣式與指令之說明文字不是存取(以禁區字樣為搜尋目標之自檢); 被搜尋之路徑照查。
                    //  節點搜尋禁區字樣屬探查, 照列
                    if (!nodeRules) {
                        if (k === 'description' || (isSearchTool && /^(pattern|query|regex)$/i.test(k))) {
                            continue
                        }
                        if (k === 'command') {
                            let ip = inlinePartsOf(v)
                            let kept = stripSearchPatterns(ip.text)
                            test(kept, '指令或路徑')
                            for (let frag of ip.removed) {
                                test(frag, '內嵌腳本之比對式', inlineMap)
                            }
                            //以原指令之比對結果為上界: 原指令命中而去掉比對式後未命中之字樣(例: 字樣跨越被去掉之片段之邊界)一律列待核對
                            let raw = stripSearchPatterns(v)
                            for (let [re, label] of pats) {
                                let kk = `${label}(內嵌腳本之比對式)`
                                if (hasPat(re, raw) && !hasPat(re, kept) && !inlineMap.has(kk)) {

                                }
                            }

                            if (!hits.has(kk)) {
                                hits.set(kk, String(v).replace(/\s+/g, ' ').slice(0, 160))
                            }
                        }
                    }
                }

                if (planOn) {

    let planAfterEval = planOn && firstDoc !== null && firstFull !== null && firstDoc > firstFull ? { docCall: firstDoc, evalCall: firstFull } : null

//evalLines, 工作夾之 .evals.log 之各列(原文); 無此檔回 null
//  until(選配; ISO 時刻): 只取 at ≤ until 之列 —— 節點之 agent 結束之後才寫入之列不是該節點之操作
//    (做夢或主控日後直接評估該工作夾內之模組時, 評估器照樣記於此檔); 讀不出 at 之列照留
let evalLines = (workDir, until = null) => {
    let fp = path.join(workDir, '.evals.log')
    if (!fs.existsSync(fp)) {
        return null
    }
    let ls = fs.readFileSync(fp, 'utf8').split('\n')
    if (typeof until !== 'string' || !until) {
        return ls
    }
    return ls.filter((l) => {
        let m = /"at":"([^"]+)"/.exec(l)
        return !m || m[1] <= until
    })
}

//evalsLogged, 評估器自己記下之評估次數(工作夾之 .evals.log; 每次 g_eval.mjs 執行一列); 無此檔回 null
let evalsLogged = (workDir, until = null) => {
    let ls = evalLines(workDir, until)
    if (!ls) {
        return null
    }
    return ls.filter((l) => l.trim().startsWith('{')).length
}

//evalsDistinct, 完整評估之次數與其相異版本數(工作夾之 .evals.log; 只計成功之 full 列); 無此檔回 null
//  nFull = 完整評估之次數; nFullSel = 其相異 selHash 數(相異之選集序列; 只差 tag、只加 --ref 或同一內容重跑者不算新版本);
//  nFullMod = 其相異模組雜湊數(相異之檔案內容; 舊紀錄無 h 者不計)
//  「試得越多, 其中最佳者越可能只是運氣」以相異 selHash 數衡量; 總次數照記(evalsLogged)
let evalsDistinct = (workDir, until = null) => {
    let ls = evalLines(workDir, until)
    if (!ls) {
        return null
    }
    let nFull = 0
    let sel = new Set()
    let mod = new Set()
    for (let l of ls) {
        if (!l.trim().startsWith('{')) {
            continue
        }
        let j
        try {
            j = JSON.parse(l)
        }
        catch (err) {
            continue
        }
        //只計評估期(回測期)之完整評估; --period train 之診斷不是版本之評估
        if (j.ok !== true || j.mode !== 'full' || (j.period && j.period !== 'backtest')) {
            continue
        }
        nFull++
        if (typeof j.selHash === 'string') {
            sel.add(j.selHash)
        }
        if (typeof j.h === 'string') {
            mod.add(j.h)
        }
    }
    return { nFull, nFullSel: sel.size, nFullMod: mod.size }
}

//noLogTried, .evals.log 內標記 noLogIgnored 之列數(非迴圈之呼叫者帶 --no-log 而被照記者)
let noLogTried = (workDir, until = null) => {
    let ls = evalLines(workDir, until)
    if (!ls) {
        return 0
    }
    return ls.filter((l) => l.includes('"noLogIgnored":true')).length
}

//lostVersions, 已以完整模式評估過、而工作夾內現已無任何檔之內容與之相同之版本(評估後覆寫了該檔 ⇒ 該次實驗無檔可重現; 任務書「一實驗一檔」)
//  依 .evals.log 各列之 h(模組內容之雜湊; g_eval.mjs 所記)與工作夾內現有程式檔之雜湊比對; 回 [{ pick, n }](依檔名彙整; 舊紀錄無 h 者略過)
let lostVersions = (workDir, until = null) => {
    let ls = evalLines(workDir, until)
    if (!ls) {
        return []
    }
    let now = new Set()
    for (let f of fs.readdirSync(workDir)) {
        let p = path.join(workDir, f)
        if (/\.(mjs|js|cjs)$/i.test(f) && fs.statSync(p).isFile()) {
            now.add(crypto.createHash('sha1').update(fs.readFileSync(p)).digest('hex').slice(0, 12))
        }
    }
    let lost = new Map()
    for (let l of ls) {
        if (!l.trim().startsWith('{')) {
            continue
        }
        let j
        try {
            j = JSON.parse(l)
        }
        catch (err) {
            continue
        }
        if (j.ok === true && j.mode === 'full' && !(j.period && j.period !== 'backtest') && typeof j.h === 'string' && !now.has(j.h)) {
            if (!lost.has(j.pick)) {
                lost.set(j.pick, new Set())
            }
            lost.get(j.pick).add(j.h)
        }
    }
    return [...lost.entries()].map(([pick, hs]) => ({ pick, n: hs.size }))
}

//isHalfOf, 成品自述為半庫版(meta.params.half 為 0 或 1; 共用工具區之 *_half.mjs 樣板): 迴圈之驗證(makeVerify 取評估器回傳之 pickMeta)與補算(rescanSideFlow 取節點工作夾內重放結果檔之 pickMeta)用同一判定,
//  使補算重產之補充列與迴圈驗證者一致(「半庫版」之說明列); 無 pickMeta 或無此自述者為 false
let isHalfOf = (pickMeta) => !!(pickMeta && pickMeta.params && (pickMeta.params.half === 0 || pickMeta.params.half === 1))
//notesOf, 節點文件之補充說明列(領域指標; 全部為評估期 = 回測期之數字)
//  seedS(選配): 迴圈換亂數種子重放所得之 S(主種子以外者)
//  train(選配): 迴圈以訓練期之診斷重放所得之選集組成 { longRatio, tags }(只記組成, 不記績效 —— 訓練期之績效不是目標)
//下單流量偏一側之機械判讀(規則 R-REC-15)
//  shortShareOf(metrics): 做空單 ÷ 全部下單(各相位平均之單數; 無下單或無此欄 ⇒ null)
//  sideFlowIssue(回測期之值, 訓練期診斷之值, cfg): 兩期皆 ≥ 全體隨機基準之 hi 倍(或皆 ≤ lo 倍)⇒ 回問題字串, 否則 null
//    基準 = 中性隨機 40 隻(templates/neutral_random40.mjs)之同一比例: 回測期約 0.16、訓練期(quick 診斷)約 0.22; 可由 settings.score.sideFlow 覆寫
//    例: 「做多取甲規則、做空取乙規則」之分側組合, 席次之做多比例兩期皆正常(固定側別之檢查不列), 但兩期之下單皆大半為做空單 ——
//    兩側之活躍度不對稱使下單集中於做空側, 等於把回測期之方向寫進模組; S、階梯與同側同活躍度虛無都照樣保留它, 故另設此檢查(不否決, 由做夢審計)
//fixedSideOf, 固定側別之判定(迴圈檢查之「固定側別」與側別曝險之判定〔verifyExposure.mjs; 規則 R-REC-48〕共用此單一定義): a = 訓練期之選集之做多比例, b = 回測期者, sizeMax = 回測期之選集之最大席數
//  選集非空(sizeMax > 0)且兩期皆 < 0.1 或皆 > 0.9 ⇒ true
let fixedSideOf = (a, b, sizeMax) => sizeMax > 0 && ((a < 0.1 && b < 0.1) || (a > 0.9 && b > 0.9))
let SIDE_FLOW = { baseShort: { backtest: 0.16, train: 0.22 }, hi: 2.5, lo: 0.4 }
let shortShareOf = (m) => {
    let b = m && m.op && m.op.bySide
    if (!b || !b.long || !b.short || typeof b.long.orders !== 'number' || typeof b.short.orders !== 'number' || b.long.orders + b.short.orders <= 0) {
        return null
    }
    return Math.round((b.short.orders / (b.long.orders + b.short.orders)) * 1000) / 1000
}
//sideFlowCfgOf, 判讀所用之設定(內建值 SIDE_FLOW 與 settings.score.sideFlow 之合併); 看板之說明(domain.mjs)取同一份基準(規則 R-REC-36: 不寫死某一策略庫之實測值)
//  取值(排程第 612 項 ②; R-EV-57): 「下單流量偏一側」為提示用(不否決、不入階梯與評估之數字)⇒ 依跨套件規格 K3: 給了而無效者取內建值(不拋; 該鍵單獨退回、其餘之鍵照給);
//  數字字串轉為數值後才使用(K4、K7: 檢測之值即使用之值)。取值之基元為評估器之 cfgNumOf(同 evalScoreOf 之單一來源: 數字與十進位數字字串、有限數)。
//  值域: hi(偏做空之倍數)> 0、lo(偏做多之倍數)≥ 0、baseShort.*(全體隨機之做空單比例)> 0 且 ≤ 1(為 0 者使「≥ hi 倍」恆成立);sideFlow、baseShort 非物件者視為未給;未知之鍵不入回傳
//  原: 以原值合併而不檢 —— null 覆蓋內建值(hi 為 null ⇒ 門檻 0 ⇒ 每個成品皆被標「下單偏做空」)、非數字使判讀恆不成立、字串之基準原樣寫入節點文件、字串或陣列被展開
let SIDE_FLOW_OK = { hi: (x) => x > 0, lo: (x) => x >= 0, share: (x) => x > 0 && x <= 1 }
let sideNumOf = (v, def, ok) => {
    try {
        let x = cfgNumOf(v, def, 'sideFlow', 'num')
        return ok(x) ? x : def
    }
    catch (err) {
        return def
    }
}
let sideObjOf = (o) => (o !== null && typeof o === 'object' ? o : {})
let sideFlowCfgOf = (cfg = null) => {
    let c = sideObjOf(cfg)
    let b = sideObjOf(c.baseShort)
    return {
        baseShort: { backtest: sideNumOf(b.backtest, SIDE_FLOW.baseShort.backtest, SIDE_FLOW_OK.share), train: sideNumOf(b.train, SIDE_FLOW.baseShort.train, SIDE_FLOW_OK.share) },
        hi: sideNumOf(c.hi, SIDE_FLOW.hi, SIDE_FLOW_OK.hi),
        lo: sideNumOf(c.lo, SIDE_FLOW.lo, SIDE_FLOW_OK.lo),
    }
}
let sideFlowIssue = (bt, tr, cfg = null) => {
    if (typeof bt !== 'number' || typeof tr !== 'number') {
        return null
    }
    let c = sideFlowCfgOf(cfg)
    let up = bt >= c.baseShort.backtest * c.hi && tr >= c.baseShort.train * c.hi
    let dn = bt <= c.baseShort.backtest * c.lo && tr <= c.baseShort.train * c.lo
    if (!up && !dn) {
        return null
    }
    return `下單流量偏一側: 做空單之比例 訓練期 ${tr} / 回測期 ${bt}(全體隨機約 ${c.baseShort.train} / ${c.baseShort.backtest};兩期皆${up ? `≥ ${c.hi} 倍` : `≤ ${c.lo} 倍`} ⇒ 下單偏${up ? '做空' : '做多'}側屬組成,回測期之方向可能經由兩側活躍度之不對稱寫進模組;由審計判定)`
}
//sideFlowIssueOfSt, 迴圈檢查之「下單流量偏一側」是否列為問題(規則 R-REC-15、R-REC-55 ③; 迴圈之驗證〔makeVerify〕與補算〔domain.mjs 之 rescanSideFlow〕共用此單一定義):
//  側別曝險之判定關閉者(settings.rsi.exposure === false; 與 verifyExposure.mjs 之 exposureCfgOf 回 null 同一條件)回 null —— 多空兩側各自運行, 某時段只有一側下單是各策略自身之條件之結果, 不列為問題(讀數照留:
//  節點文件之「下單之做空比例」一列與樹之 raw.train.shortShare); 其餘 = sideFlowIssue(門檻取 settings.score.sideFlow, 經 sideFlowIssue 內之 sideFlowCfgOf 取值〔R-EV-57 ②、K3 單一來源: 無效之值取內建值、數字字串轉為數值〕, 本函數不另讀設定; 與改前逐字相同); 側別曝險之判定本身(verifyExposure.mjs)直接用 sideFlowIssue, 不經本函數
let sideFlowIssueOfSt = (bt, tr, st) => (st && st.rsi && st.rsi.exposure === false ? null : sideFlowIssue(bt, tr, st && st.score && st.score.sideFlow))

//  replays(選配): 各次重放之 { S, comp(組成對照之超額), recent(近期活躍度之組成對照之超額), [ext](一併載入之擴充診斷之摘要; 規則 R-REC-42) }(規則 R-EV-24; 有給時該列一併列出; 帶 ext 者另列各診斷之看板值之多抽之均)
/**
 * 產生節點文件之補充說明列(領域指標;皆為回測期之數字)
 *
 * @param {Object} m 輸入評估之 metrics,讀取 period、op、nullSideAct、ext
 * @param {Array<Number>|null} [seedS=null] 輸入換亂數種子重放所得之 S 陣列,預設null
 * @param {Object|null} [train=null] 輸入訓練期診斷之選集組成 { longRatio, tags },預設null
 * @param {Array<Object>|null} [replays=null] 輸入各次重放之 { S, comp, recent, [ext] }(ext 為一併載入之擴充診斷之摘要;帶者另列各診斷之看板值之多抽之均),預設null
 * @param {Boolean} [half=false] 輸入是否為半庫版,預設false
 * @param {Object|null} [sideFlow=null] 輸入「下單流量偏一側」之設定(settings.score.sideFlow;「下單之做空比例」一列所附之全體隨機基準取自其與內建值之合併),預設null
 * @returns {Array<String>} 回傳說明列字串陣列
 * @example
 * let lines = notesOf(metrics, [1.1, 1.3], null, null, false)
 */
let notesOf = (m, seedS = null, train = null, replays = null, half = false, sideFlow = null) => {
    let f1 = (x) => (typeof x === 'number' ? x.toFixed(1) : '—')
    let out = []
    if (m.period && m.period.evalFrom) {
        out.push(`評估期間(回測期): ${String(m.period.evalFrom).slice(0, 10)} 起至 ${String(m.period.endTime).slice(0, 10)};首個計入之決策 ${String(m.period.firstEvalDecision || '').slice(0, 10)};各相位決策 ${m.op.nDecisions.join(' / ')} 次`)
    }
    out.push(`各相位之等效年化(%): ${m.op.eqYr.byPhase.map((v) => f1(v)).join(' / ')}(為正之相位 ${m.op.eqYr.byPhase.filter((v) => v > 0).length} / ${m.op.eqYr.byPhase.length})`)
    if (Array.isArray(seedS) && seedS.length) {
        //半庫版: 換同分次序即換一種切法 ⇒ 重放之 S 隨成員擺動, 不是「入選者由同分決定」
        if (half) {
            out.push('半庫版(成品之 meta.params.half 為 0 或 1; 以 ctx.meta.tie 之雜湊把策略庫切半): 下列重放 = 另兩種切法, S 隨成員擺動, 不作「分數不穩」之判讀; 判內容以正本與重放之組成對照之均')
        }
        let hasRp = Array.isArray(replays) && replays.length === seedS.length && replays.some((x) => x && (typeof x.comp === 'number' || typeof x.recent === 'number'))
        let rp = hasRp ? `;組成對照之超額 ${replays.map((x) => f1(x.comp)).join(' / ')}${m.nullSideAct && typeof m.nullSideAct.excess === 'number' ? `(正本 ${f1(m.nullSideAct.excess)})` : ''};近期活躍度之組成對照之超額 ${replays.map((x) => f1(x.recent)).join(' / ')}${recentDiagOf(m) ? `(正本 ${f1(recentDiagOf(m).excess)})` : ''}` : ''
        //各擴充診斷之看板值之多抽之均(規則 R-REC-42; 重放帶其值者才列, 否則與改前相同)
        let xm = extBoardMeansOf(m, replays)
        let xmText = xm.length ? `;擴充診斷之看板值之均(正本與各次重放):${xm.map((x) => `${x.name} ${x.value}(${x.n} 抽)`).join('、')}` : ''
        out.push(`換亂數種子與同分次序之重放(迴圈${hasRp ? ';組成對照 32 抽' : ';不含虛無'}): S ${seedS.map((v) => f1(v)).join(' / ')}${rp}${xmText} —— 每次同時換 ctx.rand 之種子與同分次序(ctx.meta.tie、同一進場棒之訊號先後);含隨機抽取、或主鍵大量同分(入選者實際由 tie 決定)之成品其結果隨之波動;各次之 S 與主分數相差很大者,主分數不可靠${hasRp ? ';**判內容以正本與各次重放之超額之平均為準,不以單抽**' : ''}`)
    }
    if (m.null) {
        out.push(`同劑量虛無(${m.null.draws} 抽;同決策日、同選集大小、同執行規則之隨機挑): 均 ${f1(m.null.mean)}%、sd ${f1(m.null.sd)};本成品之超額 ${f1(m.null.excess)} 個百分點(z ${m.null.z}、p ${m.null.p})`)
    }
    if (m.nullVol) {
        out.push(`同訊號量虛無(選集之每隻訊號數為全體隨機之 ${m.nullVol.ratio} 倍 ⇒ 改與隨機 ${f1(m.nullVol.sizeMean)} 隻、訊號量相當者比較): 均 ${f1(m.nullVol.mean)}%、sd ${f1(m.nullVol.sd)};超額 ${f1(m.nullVol.excess)} 個百分點(z ${m.nullVol.z}、p ${m.nullVol.p})—— 等效年化受訊號量之影響很大,不匹配訊號量會把較活躍或較安靜誤當成內容${m.nullAct ? ';本成品另有同活躍度虛無,判讀臂之內容以該列為準' : ''}`)
    }
    if (m.nullAct) {
        out.push(`同活躍度虛無(隨機選集之活躍度分層與成品相同:各決策依決策日以前之累計進場數分 ${m.nullAct.bins} 層、各層抽同樣隻數): 均 ${f1(m.nullAct.mean)}%、sd ${f1(m.nullAct.sd)};超額 ${f1(m.nullAct.excess)} 個百分點(z ${m.nullAct.z}、p ${m.nullAct.p})—— 選集比隨機安靜或活躍時,同訊號量之超額含「活躍度層」本身之差異,扣掉後剩多少看此列`)
    }
    //各三等分段之均(與成品之三等分並列): 成品之超額是否只集中於某一段
    let th3 = (x) => (Array.isArray(x.thirdsMean) && Array.isArray(m.op.thirds) && m.op.thirds.length === 3 ? `;三等分之均 ${x.thirdsMean.map((v) => f1(v)).join(' / ')}(成品 ${m.op.thirds.map((g) => f1(g.eqYr.mean)).join(' / ')})` : '')
    if (m.nullSideAct) {
        out.push(`同側同活躍度虛無(組成對照:把選集之每一隻換成同方向、決策日以前累計進場數之名次 ±${m.nullSideAct.half} 名內之隨機一隻): 均 ${f1(m.nullSideAct.mean)}%、sd ${f1(m.nullSideAct.sd)};超額 ${f1(m.nullSideAct.excess)} 個百分點(z ${m.nullSideAct.z}、p ${m.nullSideAct.p})${typeof m.nullSideAct.share === 'number' ? `;**組成可重現之比例 ${f1(m.nullSideAct.share * 100)}%**` : ''}${th3(m.nullSideAct)} —— ${m.nullSideAct.triggered === false ? '選集之每隻訊號數與做多比例皆與全體隨機相近(此列不入階梯);' : ''}「方向 × 活躍度」之組成本身就有績效;同組成內是否挑得比隨機好,以此列為準`)
    }
    if (m.nullSideGeo) {
        out.push(`同側同幾何同活躍度虛無(幾何對照:再匹配該側內止盈 ${m.nullSideGeo.nTp} 等分 × 止損 ${m.nullSideGeo.nSl} 等分;只供判讀、不入階梯): 均 ${f1(m.nullSideGeo.mean)}%、sd ${f1(m.nullSideGeo.sd)};超額 ${f1(m.nullSideGeo.excess)} 個百分點(z ${m.nullSideGeo.z}、p ${m.nullSideGeo.p})${typeof m.nullSideGeo.share === 'number' ? `;可重現之比例 ${f1(m.nullSideGeo.share * 100)}%` : ''}${th3(m.nullSideGeo)}`)
    }
    //擴充診斷(共用工具區 metrics/; 由做夢擴建; 每次完整評估自動算出): 每個一列, 取其自述之一行說明
    for (let [nm, r] of Object.entries(m.ext || {})) {
        out.push(`擴充診斷 ${nm}: ${r && r.error ? `未算(${r.error})` : (r && typeof r.text === 'string' ? r.text.slice(0, 600) : JSON.stringify(r).slice(0, 300))}`)
    }
    if (m.op.bySide) {
        let sd = (x) => `${f1(x.orders)} 單、損益 ${x.pnl}、勝率 ${f1(x.winRate * 100)}%`
        out.push(`損益歸因(各相位平均;損益 = ${m.unit ? '各單依其基礎單位名義計之損益和(u)' : '每單 1 單位之損益和'},合計 ${m.op.pnl}): 做多 ${sd(m.op.bySide.long)};做空 ${sd(m.op.bySide.short)}${Array.isArray(m.op.byGroup) && m.op.byGroup.length > 1 ? `;依子帳: ${m.op.byGroup.map((g) => `子帳 ${g.group}(每次決策 ${f1(g.seats)} 席)${sd(g)}`).join(' | ')}` : ''}${m.sel && Array.isArray(m.sel.tagPerf) && m.sel.tagPerf.length ? `;依 tag(前 6 種): ${m.sel.tagPerf.slice(0, 6).map((t) => `${t.tag} ${t.decisions} 次決策、${f1(t.orders)} 單、損益 ${t.pnl}`).join(' | ')}` : ''}`)
    }
    if (Array.isArray(m.op.thirds) && m.op.thirds.length) {
        out.push(`評估期間三等分(各段自空倉起算;看績效是否集中於某一段): ${m.op.thirds.map((g) => `${String(g.from).slice(0, 10)} ~ ${String(g.to).slice(0, 10)} 均 ${f1(g.eqYr.mean)}%(最差相位 ${f1(g.eqYr.min)}${typeof g.nullMean === 'number' ? `;隨機挑 ${f1(g.nullMean)}` : ''})`).join(' | ')}`)
    }
    if (train && typeof train.longRatio === 'number' && m.sel && typeof m.sel.longRatio === 'number') {
        let tg = (xs) => (Array.isArray(xs) && xs.length ? xs.slice(0, 4).map((t) => `${t.tag} ${t.share}`).join('、') : '無')
        out.push(`選集之做多比例: 訓練期(診斷)${train.longRatio} / 回測期 ${m.sel.longRatio};tag 之決策比例(前 4 種): 訓練期 ${tg(train.tags)} / 回測期 ${tg(m.sel.tags)} —— 兩期之側別或狀態比例皆固定於一端者,該規則並未隨當下狀態切換`)
        let sb = shortShareOf(m)
        if (typeof train.shortShare === 'number' && sb !== null) {
            //基準與機械判讀(sideFlowIssue)、看板之說明(domain.mjs)取同一份(規則 R-REC-36: 不寫死某一策略庫之實測值)
            let sfBase = sideFlowCfgOf(sideFlow).baseShort
            out.push(`下單之做空比例(做空單 ÷ 全部下單): 訓練期(診斷)${train.shortShare} / 回測期 ${sb};全體隨機約 ${sfBase.train} / ${sfBase.backtest} —— 席次之做多比例正常而下單偏在一側者,是兩側活躍度不對稱所致之組成;兩期皆遠高(或遠低)於隨機者,回測期之方向可能經此寫進模組`)
        }
    }
    if (m.fresh) {
        out.push(`新鮮度(${phaseText(phaseCountOfDoc(m), '相位')}均 %): ${Object.keys(m.fresh).map((H) => `${H}d ${f1(m.fresh[H].eqYr.mean)}`).join(' | ')}`)
    }
    if (m.sel) {
        out.push(`選集: 大小均 ${f1(m.sel.sizeMean)}(${m.sel.sizeMin} ~ ${m.sel.sizeMax});換手 Jaccard ${m.sel.turnoverJaccard};執行規則 ${m.hasAccept ? `有(下單率 ${f1(m.op.acceptRate * 100)}%)` : '無'};勝率 ${f1(m.op.winRate * 100)}%`)
        if (typeof m.sel.sigPerPick === 'number') {
            //選集組成: 分數受活躍度與多空組成之影響很大, 故與同劑量虛無(全體隨機)並列, 一眼可見成品是否只是換了活躍度或多空比例
            out.push(`選集組成(括號內為同劑量虛無 = 全體隨機): 每隻每個決策視窗之訊號數 ${m.sel.sigPerPick}${m.null && typeof m.null.sigPerPick === 'number' ? `(${m.null.sigPerPick})` : ''};做多比例 ${m.sel.longRatio}${m.null && typeof m.null.longRatio === 'number' ? `(${m.null.longRatio})` : ''};平均因子數 ${m.sel.nKeysMean};平均止盈 / 止損 ${m.sel.tpMean} / ${m.sel.slMean}`)
        }
        if (m.sel.orders) {
            //下單集中度: 選集很大而下單率低時, 實際下單者可能集中在少數策略 / 因子組合 / 進場棒
            out.push(`下單集中度(各相位平均): 實際下單之不同策略 ${f1(m.sel.orders.strats)} 隻、不同因子組合 ${f1(m.sel.orders.combos)} 種;下單最多之前 1% 進場棒占全部下單 ${f1(m.sel.orders.barTop1Share * 100)}%;單一進場棒最多 ${m.sel.orders.perBarMax} 單`)
        }
    }
    return out
}

//datedIn, 一段程式(已去註解)內第一個落在某期間之年份內之日期字樣(YYYY-MM 或 YYYY-MM-DD; 分隔可為 - / .); 無則 null
//  period = { start, end }(時間字串): 只看年份範圍 —— 粗篩「依評估期間內之日期寫死行為」; 訓練期之日期(例如求解窗終點)不列
let datedIn = (src, period) => {
    let y0 = Number(String(period.start).slice(0, 4))
    let y1 = Number(String(period.end).slice(0, 4))
    let m0 = Number(String(period.start).slice(5, 7))
    let years = []
    for (let y = y0; y <= y1; y++) {
        years.push(y)
    }
    let re = new RegExp(`(${years.join('|')})[-/.](0[1-9]|1[0-2])(?:[-/.](?:[0-2]\\d|3[01]))?`, 'g')
    for (let m of String(src).matchAll(re)) {
        //起始年之內, 早於期間起始月者屬期間之前(例: 回測期自 2025-10 起, 2025-04 為訓練期內之日期)
        if (Number(m[1]) === y0 && Number(m[2]) < m0) {
            continue
        }
        return m[0]
    }
    return null
}

//  refsOf(spec, nodeId)(選配): 回傳任務書列為參考之節點 id 陣列(其工作夾可讀); nodeId = 所驗證之節點之 id(評估環境之鍵改變後之舊節點, 其 id 與以規格求得者不同)
/**
 * 建立節點之迴圈驗證函數(重跑完整評估、掃操作紀錄、必要時重放回測期旗標);驗證時會起子行程,不宜於文件範例中實跑
 *
 * @param {Object} opt 輸入設定物件
 * @param {Object} opt.st 輸入設定物件(getSettings 之結果)
 * @param {String} opt.FD_ROOT 輸入專案根路徑
 * @param {Function} opt.layerOfSpec 輸入由規格取得所屬層之函數
 * @param {Function|null} [opt.refsOf=null] 輸入回傳任務書列為參考之節點 id 陣列之函數(輸入 spec 與所驗證之節點之 id),預設null
 * @returns {Function} 回傳非同步驗證函數 async ({ nodeId, spec, workDir, attempt=1, keepValidation=null, evalsUntil=null }),成功回傳 { ok:true, raw, score, summary, notes, fingerprint, issues, checks, validation, evalJson },失敗回傳 { ok:false, error, issues }
 * @example
 * let verify = makeVerify({ st, FD_ROOT, layerOfSpec })
 * let r = await verify({ nodeId, spec, workDir })
 */
let makeVerify = ({ st, FD_ROOT, layerOfSpec, refsOf = null }) => {
    let backtest = makeBacktest({ st, FD_ROOT })
    //  keepValidation(選配): 給定 { flag, at } 時不重放回測期, 沿用該旗標(評估器只改訓練期之輸出、回測期重放之結果不變時用; 見 reverifyAll)
    //  evalsUntil(選配; ISO 時刻): 評估紀錄只計至該時刻(節點之 agent 結束之時刻); 事後重跑(reverifyAll)時給定 —— 其後寫入之列是做夢或主控之評估, 不是節點之操作
    return async ({ nodeId, spec, workDir, attempt = 1, keepValidation = null, evalsUntil = null }) => {
        let rel = (p) => path.relative(FD_ROOT, p).replace(/\\/g, '/')
        let fpPick = path.join(workDir, 'pick.mjs')
        let fpOut = path.join(workDir, 'harness_full.json')
        //評估之執行面之數字(亂數種子、重放次數、重放之虛無抽數、逾時秒數): 與評估器、設定之預檢同一組函數(evalRunNumsOf; 規則 R-EV-44 ④之二): 未給 = 預設、數字字串轉為數值後才使用(原: 各處自行判斷 ——
        //  replicaSeeds 與 replicaNullDraws 不收數字字串、`seed + i` 於字串之種子為字串相接、`timeoutSec || 900` 之 falsy 取預設), 給了而無效 ⇒ 拋 invalid eval.<鍵>(組裝時之預檢已先拒絕)
        let evRun = evalRunNumsOf(st.eval, 'eval')
        if (!fs.existsSync(fpPick)) {
            let dh = deniedHalt(path.join(workDir, `agent_a${attempt}.log`))
            return { ok: false, error: `未交付 pick.mjs${dh ? `(工作階段結束於被權限拒絕之工具呼叫: ${dh.tool} ${dh.target})` : ''}`, issues: [] }
        }
        try {
            fs.rmSync(fpOut, { force: true })
        }
        catch (err) {}
        //agent 之評估次數(評估器自己記於 .evals.log)須在迴圈重跑之前讀; 迴圈之重跑帶 --no-log 不計
        let nLogged = evalsLogged(workDir, evalsUntil)
        //--no-log 只在 ARSI_HARNESS=1 時生效(g_eval.mjs); 節點自己帶 --no-log 者照記並標記
        let nNoLog = noLogTried(workDir, evalsUntil)
        let r = await runNodeProc([path.join(FD_ROOT, 'g_eval.mjs'), '--pick', fpPick, '--mode', 'full', '--out', fpOut, '--no-log'], { cwd: FD_ROOT, timeoutMs: evRun.timeoutSec * 1000, env: { ...process.env, ARSI_HARNESS: '1' } })
        let out = readJson(fpOut)
        if (r.timeout) {
            return { ok: false, error: `重跑逾時(${evRun.timeoutSec}s; 挑選模組過慢)`, issues: [] }
        }
        if (!out || out.ok !== true) {
            //結果檔帶 code 者(EVAL_UNIT_CAPITAL、PICK_RUN、LEDGER_UNIT_MISMATCH 等; 同為離開碼 3 或 2 者以之分辨)一併列於訊息與回傳
            let code = out && typeof out.code === 'string' ? out.code : null
            return { ok: false, error: `重跑失敗(離開碼 ${r.code}${code ? `; ${code}` : ''}): ${out && out.error ? out.error : r.out.split('\n').slice(-6).join(' | ').slice(0, 600)}`, issues: (out && out.issues) || [], ...(code ? { code } : {}) }
        }
        let issues = []
        let declared = out.pickMeta && Array.isArray(out.pickMeta.arms) ? out.pickMeta.arms : null
        if (!declared) {
            issues.push('成品未宣告 meta.arms')
        }
        else {
            let miss = spec.arms.filter((a) => !declared.includes(a))
            if (miss.length) {
                issues.push(`成品之 meta.arms 未含指定之臂 ${miss.join('、')}`)
            }
        }
        let src = stripComments(fs.readFileSync(fpPick, 'utf8'))
        if (/\[\s*(?:\d+\s*,\s*){20,}\d+\s*\]/.test(src)) {
            issues.push('原始碼含長整數陣列(疑似硬編序號清單)')
        }
        //把回測期之答案寫進模組之粗篩: 原始碼含評估期間內之日期字樣(依日期寫死行為); 依 ctx.k / ctx.day 之特定值切換者無法機械判定, 由審計看
        {
            let hit = datedIn(src, st.periods.backtest)
            if (hit) {
                issues.push(`原始碼含評估期間內之日期字樣「${hit}」(疑似依日期寫死行為;合約 §2「不得把回測期之答案寫進模組」)`)
            }
        }
        let own = readJson(path.join(workDir, 'result_full.json'))
        if (!own || !own.metrics) {
            issues.push('未交付 result_full.json(agent 自跑之完整評估)')
        }
        else if (own.period && own.period.name && own.period.name !== 'backtest') {
            issues.push(`agent 之 result_full.json 不是回測期之評估(${own.period.name});交付用之結果須為預設之評估期間`)
        }
        else if (own.metrics.selHash !== out.metrics.selHash) {
            issues.push(`agent 之 result_full.json 之 selHash ${own.metrics.selHash} 與重跑 ${out.metrics.selHash} 不同(交付後成品有改, 或結果不可重現)`)
        }
        //擴充診斷自帶之提示(規則 R-REC-21): 列入迴圈檢查, 不否決
        for (let f of extSurfaceOf(out.metrics).flags) {
            issues.push(extFlagIssue(f))
        }
        //換亂數種子與同分次序之重放(規則 R-EV-17、R-EV-22): 節點只看得到主種子與正本之同分次序 ⇒ 迴圈另重放 settings.eval.replicaSeeds 次(七相位、不算虛無), 記其 S
        //  每次同時換 ctx.rand 之種子(--seed)與同分次序(--tie-salt: ctx.meta.tie 與同一進場棒之訊號先後):
        //    含隨機抽取之成品隨種子波動; 主鍵大量同分之成品其入選者實際由 tie 決定(= 固定種子之隨機抽取), 只換種子驗不到(例: 換種子 S 不變、換 tie 即由正轉負)
        //  第一次重放之選集與 S 皆與主重放相同者(未用亂數、同分不影響入選、持倉上限未因先後而擋單)不再重放
        //  每次重放另附算組成對照(32 抽)與近期活躍度之組成對照(擴充診斷 nullRecentAct)之超額(規則 R-EV-24):
        //    只列 S 時看不出「內容」在重放下是否仍在; 單抽之組成對照不足以判內容(全庫隨機之兩組抽取 +1.3 / +11.6), 兩次重放即現成之第二、三抽
        let seedS = []
        let replays = []
        let isHalf = isHalfOf(out.pickMeta)
        {
            let nRep = evRun.replicaSeeds
            let seed0 = evRun.seed
            let nDraw = evRun.replicaNullDraws
            //重放一併載入之擴充診斷(規則 R-REC-42; settings.rsi.replicaExt): 預設 = 正本中帶看板值之診斷 + 正本之 neutralExec(其 delta 有值者); 'none' 或無此類診斷者之引數與改前逐字相同
            let rxScope = replicaExtOf(st.rsi)
            let extArgs = replicaExtArgsOf(rxScope, out.metrics)
            let extArgs0 = ['--ext-only', 'nullRecentAct']
            let replayOf = (i, fpS, ea) => runNodeProc([path.join(FD_ROOT, 'g_eval.mjs'), '--pick', fpPick, '--mode', 'full', '--null', String(nDraw), ...ea, '--seed', String(seed0 + i), '--tie-salt', String(i), '--out', fpS, '--no-log'], { cwd: FD_ROOT, timeoutMs: evRun.timeoutSec * 1000, env: { ...process.env, ARSI_HARNESS: '1' } })
            for (let i = 1; i <= nRep; i++) {
                let fpS = path.join(workDir, `harness_seed${i}.json`)
                try {
                    fs.rmSync(fpS, { force: true })
                }
                catch (err) {}
                let rs = await replayOf(i, fpS, extArgs)
                let o = readJson(fpS)
                //另載入之診斷使該次重放失敗者(逾時、完整性錯誤等; 診斷之拋錯本身由評估器記為 { error }, 不使之失敗)⇒ 以改前之引數再重放一次, 重放之 S 與既有之兩項不受影響
                if ((rs.timeout || !o || o.ok !== true || !o.score) && extArgs.join(' ') !== extArgs0.join(' ')) {
                    issues.push(`換亂數種子與同分次序之重放(第 ${i} 次)一併載入擴充診斷時失敗,已改為只載入 nullRecentAct 重放(該次無其他擴充診斷之值)`)
                    try {
                        fs.rmSync(fpS, { force: true })
                    }
                    catch (err) {}
                    rs = await replayOf(i, fpS, extArgs0)
                    o = readJson(fpS)
                }
                if (rs.timeout || !o || o.ok !== true || !o.score) {
                    issues.push(`換亂數種子與同分次序之重放失敗(第 ${i} 次)`)
                    break
                }
                if (i === 1 && o.metrics.selHash === out.metrics.selHash && out.score && o.score.S === out.score.S) {
                    break //結果不依賴亂數種子與同分次序
                }
                seedS.push(o.score.S)
                let nr = recentDiagOf(o.metrics) //近期活躍度版之 side(與領域、共同儲存區同一個「有值」之判準: 物件、不帶 error、超額為有限數值; 規則 R-REC-38 ⑦)
                //重放之擴充診斷之摘要(看板值與 neutralExec 之 delta; 'none' 者不存 ⇒ 紀錄與改前相同)
                let rx = rxScope === 'none' ? null : replayExtOf(o.metrics)
                replays.push({ S: o.score.S, comp: o.metrics.nullSideAct && typeof o.metrics.nullSideAct.excess === 'number' ? o.metrics.nullSideAct.excess : null, recent: nr ? nr.excess : null, ...(rx ? { ext: rx } : {}) })
            }
            //主分數為正而重放後為零或負: 分數依賴亂數抽取或同分次序(不否決, 由做夢審計)
            //  半庫版(成品自述 meta.params.half 為 0 或 1; 共用工具區之 *_half.mjs 樣板: 以 ctx.meta.tie 之雜湊切半)換同分次序即換一種切法 ⇒ 不列「分數不穩」,
            //  改於判讀加註(判準由成品自述, 由審計核對 —— 與擴充診斷之 notContent 同一作法)
            let sMain = out.score && typeof out.score.S === 'number' ? out.score.S : null
            if (!isHalf && seedS.length && sMain !== null && sMain > 0 && Math.min(...seedS) <= 0) {
                issues.push(`分數不穩: 主 S ${sMain},換亂數種子與同分次序後為 ${seedS.join(' / ')}(由正轉負;含隨機抽取或主鍵大量同分 ⇒ 入選者實際由亂數或 ctx.meta.tie 決定,由審計判定)`)
            }
        }
        //固定側別之機械判讀: 另以訓練期之診斷重放一次(quick、不算虛無), 只取選集組成(做多比例、tag 之決策比例)
        //  回測期若為單邊行情, 固定只挑一側者即可得高分而毫無自適應; 訓練期與回測期之做多比例皆 < 0.1 或皆 > 0.9 者列為問題(不否決, 由做夢審計)
        let train = null
        {
            let fpT = path.join(workDir, 'harness_train.json')
            try {
                fs.rmSync(fpT, { force: true })
            }
            catch (err) {}
            let rt = await runNodeProc([path.join(FD_ROOT, 'g_eval.mjs'), '--pick', fpPick, '--mode', 'quick', '--period', 'train', '--null', '0', '--out', fpT, '--no-log'], { cwd: FD_ROOT, timeoutMs: evRun.timeoutSec * 1000, env: { ...process.env, ARSI_HARNESS: '1' } })
            let o = readJson(fpT)
            if (!rt.timeout && o && o.ok === true && o.metrics && o.metrics.sel && typeof o.metrics.sel.longRatio === 'number') {
                train = { longRatio: o.metrics.sel.longRatio, tags: (o.metrics.sel.tags || []).slice(0, 4), ...(shortShareOf(o.metrics) === null ? {} : { shortShare: shortShareOf(o.metrics) }) }
                let a = train.longRatio
                let b = out.metrics.sel.longRatio
                if (fixedSideOf(a, b, out.metrics.sel.sizeMax)) {
                    issues.push(`固定側別: 選集之做多比例 訓練期 ${a} / 回測期 ${b}(兩期皆只挑${a < 0.1 ? '做空' : '做多'}側;回測期之行情方向可能被寫進模組,由審計判定)`)
                }
                //下單流量偏一側(規則 R-REC-15): 席次之做多比例正常, 仍可經由兩側活躍度之不對稱使下單集中在一側
                //  側別曝險之判定關閉者不列為問題(規則 R-REC-55 ③; 判定在 sideFlowIssueOfSt)
                let sf = sideFlowIssueOfSt(shortShareOf(out.metrics), train.shortShare, st)
                if (sf) {
                    issues.push(sf)
                }
            }
        }
        //寫到工作夾之外之檔(常見: 路徑少一個分隔符 ⇒ <工作區>/<節點 id><檔名>): 移入工作夾之 _stray/ 並列為問題
        {
            let wr = path.dirname(workDir)
            let strays = []
            try {
                strays = fs.readdirSync(wr).filter((f) => f.startsWith(nodeId) && f !== path.basename(workDir) && fs.statSync(path.join(wr, f)).isFile())
            }
            catch (err) {}
            if (strays.length) {
                let moved = []
                for (let f of strays) {
                    try {
                        fs.mkdirSync(path.join(workDir, '_stray'), { recursive: true })
                        fs.renameSync(path.join(wr, f), path.join(workDir, '_stray', f.slice(nodeId.length) || f))
                        moved.push(f.slice(nodeId.length) || f)
                    }
                    catch (err) {}
                }
                issues.push(`有 ${strays.length} 個檔寫在工作夾之外(${strays.slice(0, 4).join('、')}${strays.length > 4 ? ' …' : ''};路徑少了分隔符)${moved.length ? `,已移入工作夾之 _stray/` : ''}`)
            }
        }
        //可讀之他節點工作夾: 起點(spec.base)與任務書列為參考之節點(由呼叫端經 allowWork 給入)
        let tr = scanTrace({ workDir, attempt, privateRoot: backtest.cfg.privateRoot, root: FD_ROOT, writeRoots: [workDir], workRoot: st.records.workDir, allowWork: [...(spec.base ? [spec.base] : []), ...(typeof refsOf === 'function' ? refsOf(spec, nodeId) : [])], nodeRules: true, stateRoot: st.rsi && st.rsi.root })
        issues.push(...tr.issues)
        if (nNoLog > 0) {
            issues.push(`以 --no-log 執行評估 ${nNoLog} 次(該旗標只供迴圈重跑; 已照記入評估次數)`)
        }
        let lost = lostVersions(workDir, evalsUntil)
        if (lost.length) {
            issues.push(`已完整評估而現無檔可重現之版本 ${lost.reduce((s, x) => s + x.n, 0)} 個(${lost.map((x) => `${x.pick} × ${x.n}`).join('、')}): 評估後覆寫了該檔(一實驗一檔)`)
        }
        //評估次數: 以評估器之紀錄為準; 無紀錄檔(舊交付)才用操作紀錄數得者
        let nEvals = nLogged !== null ? nLogged : tr.nEvals
        //完整評估之相異版本數(迴圈之重跑帶 --no-log 不記, 故此處讀到的只有 agent 自己的評估)
        let nDist = evalsDistinct(workDir, evalsUntil)
        let layer = layerOfSpec(spec)
        let score = scoreOf(out.metrics, { limits: st.eval.limits, score: st.score, layer, nPh: phaseCountOf(st.eval) })
        let trainScore = score
        //檢定旗標(「回測期旗標」機制; 隨附之參考設定明設 settings.backtest.enabled = false(未設時 backtestCfg 視為啟用): 不啟用時評估期即回測期, 節點自己看得到數字;
        //  日後若由主控以保留期或他幣作檢定, 可沿用此接口, 見 sgud/proc/process.md §5)
        let validation = null
        if (keepValidation && typeof keepValidation.flag === 'string' && backtest.cfg.enabled) {
            validation = { name: 'backtest', flag: keepValidation.flag, label: gradeLabel(keepValidation.flag), at: keepValidation.at || null, error: null }
            score = scoreOf(out.metrics, { limits: st.eval.limits, score: st.score, layer, evidence: { backtest: keepValidation.flag }, nPh: phaseCountOf(st.eval) })
        }
        else if (backtest.cfg.enabled && score.ladder >= backtest.cfg.minLadder && !score.flags.vetoed) {
            let b = await backtest.run({ nodeId, attempt, workDir })
            validation = { name: 'backtest', flag: b.flag, label: b.flag ? gradeLabel(b.flag) : null, at: b.at, error: b.error }
            if (b.flag) {
                score = scoreOf(out.metrics, { limits: st.eval.limits, score: st.score, layer, evidence: { backtest: b.flag }, nPh: phaseCountOf(st.eval) })
            }
        }
        return {
            ok: true,
            raw: { metrics: out.metrics, selHash: out.metrics.selHash, seedS, replays, ...(train ? { train } : {}) },
            score,
            summary: summaryOf(out.metrics, trainScore),
            notes: notesOf(out.metrics, seedS, train, replays, isHalf, st.score && st.score.sideFlow),
            fingerprint: out.metrics.selHash,
            issues,
            checks: { ...(nEvals === null ? {} : { nEvals }), ...(nDist ? { ...nDist, nEvalsNote: `其中完整評估 ${nDist.nFull} 次、相異之選集(selHash)${nDist.nFullSel} 個;只差 tag、只加 --ref 或同一內容重跑者不算新版本` } : {}), ...(tr.note ? { traceNote: tr.note } : {}) },
            validation,
            evalJson: rel(fpOut),
        }
    }
}

//  board(看板之值; 字串或有限數字, 取前 32 字元)、boardLabel(該值之欄名; 取前 32 字元, 預設 = 診斷之檔名)、flag(迴圈檢查之提示; 非空字串, 取前 160 字元; 不否決)

/**
 * 取出擴充診斷自帶之看板值與迴圈提示(來自 metrics.ext 各診斷之 board、boardLabel、flag、notContent、subset;字串去除直線與換行)
 *
 * @param {Object} m 輸入評估之 metrics,讀取 m.ext
 * @returns {Object} 回傳 { cells, flags, notContent, subsets }:cells 每項 { name, label, value },flags 與 notContent 每項 { name, text },subsets 每項 { name, label }
 * @example
 * let s = extSurfaceOf({ ext: { demo: { board: 3, flag: '提示' } } })
 */
let extSurfaceOf = (m) => {
    let ext = m && m.ext && typeof m.ext === 'object' && !Array.isArray(m.ext) ? m.ext : {}
    let clean = (v, n) => String(v).replace(/[|\r\n]+/g, ' ').replace(/\s+/g, ' ').trim().slice(0, n)
    //  subset(規則 R-REC-27): 非空字串(取前 32 字元)= 診斷自述「此成品屬於某個子集」之名稱(例: 「時效 ≥ 15」)
    //    ⇒ 搜尋地圖 §1 於現行量尺旁, 另列各子集內「有內容且可信」之最高者(並列、不取代現行量尺); 條件不成立時回 null 或不帶; 子集之定義由診斷自述, 迴圈不寫死哪一個診斷
    let cells = []
    let flags = []
    let notContent = []
    let subsets = []
    for (let name of Object.keys(ext).sort()) {
        let o = ext[name]
        if (!o || typeof o !== 'object' || Array.isArray(o) || o.error) {
            continue
        }
        if (typeof o.subset === 'string' && o.subset.trim()) {
            subsets.push({ name, label: clean(o.subset, 32) })
        }
        if ((typeof o.board === 'string' && o.board.trim()) || (typeof o.board === 'number' && Number.isFinite(o.board))) {
            cells.push({ name, label: typeof o.boardLabel === 'string' && o.boardLabel.trim() ? clean(o.boardLabel, 32) : name, value: clean(o.board, 32) })
        }
        if (typeof o.flag === 'string' && o.flag.trim()) {
            flags.push({ name, text: clean(o.flag, 160) })
        }
        if (typeof o.notContent === 'string' && o.notContent.trim()) {
            notContent.push({ name, text: clean(o.notContent, 160) })
        }
    }
    return { cells, flags, notContent, subsets }
}

//boardMeanOf, 看板值之多抽之均(純函數): vals = 各抽之看板值(字串或數字; 正本在前); 每個值之數字(BOARD_NUM)換成佔位後之樣板須各抽相同, 依位置取均(小數位數 = 該位置各抽之最多者);
//  少於 2 抽、樣板不同(例:「—」與數字、文字不同、正負號以「+」書寫者與不帶者)或無數字者回 null(不硬算)
let BOARD_NUM = /-?\d+(?:\.\d+)?/g
let boardMeanOf = (vals) => {
    if (!Array.isArray(vals) || vals.length < 2) {
        return null
    }
    let parts = vals.map((v) => {
        let nums = []
        let tpl = String(v).replace(BOARD_NUM, (x) => {
            nums.push(x)
            return '\u0000'
        })
        return { tpl, nums }
    })
    if (parts[0].nums.length === 0 || parts.some((p) => p.tpl !== parts[0].tpl)) {
        return null
    }
    let mean = parts[0].nums.map((_, i) => {
        let dec = Math.max(...parts.map((p) => (p.nums[i].split('.')[1] || '').length))
        let s = (parts.reduce((a, p) => a + Number(p.nums[i]), 0) / parts.length).toFixed(dec)
        return /^-0(\.0*)?$/.test(s) ? s.slice(1) : s
    })
    return parts[0].tpl.split('\u0000').reduce((acc, seg, i) => acc + (i > 0 ? mean[i - 1] : '') + seg, '')
}

//  回 [{ name, label, value, n }](依診斷名; n = 所取之抽數): 只列正本有看板值、至少一次重放帶其看板值且可取均者; 重放皆未帶 ext 者回 [](看板與節點文件與改前相同)

    if (!rp.length) {

        let vals = [c.value, ...rp.map((x) => (x.ext[c.name] && x.ext[c.name].board !== undefined ? x.ext[c.name].board : null)).filter((v) => v !== null)]
        let v = boardMeanOf(vals)
        if (v !== null) {
            out.push({ name: c.name, label: c.label, value: v, n: vals.length })
        }
    }
    return out
}
