    let a = new Ctor(cap)
    let n = 0
    return {
        push: (v) => {
            if (n >= a.length) {
                let b = new Ctor(a.length * 2)
                b.set(a)
                a = b
            }
            a[n++] = v
        },
        size: () => n,
        done: () => a.subarray(0, n),
    }
}

//alignSeries, 把一條外部序列對位到 K 棒(純函數; 規則 R-EV-34)
/**
 * 把一條外部序列 [{ time, value }] 對位到 K 棒: time 為棒之開盤時間(與 K 線之 time 同一字串約定), 對不到棒之列略過; 值須為有限數(數字或非空白之數字字串), 其餘一律略過; 未被填入之棒為 NaN
 * 同一時間有多筆有效值且相異者(互相矛盾): 該棒取缺值 NaN(收嚴; 結果與列之順序無關), 值相同者無影響; 無效之列視為未列; K 線已於 endTime 截斷者, 時間 ≥ endTime 之列自然對不到棒 ⇒ 帳本內之序列不含「時間標記在 endTime 以後」之列所帶之值
 * 選配之 stat 物件(輸入 endTime、time0 為字串者才分類)於呼叫後填入計數, 供建帳本時提示: rows(列數)、matched(對到棒之列)、invalid(非物件之列與值無效之列)、afterEnd(對不到棒且時間 ≥ endTime)、beforeStart(對不到棒且時間 < time0)、offGrid(其餘對不到棒者; 多為時間字串格式與 K 線不同或不在棒之開盤時間)、offGridSample(第一個 offGrid 之時間)、conflict(多筆相異值而取缺值之棒數)、conflictSample(第一個之時間)
 *
 * @param {Array} arr 輸入序列陣列, 各元素為 { time, value }; 非陣列者視為空序列
 * @param {Map} kpTimeInd 輸入「棒之時間字串 → 棒序號」之對照
 * @param {Integer} n 輸入 K 棒數
 * @param {Object} [stat=null] 輸入計數物件(選配; 可帶 endTime、time0 字串; 呼叫後填入各計數),預設null
 * @returns {Float64Array} 回傳長度 n 之序列(缺值 NaN)
 * @example
 * let a = alignSeries([{ time: '2024-01-01T00:00:00', value: 1.5 }], new Map([['2024-01-01T00:00:00', 0]]), 1)
 */
let alignSeries = (arr, kpTimeInd, n, stat = null) => {
    let a = new Float64Array(n).fill(NaN)
    let st = stat && typeof stat === 'object' ? stat : null
    if (st) {
        Object.assign(st, { rows: 0, matched: 0, invalid: 0, afterEnd: 0, beforeStart: 0, offGrid: 0, offGridSample: null, conflict: 0, conflictSample: null })
    }
    let conf = null //有多筆相異值之棒(1); 一旦矛盾即維持缺值, 後續同棒之列不得再填入 ⇒ 結果與列之順序無關
    for (let v of (Array.isArray(arr) ? arr : [])) {
        if (st) {
            st.rows++
        }
        if (!v || typeof v !== 'object') {
            if (st) {
                st.invalid++
            }
            continue
        }
        let ind = kpTimeInd.get(v.time)
        if (ind === undefined) {
            if (st) {
                let isStr = typeof v.time === 'string'
                if (isStr && typeof st.endTime === 'string' && v.time >= st.endTime) {
                    st.afterEnd++
                }
                else if (isStr && typeof st.time0 === 'string' && v.time < st.time0) {
                    st.beforeStart++
                }
                else {
                    st.offGrid++
                    if (st.offGridSample === null) {
                        st.offGridSample = String(v.time)
                    }
                }
            }
            continue
        }
        if (st) {
            st.matched++
        }
        //只收數字與非空白之數字字串(Number('') 與 Number('  ') 為 0、Number([]) 為 0、Number(true) 為 1, 皆不得當成數值)
        let raw = v.value
        let x = typeof raw === 'number' ? raw : (typeof raw === 'string' && raw.trim() !== '' ? Number(raw) : NaN)
        if (Number.isFinite(x)) {
            if (conf !== null && conf[ind] === 1) {
                continue
            }
            if (!Number.isNaN(a[ind]) && a[ind] !== x) {
                //同一時間之兩個有效值相異(互相矛盾): 取缺值(原則: 遇衝突取最嚴之解讀, 不靜默擇一)
                if (conf === null) {
                    conf = new Uint8Array(n)
                }
                conf[ind] = 1
                a[ind] = NaN
                if (st) {
                    st.conflict++
                    if (st.conflictSample === null) {
                        st.conflictSample = String(v.time)
                    }
                }
            }
            else {
                a[ind] = x === 0 ? 0 : x //-0 一律存為 0(位元組亦與列之順序無關)
            }
        }
        else if (st) {
            st.invalid++
        }
    }
    return a
}


/**

 * K 線於 endTime 截斷(不含); 語義與輸出欄見檔頭說明

 * @param {String} [opt.seriesDir=null] 輸入外部序列資料夾路徑(選配; 其下每個 <名>.json 為一條序列 [{ time, value }], time 為K棒之開盤時間, 檔名以底線起首者略過; 名稱須英文字母起首、其後英數或底線、至多32字),預設null(不建入; 資料夾不存在者亦不建入並於記錄註明)
 * @param {Object} [opt.unit=null] 輸入基礎單位之設定(選配; 規則 R-EV-43){ rule: { stepSize, minQty, minNotional, asOf?, source? }, buffer = 0.1, uIni }: 給者另存每根棒之基礎單位名義(unit.f64)、meta.unit 與 meta.unitStat, 交易規則與緩衝入帳本鍵; 單一策略之全期最大同時持倉名義只記錄(最大者與超過 uIni 之隻數; 不拋錯、照寫出); 不合者拋 invalid unit…之錯誤,預設null(不啟用; 帳本與未加入本功能時逐位相同)
 * @param {Number} [opt.seriesOffGridMax=null] 輸入外部序列對不到棒之比例上限(0 ~ 1; 數字或數字字串; 選配; 規則 R-EV-44): 給者, 任一序列之時間落在帳本範圍內而對不到棒之列之占比超過即拋 invalid series…(不寫出); 不合者於讀任何檔之前拋 invalid seriesOffGridMax…,預設null(只記錄; 行為不變)
 * @param {Function} [opt.log=console.log] 輸入記錄函數,預設console.log
 * @returns {Promise<Object>} 回傳Promise,resolve為帳本 meta 物件(含 version、key、endTime、time0、nBars、nStrategies、nOrders、nUnsettled、check、keyNames、seriesNames 等; 給 startTime 者另含 startTime; 給 unit 者另含 unit、unitStat; check 於有策略因求解窗起點不在帳本之棒上而未納入核對時另含 skipped)
 * @example
 * let meta = await buildLedger({ libraryRoot: './data/library', keyOhlc: 'BTCUSDT_4h', endTime: '2024-01-01T00:00:00', outDir: './data/ledger/eval' })
 * let meta2 = await buildLedger({ libraryRoot: './data/library', keyOhlc: 'BTCUSDT_4h', startTime: '2022-07-01T00:00:00', endTime: '2024-01-01T00:00:00', outDir: './data/ledger/eval' })
 */
let buildLedger = async ({ libraryRoot, keyOhlc, endTime, startTime = null, outDir, barHours = 4, limit = 0, seriesDir = null, unit = null, seriesOffGridMax = null, log = console.log }) => {

    let t0 = Date.now()
    //每根之時長與外部序列之門檻(規則 R-EV-44): 先於讀任何檔檢查; barHours 通過即轉為數值(其後一律用轉換值)
    barHours = normBarHours(barHours, 'barHours')
    let barSec = Math.round(barHours * 3600)
    let offGridMax = null
    if (seriesOffGridMax !== undefined && seriesOffGridMax !== null) {
        offGridMax = isenum(seriesOffGridMax, { min: 0, max: 1 }) ? Number(seriesOffGridMax) : NaN
        if (!(offGridMax >= 0 && offGridMax <= 1)) {
            throw new Error('invalid seriesOffGridMax: must be a number from 0 to 1')
        }
    }
    //帳本之起點(規則 R-EV-45; 選配): 型別於讀任何檔之前檢查(須為可解析之時間字串); 「是 K 線檔內某一根之開盤時間且早於 endTime」須讀 K 線檔才能判定, 於其後、任何寫出之前檢查; 訊息不帶輸入值
    let hasStart = startTime !== undefined && startTime !== null
    if (hasStart && !(typeof startTime === 'string' && Number.isFinite(Date.parse(`${startTime}Z`)))) {
        throw new Error(START_TIME_MSG)
    }
    //基礎單位之設定(規則 R-EV-43): 先於讀任何檔檢查(不合者不耗時)
    let nu = normUnit(unit, 'unit')
    //每根棒之名義之函數(回測套件之 calcBaseUnitQuantity): 給了 unit 才動態匯入 ⇒ 未給者不依賴之; 相依之回測套件不含此函數(或其所匯入之相依缺漏)者拋 invalid unit…(不留載入失敗之堆疊; 原錯誤保留於 err.cause)
    let unitNotionalOf = null
    if (nu) {
        try {
            unitNotionalOf = (await import('./unitNotional.mjs')).unitNotionalOf
        }
        catch (err) {
            if (err && err.code === 'ERR_MODULE_NOT_FOUND') {
                throw new Error('invalid unit: must be used with a w-trade-backtest that provides calcBaseUnitQuantity and its dependencies (upgrade or reinstall w-trade-backtest)', { cause: err })
            }
            throw err
        }
    }

    //ohlc(於 endTime 截斷)
    let arrAll = JSON.parse(fs.readFileSync(path.join(libraryRoot, 'ohlc', `${keyOhlc}.json`), 'utf8'))
    //起點(規則 R-EV-45): 須為 K 線檔內某一根之開盤時間且早於 endTime(與 endTime 同樣以字串比較); 不合者於任何寫出之前拋錯
    if (hasStart && !(startTime < endTime && arrAll.some((r) => r.time === startTime))) {
        throw new Error(START_TIME_MSG)
    }
    let arrOhlc = arrAll.filter((r) => (hasStart ? r.time >= startTime : true) && r.time < endTime)
    let n = arrOhlc.length
    if (n < 100) {
        if (hasStart) {
            throw new Error(START_FEW_MSG)
        }
        throw new Error(`K 線過少(${n}); endTime ${endTime}`)
    }
    let barMs = barHours * 3600000
    let time0 = arrOhlc[0].time
    for (let i = 1; i < n; i++) {
        if (Date.parse(`${arrOhlc[i].time}Z`) - Date.parse(`${arrOhlc[i - 1].time}Z`) !== barMs) {
            throw new Error(`K 線不連續: ${arrOhlc[i - 1].time} → ${arrOhlc[i].time}`)
        }
    }
    let C = new Float64Array(n)
    let kpTimeInd = new Map()
    for (let i = 0; i < n; i++) {
        C[i] = arrOhlc[i].Close
        kpTimeInd.set(arrOhlc[i].time, i)
    }
    log(`ohlc ${keyOhlc}: ${n} 根, ${time0} ~ ${arrOhlc[n - 1].time}(${hasStart ? `startTime ${startTime} 含; ` : ''}endTime ${endTime} 不含)`)
    //每根棒之基礎單位名義(規則 R-EV-43; 給 unit 者): 進場價 = 進場棒之收盤價 ⇒ 每單之名義 = uN[進場棒]
    let uN = nu ? unitNotionalOf(C, nu, arrOhlc.map((r) => r.time)) : null
    //dN, 逐策略之持倉名義之差分(單一策略之峰值名義之記錄; 每隻策略算完即歸零重用)
    let dN = nu ? new Float64Array(n + 1) : null
    let uPeak = 0
    let uPeakFile = null
    let uOver = 0
    //uOverTop, 超過 uIni 者之峰值最高之至多 10 隻 [峰值, 策略序號](供核查; 峰值降冪、同值依策略序)
    let uOverTop = []
    if (nu) {
        log(`基礎單位: 交易規則 ${unitKeyOf(nu)}(步長|最小數量|最小名義價值|緩衝), 每根棒之名義 ${uN.reduce((a, v) => Math.min(a, v), Infinity)} ~ ${uN.reduce((a, v) => Math.max(a, v), -Infinity)}u, 研究用本金 ${nu.uIni}u`)
    }

    //因子序列對位(缺值 NaN)
    let fdParam = path.join(libraryRoot, 'param')
    let keyNames = fs.readdirSync(fdParam).filter((f) => f.endsWith('.json')).map((f) => f.slice(0, -5)).sort()
    let keyIdxOf = new Map(keyNames.map((k, i) => [k, i]))
    let aligned = new Array(keyNames.length)
    for (let i = 0; i < keyNames.length; i++) {
        let arr = JSON.parse(fs.readFileSync(path.join(fdParam, `${keyNames[i]}.json`), 'utf8'))
        let a = new Float64Array(n).fill(NaN)
        for (let v of arr) {
            let ind = kpTimeInd.get(v.time)
            if (ind !== undefined) {
                a[ind] = v.param
            }
        }
        aligned[i] = a
    }
    log(`因子序列 ${keyNames.length} 個已對位(${((Date.now() - t0) / 1000).toFixed(1)}s)`)

    //外部序列對位(選配; 規則 R-EV-34; 供挑選函數以 ctx.series 取決策棒以前之值): seriesDir 下每個 <名>.json 為一條序列(格式同因子檔: [{ time, value }]),
    //  名稱 = 檔名(依字典序); 檔名以底線起首者(報表等)略過。K 線已於 endTime 截斷 ⇒ 時間 ≥ endTime 之列對不到棒 ⇒ 帳本內之序列不含時間標記在 endTime 以後之列所帶之值
    //  (endTime 與 K 線之 time 以字串比較: 須同一字串格式且落在棒界, 為既有之約定)
    //  先於任何寫出完成(名稱或內容不合者拋錯時不留下半份帳本)
    let seriesNames = []
    let alignedSer = []
    let stats = []
    if (seriesDir) {
        if (!fs.existsSync(seriesDir)) {
            log(`注意: 外部序列夾不存在, 不建入外部序列: ${seriesDir}`)
        }
        else {
            seriesNames = fs.readdirSync(seriesDir).filter((f) => f.endsWith('.json') && !f.startsWith('_')).map((f) => f.slice(0, -5)).sort()
            for (let nm of seriesNames) {
                if (!RE_SERIES_NAME.test(nm)) {
                    throw new Error(`外部序列之名稱不合(英文字母起首, 其後英數或底線, 至多 32 字): ${nm}`)
                }
                let arr
                try {
                    arr = JSON.parse(fs.readFileSync(path.join(seriesDir, `${nm}.json`), 'utf8'))
                }
                catch (err) {
                    throw new Error(`外部序列檔 ${nm}.json 讀取或解析失敗: ${err.message}`)
                }
                if (!Array.isArray(arr)) {
                    throw new Error(`外部序列檔 ${nm}.json 須為 [{ time, value }] 陣列`)
                }
                let stat = { endTime, time0 }
                alignedSer.push(alignSeries(arr, kpTimeInd, n, stat))
                stats.push(stat)
            }
            let nValid = alignedSer.map((a) => a.reduce((c, v) => c + (Number.isFinite(v) ? 1 : 0), 0))
            log(`外部序列 ${seriesNames.length} 個已對位${seriesNames.length ? `: ${seriesNames.map((nm, i) => `${nm}(列 ${stats[i].rows}、有值 ${nValid[i]} / ${n})`).join('、')}` : ''}`)
            //提示(不拋錯): 時間不在棒之開盤時間(多為字串格式與 K 線不同)、值無效而略過、一個值也對不到
            for (let i = 0; i < seriesNames.length; i++) {
                let s = stats[i]
                let notes = []
                if (s.offGrid > 0) {
                    notes.push(`${s.offGrid} 列之時間對不到棒(例 ${s.offGridSample}; time 須為棒之開盤時間且與 K 線同一字串格式, 例 ${arrOhlc[0].time})`)
                }
                if (s.invalid > 0) {
                    notes.push(`${s.invalid} 列之值無效而略過(須為有限數字或數字字串)`)
                }
                if (s.conflict > 0) {
                    notes.push(`${s.conflict} 根棒有同一時間之多筆相異值而取缺值(例 ${s.conflictSample})`)
                }
                if (nValid[i] === 0) {
                    notes.push('無任何值對到棒(整條序列皆為缺值)')
                }
                if (notes.length) {
                    log(`注意: 外部序列 ${seriesNames[i]}: ${notes.join('; ')}`)
                }
            }
            //對不到棒之比例上限(規則 R-EV-44; 給了 seriesOffGridMax 者): 占比 = offGrid ÷(matched + offGrid)—— 只計時間落在帳本範圍內之列(終點以後、起點以前者不計)
            //  於寫出之前拋錯(不留半份帳本); 例: 每根 4 小時之序列餵給每根 1 日之帳本, 每日只對到 1 列(占比約 5/6)
            //  時間落在帳本範圍內之列為 0 者(分母 0; 整條序列皆在帳本之外或為空)亦拒絕 —— 給了門檻即要求序列確實對位到本帳本
            //  (覆蓋率不在本檢查: 較粗之序列餵給較細之帳本〔例: 每日之序列給每根 4 小時之帳本〕其列皆對得到棒, 占比為 0, 而多數棒為缺值)
            if (offGridMax !== null) {
                for (let i = 0; i < seriesNames.length; i++) {
                    let s = stats[i]
                    let den = s.matched + s.offGrid
                    if (den === 0) {
                        throw new Error(`invalid series ${seriesNames[i]}: must have rows within the ledger (required by seriesOffGridMax)`)
                    }
                    if (s.offGrid / den > offGridMax) {
                        throw new Error(`invalid series ${seriesNames[i]}: off-grid rows exceed seriesOffGridMax (${s.offGrid} of ${den} rows within the ledger; time must be the bar open time in the same string format as the K lines)`)
                    }
                }
            }
        }
    }

    //first-touch 表(每 方向 × tp × sl 一份)
    let kpFt = new Map()
    let ftOf = (mode, tp, sl) => {
        let k = `${mode}_${tp}_${sl}`
        let ft = kpFt.get(k)
        if (!ft) {
            ft = genKLinesTpSl(arrOhlc, { tp, sl, mode })
            kpFt.set(k, ft)
        }
        return ft
    }

    //策略
    let fdSt = path.join(libraryRoot, 'strategies')
    let files = fs.readdirSync(fdSt).filter((f) => f.endsWith('.json')).sort()
    if (limit > 0) {
        files = files.slice(0, limit)
    }
    let ns = files.length
    let off = new Uint32Array(ns + 1)
    let ent = grow(Int32Array, 1 << 22)
    let ext = grow(Int32Array, 1 << 22)
    let pnl = grow(Float32Array, 1 << 22)
    let mode = new Int8Array(ns)
    let tp = new Float32Array(ns)
    let sl = new Float32Array(ns)
    let nKeys = new Uint8Array(ns)
    let lv = new Int32Array(ns)
    let fitNt = new Int32Array(ns)
    let cls = new Uint8Array(ns * 3)
    let keyOff = new Uint32Array(ns + 1)
    let keyIdx = grow(Uint16Array, 1 << 19)
    //進場規則之各條件(與 keyIdx 逐格對齊; 規則 R-EV-29): 比較方向(+1 = '>'、-1 = '<')、門檻(原值, 不降精度)、是否為 or 條件
    let keySym = grow(Int8Array, 1 << 19)
    let keyTh = grow(Float64Array, 1 << 19)
    let keyOr = grow(Uint8Array, 1 << 19)
    let rule = { nConds: 0, nGt: 0, nLt: 0, nOr: 0, nSymOther: 0 } //nSymOther: 比較符號非 '>' / '<' 者(觸發時一律當 '<', 與既有語義相同; 應為 0)
    let nUnsettled = 0
    let chk = { n: 0, same: 0, diffs: [], skipped: 0, skippedFirst: null } //skippedFirst: 第一個被略過者之檔名與其起點欄(只供記錄, 不入 meta);與策略檔所存求解窗成交數之一致性(求解窗終點於帳本範圍內者); skipped: 求解窗起點不在本帳本之棒上而未納入核對之隻數(規則 R-EV-45)
    for (let si = 0; si < ns; si++) {
        let s = JSON.parse(fs.readFileSync(path.join(fdSt, files[si]), 'utf8'))
        let md = s.mode
        if (md !== 'long' && md !== 'short') {
            throw new Error(`${files[si]}: mode ${md}`)
        }
        let st = s.settings || {}
        //每根秒數之宣告(規則 R-EV-44): 有帶者須等於 barHours × 3600(數字或數字字串); 未帶者不檢
        let isec = st.intervalSec
        if (isec !== undefined && isec !== null && !(isenum(isec) && Number(isec) === barSec)) {
            throw new Error(`invalid settings.intervalSec of strategy ${files[si]}: must equal barHours * 3600`)
        }
        let rtp = st.rTakeProfit
        let rsl = st.rStopLoss
        let fee = typeof st.rFee === 'number' ? st.rFee : 0.0005
        let conds = s.conds || []
        if (conds.length === 0) {
            throw new Error(`${files[si]}: 無 conds`)
        }
        let cs = conds.map((c) => {
            let ki = keyIdxOf.get(c.key)
            if (ki === undefined) {
                throw new Error(`${files[si]}: 因子 ${c.key} 不在 param`)
            }
            let sym = c.sym === undefined ? '>' : c.sym
            let opr = c.opr === 'or' ? 'or' : 'and'
            if (sym !== '>' && sym !== '<') {
                rule.nSymOther++
            }
            return { ki, arr: aligned[ki], gt: sym === '>', th: c.th === undefined ? 0 : c.th, isOr: opr === 'or' }
        })
        let ft = ftOf(md, rtp, rsl)
        let win = ft.win
        let hold = ft.hold
        off[si] = ent.size()
        for (let i = 0; i < n; i++) {
            let bSkip = false
            let okAnd = true
            let hasOr = false
            let okOr = false
            for (let c of cs) {
                let y = c.arr[i]
                if (Number.isNaN(y)) {
                    bSkip = true
                    break
                }
                let b = c.gt ? y > c.th : y < c.th
                if (c.isOr) {
                    hasOr = true
                    okOr = okOr || b
                }
                else {
                    okAnd = okAnd && b
                }
            }
            if (bSkip || !okAnd || (hasOr && !okOr)) {
                continue
            }
            let c0 = C[i]
            let w = win[i]
            ent.push(i)
            if (w >= 0) {
                let priceEnd
                if (w === 1) {
                    priceEnd = md === 'long' ? (1 + rtp) * c0 : (1 - rtp) * c0
                }
                else {
                    priceEnd = md === 'long' ? (1 - rsl) * c0 : (1 + rsl) * c0
                }
                ext.push(i + hold[i])
                pnl.push(settleOrderPnl(md, 1, c0, priceEnd, fee))
            }
            else {
                ext.push(-1)
                pnl.push(0)
                nUnsettled++
            }
        }
        //單一策略之峰值名義(規則 R-EV-43; 給 unit 者; 只記錄): 本策略於帳本全期之最大同時持倉名義 —— 其各單於 [進場棒, 出場棒) 持倉(與評估器之持倉計法相同), 未結者持倉至期末
        //  持倉只在進場棒增加 ⇒ 峰值落在某一進場棒; 掃描至最後一個進場棒或出場棒(其間之差分全部歸零, 供下一隻重用)
        if (nu) {
            let E0 = ent.done()
            let X0 = ext.done()
            let a = off[si]
            let z = E0.length
            if (z > a) {
                let lo = E0[a]
                let hi = E0[z - 1]
                for (let j = a; j < z; j++) {
                    let w = uN[E0[j]]
                    dN[E0[j]] += w
                    if (X0[j] >= 0) {
                        dN[X0[j]] -= w
                        if (X0[j] > hi) {
                            hi = X0[j]
                        }
                    }
                }
                let cur = 0
                let pk = 0
                for (let i = lo; i <= hi; i++) {
                    cur += dN[i]
                    if (cur > pk) {
                        pk = cur
                    }
                    dN[i] = 0
                }
                if (pk > uPeak) {
                    uPeak = pk
                    uPeakFile = files[si]
                }
                if (pk > nu.uIni) {
                    uOver++
                    uOverTop.push([pk, si])
                    uOverTop.sort((a, b) => b[0] - a[0] || a[1] - b[1])
                    if (uOverTop.length > 10) {
                        uOverTop.pop()
                    }
                }
            }
        }
        mode[si] = md === 'long' ? 1 : -1
        tp[si] = rtp
        sl[si] = rsl
        nKeys[si] = cs.length
        lv[si] = Number(s.levelNumTrade) || 0
        fitNt[si] = Number(s.numTrade) || 0
        keyOff[si] = keyIdx.size()
        for (let c of cs) {
            keyIdx.push(c.ki)
            keySym.push(c.gt ? 1 : -1)
            keyTh.push(c.th)
            keyOr.push(c.isOr ? 1 : 0)
            rule.nConds++
            rule[c.gt ? 'nGt' : 'nLt']++
            rule.nOr += c.isOr ? 1 : 0
            cls[si * 3 + clsOf(keyNames[c.ki])]++
        }
        //一致性: 策略檔之 summary.numTrade 為求解窗 [timeOhlcStart, timeOhlcEnd] 內之進場數; 帳本同窗之進場數應相同
        //  下界(規則 R-EV-45): 起點棒 = timeOhlcStart 於本帳本之棒序; 未帶起點欄(undefined / null)者自第 0 根; 起點不在本帳本之棒上(早於帳本起點 ⇒ 帳本未涵蓋完整之求解窗; 或不在棒之開盤時間)
        //  者不納入核對(不計入 chk.n)、只計隻數(chk.skipped)—— 起點 = 帳本起點時結果與只計終點者逐位相同
        //  「未帶」= undefined 或 null; 其餘(含空字串、0、false 等非字串)而不在本帳本之棒上者一律不納入核對; 起點晚於終點者照核對(計數為 0, 與 numTrade 不同即計為不符)
        //  進場數之來源(排程第 564 項; Gitea #194 §6 第 2 點): 頂層之 numTrade(數字)為先; 頂層未帶(undefined 或 null)者以 summary.numTrade(數字)為後備(策略檔之形有兩種: 頂層、summary 內);
        //  兩者皆無(或頂層已帶而不是數字)者不納入核對(同改前; 不計入 compared、不計入 skipped); 後備使 meta.check.compared 增加(原: 無頂層 numTrade 者整隻不核對)
        let tEnd = s.summary && s.summary.timeOhlcEnd
        let iEnd = tEnd ? kpTimeInd.get(tEnd) : undefined
        let ntFile = s.numTrade === undefined || s.numTrade === null ? (s.summary && typeof s.summary.numTrade === 'number' ? s.summary.numTrade : undefined) : (typeof s.numTrade === 'number' ? s.numTrade : undefined)
        if (iEnd !== undefined && ntFile !== undefined) {
            let tStart = s.summary.timeOhlcStart
            let iStart = tStart === undefined || tStart === null ? 0 : kpTimeInd.get(tStart)
            if (iStart === undefined) {
                chk.skipped++
                if (chk.skippedFirst === null) {
                    chk.skippedFirst = { f: files[si], t: tStart }
                }
            }
            else {
                let e = ent.done()
                let cnt = 0
                for (let j = off[si]; j < e.length; j++) {
                    if (e[j] >= iStart && e[j] <= iEnd) {
                        cnt++
                    }
                }
                chk.n++
                if (cnt === ntFile) {
                    chk.same++
                }
                else if (chk.diffs.length < 10) {
                    chk.diffs.push({ f: files[si], ledger: cnt, file: ntFile })
                }
            }
        }
        if ((si + 1) % 10000 === 0) {
            log(`  策略 ${si + 1}/${ns}, 訂單 ${ent.size()}(${((Date.now() - t0) / 1000).toFixed(0)}s)`)
        }
    }
    off[ns] = ent.size()
    keyOff[ns] = keyIdx.size()
    //單一策略之峰值名義(規則 R-EV-43): 只記錄 —— 不設門檻、不中止(研究用本金之「超過即中止」只施於挑選後之持倉組合, 由評估器檢)
    if (nu) {
        log(`單一策略之全期最大同時持倉名義: 最大者 ${uPeakFile} ${uPeak}u;超過研究用本金 ${nu.uIni}u 者 ${uOver} 隻(只記錄、不中止;研究用本金之檢查只施於挑選後之持倉組合,由評估器檢)`)
    }

    //寫出
    fs.mkdirSync(outDir, { recursive: true })
    //重建於已有帳本之資料夾時, 先移除舊之 meta.json(最後才寫新者): 寫出中途失敗或中止者, 留下之半份帳本因無 meta 而載入即失敗,
    //  不致以舊 meta 配上新欄(新舊欄長度相同時, 無從由長度察覺; 例如序列欄)
    fs.rmSync(path.join(outDir, 'meta.json'), { force: true })
    let E = ent.done()
    let X = ext.done()
    let Pn = pnl.done()
    writeBin(path.join(outDir, 'off.u32'), off)
    writeBin(path.join(outDir, 'ent.i32'), E)
    writeBin(path.join(outDir, 'ext.i32'), X)
    writeBin(path.join(outDir, 'pnl.f32'), Pn)
    writeBin(path.join(outDir, 'mode.i8'), mode)
    writeBin(path.join(outDir, 'tp.f32'), tp)
    writeBin(path.join(outDir, 'sl.f32'), sl)
    writeBin(path.join(outDir, 'nKeys.u8'), nKeys)
    writeBin(path.join(outDir, 'lv.i32'), lv)
    writeBin(path.join(outDir, 'fitNt.i32'), fitNt)
    writeBin(path.join(outDir, 'cls.u8'), cls)
    writeBin(path.join(outDir, 'keyOff.u32'), keyOff)
    writeBin(path.join(outDir, 'keyIdx.u16'), keyIdx.done())
    writeBin(path.join(outDir, 'keySym.i8'), keySym.done())
    writeBin(path.join(outDir, 'keyTh.f64'), keyTh.done())
    writeBin(path.join(outDir, 'keyOr.u8'), keyOr.done())
    //因子序列欄(規則 R-EV-29; 供挑選函數以 ctx.factor 取決策棒以前之值): 列 = 因子(序同 keyNames)、行 = K 棒, fac[ki * nBars + i]; 缺值 NaN
    //  即觸發進場所用之同一份序列(已於 endTime 截斷), 原值不降精度 ⇒ 以三個條件欄與本欄重算之進場 = 帳本之進場
    let fac = new Float64Array(keyNames.length * n)
    for (let i = 0; i < keyNames.length; i++) {
        fac.set(aligned[i], i * n)
    }
    writeBin(path.join(outDir, 'fac.f64'), fac)
    //外部序列欄(規則 R-EV-34; 供挑選函數以 ctx.series 取決策棒以前之值): 列 = 序列(序同 meta.seriesNames)、行 = K 棒, ser[si * nBars + i]; 缺值 NaN
    //  未建入者不留檔: 重建於同一資料夾時移除舊檔(免得 meta.seriesNames 與欄不符)
    if (seriesNames.length) {
        let ser = new Float64Array(seriesNames.length * n)
        for (let i = 0; i < seriesNames.length; i++) {
            ser.set(alignedSer[i], i * n)
        }
        writeBin(path.join(outDir, 'ser.f64'), ser)
    }
    else {
        fs.rmSync(path.join(outDir, 'ser.f64'), { force: true })
    }
    //每根棒之基礎單位名義(規則 R-EV-43; 給 unit 者): unit.f64 = Float64Array(nBars); 未給者不留檔(重建於同一資料夾時移除舊檔)
    if (nu) {
        writeBin(path.join(outDir, 'unit.f64'), uN)
    }
    else {
        fs.rmSync(path.join(outDir, 'unit.f64'), { force: true })
    }
    if (rule.nSymOther > 0) {
        log(`注意: 比較符號非 '>' / '<' 之條件 ${rule.nSymOther} 個(觸發與 keySym 皆當 '<')`)
    }
    //標的之 K 線欄(開高低收量; 供挑選函數以 ctx.market 取決策棒以前之市況)
    let px = new Float64Array(n * 5)
    for (let i = 0; i < n; i++) {
        let r = arrOhlc[i]
        px[i * 5] = r.Open
        px[i * 5 + 1] = r.High
        px[i * 5 + 2] = r.Low
        px[i * 5 + 3] = r.Close
        px[i * 5 + 4] = Number(r.Volumn) || 0
    }
    writeBin(path.join(outDir, 'px.f64'), px)
    fs.writeFileSync(path.join(outDir, 'ids.json'), JSON.stringify(files), 'utf8')
    let h = crypto.createHash('sha1')
    h.update(`${LEDGER_VERSION}|${endTime}|${time0}|${n}|${ns}|${E.length}|`)
    h.update(files.join(','))
    h.update(Buffer.from(E.buffer, E.byteOffset, E.byteLength))
    //基礎單位(給 unit 者): 交易規則與緩衝入鍵(規則改變 ⇒ 鍵改變); 未給者不加 ⇒ 鍵與未加入本功能時相同
    if (nu) {
        h.update(`|unit|${unitKeyOf(nu)}`)
    }
    let key = h.digest('hex').slice(0, 16)
    let meta = {
        version: LEDGER_VERSION,
        key,
        builtAt: new Date().toISOString(),
        libraryRoot,
        keyOhlc,
        endTime,
        ...(hasStart ? { startTime } : {}),
        time0,
        barHours,
        nBars: n,
        nStrategies: ns,
        nOrders: E.length,
        nUnsettled,
        nFirstTouch: kpFt.size,
        check: { compared: chk.n, same: chk.same, diffs: chk.diffs, ...(chk.skipped > 0 ? { skipped: chk.skipped } : {}) },
        rule,
        keyNames,
        seriesNames,
        //基礎單位(給 unit 者才有此二欄): unit = 交易規則、緩衝與研究用本金(其他標的之帳本自帶其交易規則; 評估器依此欄與 unit.f64);
        //  unitStat = 每根棒之名義之範圍、單一策略之全期最大同時持倉名義之最大者與其策略、其超過研究用本金之隻數(nStrategyOverUIni; 恰等於者不算)與峰值最高之至多 10 隻之檔名(strategyOverFiles)
        //    只記錄於摘要(不另存逐策略之欄); 以帳本全期算得 ⇒ 不交給挑選函數與擴充診斷(loadLedger 之 L.meta 雖帶之, 評估器之 ctx.meta / api.meta 不含)
        ...(nu
            ? {
                unit: { rule: { ...nu.rule }, buffer: nu.buffer, uIni: nu.uIni },
                unitStat: { notionalMin: uN.reduce((a, v) => Math.min(a, v), Infinity), notionalMax: uN.reduce((a, v) => Math.max(a, v), -Infinity), strategyPeakNotional: uPeak, strategyPeakFile: uPeakFile, nStrategyOverUIni: uOver, strategyOverFiles: uOverTop.map(([, si]) => files[si]) },
            }
            : {}),
        buildSec: Number(((Date.now() - t0) / 1000).toFixed(1)),
    }
    fs.writeFileSync(path.join(outDir, 'meta.json'), JSON.stringify(meta, null, 1), 'utf8')
    if (chk.skipped > 0) {
        //記錄(只供閱讀, 不入 meta): 附第一個被略過者之檔名與其起點欄(JSON 字串化後截至 40 字; 字串格式與 K 線不同者即由此看出); 全數略過(一隻也未受核對)者以 WARN 起首
        let tfs = String(JSON.stringify(chk.skippedFirst.t)).slice(0, 40)
        log(`${chk.n === 0 ? 'WARN' : '注意:'} ${chk.skipped} 隻策略之求解窗起點(summary.timeOhlcStart)不在本帳本之棒上(帳本起點 ${time0}; 早於帳本起點者即帳本未涵蓋完整之求解窗, 或不在棒之開盤時間), 未納入一致性核對${chk.n === 0 ? '(全數略過 ⇒ 無一隻受核對)' : ''}; 第一個: ${chk.skippedFirst.f} 之 summary.timeOhlcStart = ${tfs}`)
    }
    log(`帳本完成: 策略 ${ns}, 訂單 ${E.length}(未結 ${nUnsettled}), first-touch 表 ${kpFt.size}, 一致性 ${chk.same}/${chk.n}, 鍵 ${key}, ${meta.buildSec}s → ${outDir}`)
    return meta
}
