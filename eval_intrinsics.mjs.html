//  挑選模組與擴充診斷模組與評估器同在一個 realm: 它們改得到內建物件(原型之方法、全域之建構子)。評估器若經可被改寫之方法處理帳本或自身之狀態,
//  被改寫之方法即可拿到帳本陣列本身(例: %TypedArray%.prototype.slice 被換掉 ⇒ 呼叫它之 this 就是帳本之欄)或改動評估器之狀態。對策兩層:
//  ① 擷取: 評估器用到之內建方法於本模組載入時(任何挑選模組與擴充診斷模組載入以前)擷取, 評估器只用擷取之參照(不經原型查找);

/**
 * 複製定型陣列 src 之 [a, z) 區段為 Ctor 之新陣列(只用擷取之參照: 以建構子於 src 之緩衝建視圖再以 set 複製位元組; 不經 slice / subarray / from 與 species 建構子)
 * Ctor 須與 src 同型(位元組逐一複製; NaN 之位元與原值相同); 區段超出 src 者拋錯(呼叫端之區段一律已夾在範圍內, 拋錯代表程式錯誤)
 *
 * @param {Function} Ctor 輸入定型陣列建構子(須為 BPE 所列之一, 且與 src 同型)
 * @param {TypedArray} src 輸入來源定型陣列
 * @param {Integer} a 輸入起點(含)
 * @param {Integer} z 輸入終點(不含)
 * @returns {TypedArray} 回傳新陣列(長度 max(0, z - a))
 * @example
 * let out = copyRange(Int32Array, Int32Array.from([1, 2, 3]), 1, 3)
 */
let copyRange = (Ctor, src, a, z) => {
    let len = z - a
    if (!(len > 0)) {
        return new Ctor(0)
    }
    let bpe = MapPrototypeGet(BPE, Ctor)
    if (bpe === undefined || a < 0 || z > TypedArrayPrototypeGetLength(src) || !NumberIsInteger(a) || !NumberIsInteger(z)) {
        throw new ErrorCtor(`copyRange: 區段不合法 ${a} ~ ${z}`)
    }
    let out = new Ctor(len)
    TypedArrayPrototypeSet(out, new Ctor(TypedArrayPrototypeGetBuffer(src), TypedArrayPrototypeGetByteOffset(src) + a * bpe, len))
    return out
}

/**
 * 複製整個定型陣列(copyRange 之全段; null 照回 null)
 *
 * @param {Function} Ctor 輸入定型陣列建構子(與 src 同型)
 * @param {TypedArray|null} src 輸入來源定型陣列
 * @returns {TypedArray|null} 回傳新陣列
 * @example
 * let out = copyAll(Float64Array, new Float64Array(3))
 */
let copyAll = (Ctor, src) => (src === null || src === undefined ? null : copyRange(Ctor, src, 0, TypedArrayPrototypeGetLength(src)))

/**
 * 複製 Buffer(或任一 Uint8Array)之位元組為獨立之 ArrayBuffer 後以 Ctor 檢視(帳本檔之讀入用; 只用擷取之參照, 不經 ArrayBuffer.prototype.slice 與其 species 建構子)
 *
 * @param {Uint8Array} b 輸入位元組
 * @param {Function} Ctor 輸入定型陣列建構子
 * @returns {TypedArray} 回傳 Ctor 之陣列(長度 = 位元組數 ÷ 每元素位元組數)
 * @example
 * let a = bytesAs(Buffer.from([0, 0, 128, 63]), Float32Array)
 */
let bytesAs = (b, Ctor) => {
    let n = TypedArrayPrototypeGetByteLength(b)
    let u = new Uint8ArrayCtor(n)
    TypedArrayPrototypeSet(u, new Uint8ArrayCtor(TypedArrayPrototypeGetBuffer(b), TypedArrayPrototypeGetByteOffset(b), n))
    return new Ctor(TypedArrayPrototypeGetBuffer(u))
}

//SNAP_KIND, tag 之快照可含之物件型別(依結構化複製後之原型; 皆在受保護清單內): 'freeze' = 凍結即不可變; 'mutable' = 有可變之內部狀態(凍結擋不住), 每次交出另給複本
//  其他原型(例: Blob 等宿主物件)不收 —— 其原型不在受保護清單, 比對之後之序列化會執行到其上之 toJSON

for (const p of [Object.prototype, Array.prototype, Boolean.prototype, Number.prototype, String.prototype, BigInt.prototype, Error.prototype, EvalError.prototype, RangeError.prototype, ReferenceError.prototype, SyntaxError.prototype, TypeError.prototype, URIError.prototype]) {

let descKey = (d) => {
    if (d === undefined) {
        return null
    }
    let acc = ObjectHasOwn(d, 'get')
    return { acc, value: acc ? undefined : d.value, get: acc ? d.get : undefined, set: acc ? d.set : undefined, flags: (acc ? 8 : (d.writable ? 1 : 0)) | (d.enumerable ? 2 : 0) | (d.configurable ? 4 : 0) }
}
let keyName = (k) => (typeof k === 'symbol' ? `[${SymbolPrototypeToString(k)}]` : k)
//pushOwn, 於陣列尾端定義自有之元素(不經 push 之 [[Set]]: 原型上同名之存取子〔例: Array.prototype 之 '0' 之 setter〕不被觸發); 比對與建錯誤時用
let pushOwn = (arr, v) => {
    ReflectDefineProperty(arr, arr.length, { __proto__: null, value: v, writable: true, enumerable: true, configurable: true })
}

//sameAsDesc, 快照之描述 x 與現行之屬性描述 d(Reflect.getOwnPropertyDescriptor 之回傳)是否相同(不另建物件; 快速路徑用)
let sameAsDesc = (x, d) => {
    if (d === undefined || x === null) {
        return false
    }
    let acc = ObjectHasOwn(d, 'get')
    if (acc !== x.acc || ((acc ? 8 : (d.writable ? 1 : 0)) | (d.enumerable ? 2 : 0) | (d.configurable ? 4 : 0)) !== x.flags) {
        return false
    }
    return acc ? (d.get === x.get && d.set === x.set) : ObjectIs(d.value, x.value)
}

//SNAP, 載入時之快照: 各根之 { label, obj, proto, ext, keys: Map(鍵 → 描述), order(鍵之順序), descs(與 order 對齊之描述), n }; 全域綁定之 Map(名稱 → 描述)
//  比對之快速路徑: Reflect.ownKeys 之順序固定(整數鍵、字串鍵依加入序、符號鍵依加入序), 未被改者逐位與 order 相同 ⇒ 依序比描述即可; 不同才走以鍵查表之慢速路徑
const SNAP = []
for (const [label, obj] of PROTECTED_ROOTS) {
    let keys = new MapCtor()
    let ks = ReflectOwnKeys(obj)
    let descs = []
    for (let i = 0; i < ks.length; i++) {
        let d = descKey(ReflectGetOwnPropertyDescriptor(obj, ks[i]))
        MapPrototypeSet(keys, ks[i], d)
        ArrayPrototypePush(descs, d)
    }
    ArrayPrototypePush(SNAP, { label, obj, proto: ReflectGetPrototypeOf(obj), ext: ReflectIsExtensible(obj), keys, order: ks, descs, n: ks.length })
}
const SNAP_GLOBAL = new MapCtor()
for (const name of GLOBAL_NAMES) {
    ReflectGet(GLOBAL, name) //存取一次: 延遲定義之全域綁定先定型再記錄
    MapPrototypeSet(SNAP_GLOBAL, name, descKey(ReflectGetOwnPropertyDescriptor(GLOBAL, name)))
}
const N_ITEMS = SNAP.reduce((a, r) => a + r.n + 2, 0) + GLOBAL_NAMES.length

/**
 * 比對受保護之內建物件清單與載入時之快照
 * 只用擷取之參照(比對本身不經可被改寫之方法; 讀描述不觸發存取子)
 *
 * @returns {Array} 回傳被改之項目字串陣列(例: '%TypedArray%.prototype.slice'、'Array.prototype.foo(新增)'、'globalThis.Math'); 空陣列代表一致
 * @example
 * let items = checkIntrinsics()
 */
let checkIntrinsics = () => {
    let bad = []
    for (let i = 0; i < SNAP.length; i++) {
        let r = SNAP[i]
        let obj = r.obj
        if (ReflectGetPrototypeOf(obj) !== r.proto) {
            pushOwn(bad, `${r.label}(原型)`)
        }
        if (ReflectIsExtensible(obj) !== r.ext) {
            pushOwn(bad, `${r.label}(可擴充性)`)
        }
        let ks = ReflectOwnKeys(obj)
        if (ks.length === r.n) {
            let ok = true
            let order = r.order
            let descs = r.descs
            for (let j = 0; j < ks.length; j++) {
                if (ks[j] !== order[j] || !sameAsDesc(descs[j], ReflectGetOwnPropertyDescriptor(obj, ks[j]))) {
                    ok = false
                    break
                }
            }
            if (ok) {
                continue
            }
        }
        let seen = 0
        for (let j = 0; j < ks.length; j++) {
            let k = ks[j]
            if (!MapPrototypeHas(r.keys, k)) {
                pushOwn(bad, `${r.label}.${keyName(k)}(新增)`)
                continue
            }
            seen++
            if (!sameAsDesc(MapPrototypeGet(r.keys, k), ReflectGetOwnPropertyDescriptor(obj, k))) {
                pushOwn(bad, `${r.label}.${keyName(k)}`)
            }
        }
        if (seen !== r.n) {
            let it = MapPrototypeKeys(r.keys)
            for (let x = MapIteratorNext(it); !x.done; x = MapIteratorNext(it)) {
                if (ReflectGetOwnPropertyDescriptor(obj, x.value) === undefined) {
                    pushOwn(bad, `${r.label}.${keyName(x.value)}(刪除)`)
                }
            }
        }
    }
    let it = MapPrototypeKeys(SNAP_GLOBAL)
    for (let x = MapIteratorNext(it); !x.done; x = MapIteratorNext(it)) {
        let d0 = MapPrototypeGet(SNAP_GLOBAL, x.value)
        let d1 = ReflectGetOwnPropertyDescriptor(GLOBAL, x.value)
        if (d0 === null ? d1 !== undefined : !sameAsDesc(d0, d1)) {
            pushOwn(bad, `globalThis.${x.value}`)
        }
    }
    return bad
}

//INTEGRITY_ERRORS, 本模組所建之完整性錯誤(以 WeakSet 辨識; 模組自行拋出之同 code 物件不算)
const INTEGRITY_ERRORS = new WeakSetCtor()

//defineData, 以無原型之描述物件定義自有資料屬性(不觸發原型上之存取子)
let defineData = (o, k, v) => {
    ReflectDefineProperty(o, k, { __proto__: null, value: v, writable: true, enumerable: k !== 'stack' && k !== 'message' && k !== 'name', configurable: true })
}

/**
 * 建立完整性錯誤(code 為 EVAL_INTEGRITY; where = 偵測之時點; items = 被改之項目)
 * 錯誤之 message、stack 皆為自有之資料屬性(讀取時不經 Error.prepareStackTrace)
 *
 * @param {String} where 輸入偵測之時點(例: 'pick'、'accept'、'ext'、'pick-load')
 * @param {Array} items 輸入被改之項目字串陣列
 * @param {String} [note] 輸入說明(選配; 呼叫端所處置之說明, 例: 更新已中止), 置於訊息之開頭 EVAL_INTEGRITY 之後; 未給者訊息不變
 * @returns {Error} 回傳錯誤物件
 * @example
 * let e = integrityError('pick', ['Math.floor'])
 */
let integrityError = (where, items, note) => {
    //只用擷取之參照(此時內建物件已被改): 逐一複製, 不用 slice / 展開 / Array.from(後兩者經可被改寫之迭代器)
    let all = []
    let shown = []
    for (let i = 0; i < items.length; i++) {
        pushOwn(all, StringCtor(items[i]))
        if (i < 12) {
            pushOwn(shown, StringCtor(items[i]))
        }
    }
    if (items.length > 12) {
        pushOwn(shown, `…等 ${items.length} 項`)
    }
    let msg = `EVAL_INTEGRITY ${note === undefined ? '' : `${StringCtor(note)}: `}評估器之內建物件被改(於 ${where}; 違反合約:不得改寫內建物件): ${ArrayPrototypeJoin(shown, '、')}`
    let e = new ErrorCtor(msg)
    defineData(e, 'stack', `Error: ${msg}`)
    defineData(e, 'code', 'EVAL_INTEGRITY')
    defineData(e, 'where', where)
    defineData(e, 'items', ObjectFreeze(all))
    //凍結: 模組攔下此錯誤後改不了其 code / where / items / message(黏著偵測其後轉拋之即同一物件)
    ObjectFreeze(e)
    WeakSetPrototypeAdd(INTEGRITY_ERRORS, e)
    return e
}

/**
 * 比對受保護之內建物件; 不一致即拋完整性錯誤(code 為 EVAL_INTEGRITY)
 *
 * @param {String} where 輸入時點標示(寫入錯誤之 where 與訊息)
 * @param {String} [note] 輸入說明(選配; 見 integrityError)
 * @returns {undefined} 一致時無回傳
 * @example
 * assertIntrinsics('pick-load')
 */
//VIOL, 本行程內偵測到之次數與最後一次之錯誤: 呼叫端記下呼叫模組前之 violationMark(), 呼叫後以 violationSince(mark) 查其間有無偵測
//  (模組攔下評估器於其執行期間拋出之完整性錯誤〔例: 延遲讀取前之比對〕並改回者, 該次評估仍失敗)
const VIOL = { seq: 0, last: null }
let assertIntrinsics = (where, note) => {
    let bad = checkIntrinsics()
    if (bad.length) {
        let e = integrityError(where, bad, note)
        VIOL.seq++
        VIOL.last = e
        throw e
    }
}

/**
 * 記下目前之偵測次數(與 violationSince 成對使用)
 *
 * @returns {Integer} 回傳目前之偵測次數
 * @example
 * let mk = violationMark()
 */
let violationMark = () => VIOL.seq

/**
 * mark 以後有無偵測到被改(含被模組攔下者)
 *
 * @param {Integer} mark 輸入 violationMark 之回傳值
 * @returns {Error|null} 回傳其間最後一次之完整性錯誤, 無則 null
 * @example
 * let e = violationSince(mk)
 */
let violationSince = (mark) => (VIOL.seq !== mark ? VIOL.last : null)

/**
 * 是否為本模組所建之完整性錯誤(以 WeakSet 辨識, 不看 code 字串)
 *
 * @param {*} e 輸入任意值
 * @returns {Boolean} 回傳是否為完整性錯誤
 * @example
 * let b = isIntegrityError(err)
 */
let isIntegrityError = (e) => e !== null && (typeof e === 'object' || typeof e === 'function') && WeakSetPrototypeHas(INTEGRITY_ERRORS, e)

/**
 * 收下模組拋出之值: 只取其名稱、訊息、堆疊與 code 之字串(可能經模組之存取子), 改以評估器自建之錯誤物件轉拋
 * 呼叫端之後讀錯誤之欄位不再執行模組之程式; name / message / stack / code 與原值相同(堆疊非字串者以 String(原值) 代之)
 *
 * @param {*} e 輸入模組拋出之值
 * @returns {Error} 回傳評估器自建之錯誤物件(本模組之完整性錯誤原樣回傳)
 * @example
 * let e2 = adoptError(new TypeError('x'))
 */
let adoptError = (e) => {
    if (isIntegrityError(e)) {
        return e
    }
    let name = 'Error'
    let message = ''
    let stack = null
    let code
    if (e !== null && (typeof e === 'object' || typeof e === 'function')) {
        try {
            let v = e.name
            name = typeof v === 'string' ? v : name
            v = e.message
            message = typeof v === 'string' ? v : (v === undefined ? '' : StringCtor(v))
            v = e.stack
            stack = typeof v === 'string' ? v : null
            v = e.code
            code = typeof v === 'string' || typeof v === 'number' ? v : undefined
        }
        catch (err) {
            message = message || '(無法讀取之錯誤)'
        }
        if (!message && stack === null) {
            try {
                message = StringCtor(e)
            }
            catch (err) {
                message = '(無法讀取之錯誤)'
            }
        }
    }
    else {
        message = typeof e === 'symbol' ? SymbolPrototypeToString(e) : StringCtor(e)
        stack = message
    }
    let x = new ErrorCtor(message)
    defineData(x, 'name', name)
    defineData(x, 'stack', stack === null ? `${name}: ${message}` : stack)
    if (code !== undefined) {
        defineData(x, 'code', code)
    }
    return x
}

/**
 * 取資料之快照(挑選函數回傳之 tag 等): 以結構化複製取得獨立之複本並深度凍結
 * 原始值照回; 含函數、符號等無法結構化複製者拋錯; 含共享記憶體(SharedArrayBuffer)者拋錯(複本仍與原值共用記憶體); 型別限 SNAP_KIND 所列(其他宿主物件拋錯)
 * 複本內有凍結擋不住之內部狀態者(定型陣列、ArrayBuffer、DataView、Map、Set、Date、RegExp)記為「可變」: 交出時以 handOut 每次另給複本
 *
 * @param {*} v 輸入任意值
 * @returns {*} 回傳凍結之複本(原始值照回)
 * @example
 * let t = snapData({ side: 'long', n: [1, 2] })
 */
const MUTABLE_SNAPS = new WeakSetCtor()
let isBrand = (fn, x) => {
    try {
        fn(x)
        return true
    }
    catch (err) {
        return false
    }
}
let snapData = (v) => {
    if (v === null || (typeof v !== 'object' && typeof v !== 'function')) {
        return v
    }
    let c = StructuredClone(v)
    let mutable = false
    let seen = new WeakSetCtor()
    let walk = (o) => {
        if (o === null || typeof o !== 'object' || WeakSetPrototypeHas(seen, o)) {
            return
        }
        WeakSetPrototypeAdd(seen, o)
        //型別白名單(SNAP_KIND): 依結構化複製後之原型(複本之原型皆為本 realm 之內建原型)
        let pr = ReflectGetPrototypeOf(o)
        let kind = pr === null ? 'freeze' : MapPrototypeGet(SNAP_KIND, pr)
        if (kind === undefined) {
            throw new ErrorCtor('含不支援之型別(只收數值、字串、布林、物件、陣列、Map、Set、Date、RegExp、定型陣列、ArrayBuffer、DataView 與錯誤物件)')
        }
        if (kind === 'mutable') {
            mutable = true
        }
        let buf = null
        if (ArrayBufferIsView(o)) {
            buf = isBrand(TypedArrayPrototypeGetLength, o) ? TypedArrayPrototypeGetBuffer(o) : DataViewPrototypeGetBuffer(o)
        }
        else if (isBrand(ArrayBufferPrototypeGetByteLength, o)) {
            buf = o
        }
        if (SharedArrayBufferPrototypeGetByteLength && (isBrand(SharedArrayBufferPrototypeGetByteLength, o) || (buf !== null && isBrand(SharedArrayBufferPrototypeGetByteLength, buf)))) {
            throw new ErrorCtor('含共享記憶體(SharedArrayBuffer)之資料不能取快照')
        }
        //Map / Set 之內容不是自有屬性: 逐一走訪其鍵與值(其中之共享記憶體同樣拒收)
        if (isBrand(MapPrototypeGetSize, o)) {
            let it = MapPrototypeKeys(o)
            for (let x = MapIteratorNext(it); !x.done; x = MapIteratorNext(it)) {
                walk(x.value)
            }
            it = MapPrototypeValues(o)
            for (let x = MapIteratorNext(it); !x.done; x = MapIteratorNext(it)) {
                walk(x.value)
            }
        }
        else if (isBrand(SetPrototypeGetSize, o)) {
            let it = SetPrototypeValues(o)
            for (let x = SetIteratorNext(it); !x.done; x = SetIteratorNext(it)) {
                walk(x.value)
            }
        }
        let ks = ReflectOwnKeys(o)
        for (let i = 0; i < ks.length; i++) {
            let d = ReflectGetOwnPropertyDescriptor(o, ks[i])
            if (d && ObjectHasOwn(d, 'value')) {
                walk(d.value)
            }
        }
        if (kind === 'freeze') {
            ObjectFreeze(o)
        }
    }
    walk(c)
    if (mutable) {
        WeakSetPrototypeAdd(MUTABLE_SNAPS, c)
    }
    return c
}

/**
 * 交出快照: 凍結之快照照交; 記為可變者(見 snapData)另給一份複本
 *
 * @param {*} v 輸入 snapData 之回傳值
 * @returns {*} 回傳可交給模組之值
 * @example
 * let t = handOut(snapData({ a: 1 }))
 */
let handOut = (v) => (v !== null && typeof v === 'object' && WeakSetPrototypeHas(MUTABLE_SNAPS, v) ? StructuredClone(v) : v)

/**
 * 序列化為 JSON 文字而不經原型上之 toJSON(內建物件可能已被改時寫結果檔與紀錄用)
 * 只走訪自有、可列舉之資料屬性(陣列之索引、物件之字串鍵);原始值以擷取之 JSON.stringify(原始值不查 toJSON)
 * 輸出與 JSON.stringify(v) / JSON.stringify(v, null, 2) 對一般資料(物件、陣列、字串、數值、布林、null)相同;值為 undefined 之物件欄略過、陣列元素為 null
 *
 * @param {*} v 輸入資料
 * @param {Boolean} [pretty=false] 輸入是否縮排 2 格,預設false
 * @returns {String} 回傳 JSON 文字
 * @example
 * let s = safeJson({ ok: false, items: ['a'] }, true)
 */
let safeJson = (v, pretty = false, ind = '') => {
    if (v === null || typeof v !== 'object') {
        let s = typeof v === 'function' || typeof v === 'symbol' ? undefined : JSONStringify(v)
        return s === undefined ? 'null' : s
    }
    let inner = pretty ? `${ind}  ` : ''
    let parts = []
    if (ArrayIsArray(v)) {
        for (let i = 0; i < v.length; i++) {
            let d = ReflectGetOwnPropertyDescriptor(v, i)
            pushOwn(parts, inner + safeJson(d && ObjectHasOwn(d, 'value') ? d.value : null, pretty, inner))
        }
        return parts.length ? (pretty ? `[\n${ArrayPrototypeJoin(parts, ',\n')}\n${ind}]` : `[${ArrayPrototypeJoin(parts, ',')}]`) : '[]'
    }
    let ks = ReflectOwnKeys(v)
    for (let i = 0; i < ks.length; i++) {
        let k = ks[i]
        let d = typeof k === 'string' ? ReflectGetOwnPropertyDescriptor(v, k) : undefined
        if (!d || !d.enumerable || !ObjectHasOwn(d, 'value') || d.value === undefined || typeof d.value === 'function' || typeof d.value === 'symbol') {
            continue
        }
        pushOwn(parts, `${inner}${JSONStringify(k)}:${pretty ? ' ' : ''}${safeJson(d.value, pretty, inner)}`)
    }
    return parts.length ? (pretty ? `{\n${ArrayPrototypeJoin(parts, ',\n')}\n${ind}}` : `{${ArrayPrototypeJoin(parts, ',')}}`) : '{}'
}

let isMutableSnap = (v) => v !== null && typeof v === 'object' && WeakSetPrototypeHas(MUTABLE_SNAPS, v)
