import fs from 'fs'
import path from 'path'
import { spawnSync } from 'child_process'
import { rsiNumOf } from './rsiNums.mjs'
import { isPlain, optOf, fnReq, fnOpt, strReq, numReq } from './inputs.mjs'


//多輪執行之通用驅動(不含領域用語): 一次執行(run)連跑數輪, 每輪後回報並依條件決定是否續跑
//  「每輪幾個節點、同時幾個、一次跑幾輪」由使用端依自身設備與監督方式給定:
//    resolveRunConfig(cfg, cli) → { nodesPerRound, parallelNodes, roundsPerRun }
//      cfg  設定檔之值(可缺); cli 命令列覆寫(可缺); 皆須為正整數, parallelNodes 不大於 nodesPerRound
//  runRounds({ rounds, runRound, shouldStop, onRoundDone, log }) → { done: [各輪之結果], stopped: 提前停止之原因 | null }
//    runRound({ index })      跑一輪; 回該輪之結果物件; 結果之 halt(字串)非空 ⇒ 該輪後停止(例: 輪末回調失敗, 須人處理後才續跑)
//    shouldStop({ index, done })  每輪開始前詢問; 回非空字串 ⇒ 不再起新輪(例: 停止檔存在、額度不足、資源持續超限)
//    onRoundDone(result, { index, rounds })  每輪結束後之回調(回報、寫摘要)
//  無人值守(由排程器定時觸發同一道起跑指令)所需之三件事 —— 皆以檔案為介面, 不依賴任何常駐服務:
//    acquireRunLock(fpLock, opt)   單一執行之鎖: 同一狀態根同時只允許一次執行; 上一次尚未結束時, 本次即不啟動
//    appendEvent(fpEvents, obj) / readEvents(fpEvents)   事件檔(每列一個 JSON; 只增): 起跑、每輪結束、整次執行結束、未啟動(及其原因)
//      監看端(人、代理、其他程式)只須等待此檔出現新的結束類事件, 不必同時看狀態檔與行程
//    trailingFailures(events)      事件檔末尾連續失敗之輪數(起跑前自檢用: 連續失敗達上限即不再自動起跑, 等人處理)
//posInt: 正整數(數字或其數字字串, 轉為數值後使用; 排程第 595 項, K4、K7: 原 Number.isInteger 使 '4' 被當成未給); 無效回 null(用預設)
let posInt = (v) => rsiNumOf(v, 'posInt')

/**
 * 解析每輪節點數、同時節點數與每次執行輪數(命令列優先於設定檔,皆須為正整數〔數字字串轉為數值〕,否則用預設;cfg、cli 為 null 或非物件者視為未給)
 *
 * @param {Object} [cfg={}] 輸入設定檔之值,預設{}
 * @param {Object} [cli={}] 輸入命令列覆寫值,預設{}
 * @returns {Object} 回傳 { nodesPerRound(預設3), parallelNodes(預設為 nodesPerRound,且不大於 nodesPerRound), roundsPerRun(預設1) }
 * @example
 * let c = resolveRunConfig({ nodesPerRound: 4 }, { parallelNodes: 2 })
 * console.log(c)
 * // => { nodesPerRound: 4, parallelNodes: 2, roundsPerRun: 1 }
 */
let resolveRunConfig = (cfg = {}, cli = {}) => {
    //cfg、cli 為選填之物件(跨套件規格 V4、K3): null、非物件視為未給(原: null 於讀鍵時拋原生 TypeError)
    let cfgO = cfg !== null && typeof cfg === 'object' ? cfg : {}
    let cliO = cli !== null && typeof cli === 'object' ? cli : {}
    let pick = (k, def) => posInt(cliO[k]) || posInt(cfgO[k]) || def
    let nodesPerRound = pick('nodesPerRound', 3)
    let parallelNodes = Math.min(nodesPerRound, pick('parallelNodes', nodesPerRound))
    let roundsPerRun = pick('roundsPerRun', 1)
    return { nodesPerRound, parallelNodes, roundsPerRun }
}

/**
 * 多輪執行之通用驅動:連跑數輪,每輪前詢問是否停止、每輪後回調,結果 halt 非空即於該輪後停止
 *
 * @param {Object} opt 輸入設定物件
 * @param {Number|String} [opt.rounds=1] 輸入最多跑幾輪(非負整數或其數字字串),預設1;給了而無效(小數、負數、布林、非數字字串)reject
 * @param {Function} opt.runRound 輸入跑一輪之函數,形式 async ({ index }) → 該輪結果物件,結果之 halt 非空則於該輪後停止;非函數 reject
 * @param {Function} [opt.shouldStop=()=>null] 輸入每輪開始前之詢問函數,形式 async ({ index, done }),回傳非空字串即不再起新輪,預設回傳null;給了而非函數 reject(停止之閘不靜默失效)
 * @param {Function} [opt.onRoundDone=()=>{}] 輸入每輪結束後之回調,形式 async (result, { index, rounds }),預設空函數;非函數視為未給
 * @param {Function} [opt.log=()=>{}] 輸入記錄函數,預設空函數;非函數視為未給
 * @returns {Promise<Object>} 回傳 Promise,resolve 為 { done(各輪結果陣列), stopped(提前停止之原因字串,未提前停止為null) };輸入無效時 reject Error(invalid opt…: must be …,固定訊息、不帶輸入值),先於任何一輪
 * @example
 * let r = await runRounds({ rounds: 3, runRound: async ({ index }) => ({ index, halt: index === 1 ? '停' : null }) })
 * console.log(r)
 * // => { done: [ { index: 0, halt: null }, { index: 1, halt: '停' } ], stopped: '停' }
 */
let runRounds = async (opt) => {
    //輸入之檢測(跨套件規格 V4、§5.1、K3、K5; R-EV-54; 回傳 Promise 者以 reject 通知, 先於任何一輪):
    //  opt 非一般物件、runRound 非函數 ⇒ reject invalid opt / opt.runRound: must be …(原: 原生 TypeError); rounds 為規模(與 --rounds 同一量)⇒ 未給(undefined 或 null)= 1, 給了須為非負整數(數字字串轉為數值),
    //  否則 reject invalid opt.rounds: must be a non-negative integer(原: true ⇒ 1 輪、2.5 ⇒ 3 輪、'abc' ⇒ 0 輪仍成功); shouldStop 為停止之閘(額度不足、資源持續超限)⇒ 給了而非函數即 reject(原: 靜默無閘, null 則 TypeError);
    //  onRoundDone、log 為觀察用 ⇒ 非函數視為未給(K3)
    let o = optOf(opt)
    let runRound = fnReq(o.runRound, 'opt.runRound')
    let rounds = numReq(o.rounds, 1, 'opt.rounds', 'nonNegInt')
    let shouldStop = fnOpt(o.shouldStop, 'opt.shouldStop') || (() => null)
    let onRoundDone = typeof o.onRoundDone === 'function' ? o.onRoundDone : () => {}
    let log = typeof o.log === 'function' ? o.log : () => {}
    let done = []
    for (let i = 0; i < rounds; i++) {
        let why = await shouldStop({ index: i, done })
        if (why) {
            log(`不再起新輪(已跑 ${done.length} / ${rounds}): ${why}`)
            return { done, stopped: String(why) }
        }
        let r = await runRound({ index: i })
        done.push(r)
        await onRoundDone(r, { index: i, rounds })
        if (r && r.halt) {
            log(`本次執行於第 ${i + 1} / ${rounds} 輪後停止: ${r.halt}`)
            return { done, stopped: String(r.halt) }
        }
    }
    return { done, stopped: null }
}

//pidAlive, 行程是否仍在(訊號 0 只探測、不送訊號); EPERM = 存在但無權限
let pidAlive = (pid) => {
    try {
        process.kill(pid, 0)
        return true
    }
    catch (err) {
        return !!(err && err.code === 'EPERM')
    }
}

//procNameOf, 行程之映像名(小寫); 查不到回 null。用於辨認「鎖檔之 pid 已被別的程式重用」(機器重開後常見)
let procNameOf = (pid) => {
    try {
        if (process.platform === 'win32') {
            let r = spawnSync('tasklist', ['/FI', `PID eq ${pid}`, '/FO', 'CSV', '/NH'], { windowsHide: true, encoding: 'utf8' })
            let m = /^"([^"]+)"/m.exec(String(r.stdout || ''))
            return m ? m[1].toLowerCase() : null
        }
        return fs.readFileSync(`/proc/${pid}/comm`, 'utf8').trim().toLowerCase()
    }
    catch (err) {
        return null
    }
}

//acquireRunLock, 以「獨占建立」寫鎖檔 { pid, startedAt, beatAt }
//  opt: { staleMs(鎖之心跳逾此時間視為疑似卡住; 預設 6 小時), isSameProgram(name)(判斷鎖檔之 pid 是否仍為本程式; 預設映像名以 node 開頭), pid(測試用) }
//  回傳:
//    { ok: true, release(), beat() }           取得; release 只刪自己的鎖; beat 更新心跳(每輪開始與結束時呼叫)
//    { ok: false, why: 'running', pid, startedAt, beatAt }   另一個執行仍在進行
//    { ok: false, why: 'stale', pid, startedAt, beatAt }     鎖之行程仍在而心跳已逾時(疑似卡住)—— 不接手, 等人處理
//  鎖檔之行程已不在、或其 pid 已被別的程式重用 ⇒ 視為殘留, 刪除後接手
/**
 * 取得單一執行之鎖(以獨占建立鎖檔,同一狀態根同時只允許一次執行)
 *
 * 鎖檔之行程已不在或其 pid 已被別的程式重用時視為殘留,刪除後接手;其餘情形見回傳說明。
 * fpLock 非非空字串、opt 非一般物件(null、undefined 視為未給)⇒ 同步拋 invalid fpLock / opt: must be …(固定訊息;先於任何副作用)。
 *
 * @param {String} fpLock 輸入鎖檔路徑
 * @param {Object} [opt={}] 輸入設定物件,預設{}
 * @param {Number} [opt.staleMs=21600000] 輸入心跳逾此毫秒視為疑似卡住,預設21600000(6 小時)
 * @param {Function} [opt.isSameProgram] 輸入判斷鎖檔 pid 是否仍為本程式之函數,形式 (name) → Boolean,預設為映像名為null或以 node 開頭
 * @param {Number} [opt.pid] 輸入本方 pid(測試用),預設process.pid
 * @returns {Object} 取得時回傳 { ok: true, release(只刪自己的鎖), beat(更新心跳) };未取得回傳 { ok: false, why: 'running' 或 'stale', pid, startedAt, beatAt }
 * @example
 * let lock = acquireRunLock('./state/run.lock')
 * if (lock.ok) {
 *     lock.beat()
 *     lock.release()
 * }
 */
let acquireRunLock = (fpLock, opt = {}) => {
    //輸入之檢測(§5.1、V4; 同步且有副作用者 ⇒ 先於任何副作用拋固定訊息): fpLock 須為非空字串(原: 非字串於 mkdirSync 前之 path.dirname 拋原生 TypeError)、opt 本身須為一般物件(null、undefined 視為未給)
    //  opt.staleMs(只與時間差比較, 不是計時器)、opt.pid 無效者用預設(K3); opt.isSameProgram 非函數者用預設
    strReq(fpLock, 'fpLock')
    let o = optOf(opt)
    let staleMs = Number.isFinite(o.staleMs) && o.staleMs > 0 ? o.staleMs : 6 * 3600000
    let isSame = typeof o.isSameProgram === 'function' ? o.isSameProgram : (name) => name === null || /^node/.test(name)
    let me = Number.isInteger(o.pid) ? o.pid : process.pid
    fs.mkdirSync(path.dirname(fpLock), { recursive: true })
    let read = () => {
        try {
            return JSON.parse(fs.readFileSync(fpLock, 'utf8'))
        }
        catch (err) {
            return null
        }
    }
    for (let i = 0; i < 2; i++) {
        try {
            let fd = fs.openSync(fpLock, 'wx')
            let now = new Date().toISOString()
            fs.writeSync(fd, JSON.stringify({ pid: me, startedAt: now, beatAt: now }))
            fs.closeSync(fd)
            let mine = () => {
                let j = read()
                return j && j.pid === me ? j : null
            }
            return {
                ok: true,
                release: () => {
                    if (mine()) {
                        try {
                            fs.rmSync(fpLock, { force: true })
                        }
                        catch (err) {}
                    }
                },
                beat: () => {
                    let j = mine()
                    if (j) {
                        try {
                            fs.writeFileSync(fpLock, JSON.stringify({ ...j, beatAt: new Date().toISOString() }))
                        }
                        catch (err) {}
                    }
                },
            }
        }
        catch (err) {
            if (!err || err.code !== 'EEXIST') {
                throw err
            }
            let j = read()
            if (j && Number.isInteger(j.pid) && j.pid !== me && pidAlive(j.pid) && isSame(procNameOf(j.pid))) {
                let beat = Date.parse(j.beatAt || j.startedAt || '')
                let why = Number.isFinite(beat) && Date.now() - beat > staleMs ? 'stale' : 'running'
                return { ok: false, why, pid: j.pid, startedAt: j.startedAt || null, beatAt: j.beatAt || null }
            }
            //殘留之鎖(行程已不在、pid 已被別的程式重用、或鎖檔損壞): 刪除後重試一次
            try {
                fs.rmSync(fpLock, { force: true })
            }
            catch (e2) {}
        }
    }
    return { ok: false, why: 'running', pid: null, startedAt: null, beatAt: null }
}

//peekRunLock, 只看不動: 鎖檔所記之執行是否仍在進行(供起跑包裝在產生子行程之前先判斷, 以免覆寫進行中那一次之 pid 檔與日誌路徑)
//  回 { running, stale, pid, startedAt, beatAt }; 無鎖檔、行程已不在、或 pid 已被別的程式重用 ⇒ running = false
/**
 * 只看不動:鎖檔所記之執行是否仍在進行
 * fpLock 非非空字串、opt 非一般物件(null、undefined 視為未給)⇒ 同步拋 invalid fpLock / opt: must be …(原: 無效之路徑回 running 為 false, 使守門靜默放行)。
 *
 * @param {String} fpLock 輸入鎖檔路徑
 * @param {Object} [opt={}] 輸入設定物件,預設{}
 * @param {Number} [opt.staleMs=21600000] 輸入心跳逾此毫秒視為疑似卡住,預設21600000(6 小時)
 * @param {Function} [opt.isSameProgram] 輸入判斷鎖檔 pid 是否仍為本程式之函數,形式 (name) → Boolean,預設為映像名為null或以 node 開頭
 * @returns {Object} 回傳 { running, stale, pid, startedAt, beatAt };無鎖檔、行程已不在或 pid 已被別的程式重用時 running 為 false
 * @example
 * let s = peekRunLock('./state/run.lock')
 * console.log(s.running)
 */
let peekRunLock = (fpLock, opt = {}) => {
    //輸入之檢測(§5.1、V4): fpLock 須為非空字串、opt 須為一般物件(null、undefined 視為未給); 原: 非字串者於讀檔時拋錯而被吞 ⇒ 回 running = false(「輪次進行中不得執行」之守門於無效之路徑下靜默放行)
    strReq(fpLock, 'fpLock')
    let o = optOf(opt)
    let staleMs = Number.isFinite(o.staleMs) && o.staleMs > 0 ? o.staleMs : 6 * 3600000
    let isSame = typeof o.isSameProgram === 'function' ? o.isSameProgram : (name) => name === null || /^node/.test(name)
    let j = null
    try {
        j = JSON.parse(fs.readFileSync(fpLock, 'utf8'))
    }
    catch (err) {}
    if (!j || !Number.isInteger(j.pid) || !pidAlive(j.pid) || !isSame(procNameOf(j.pid))) {
        return { running: false, stale: false, pid: j && Number.isInteger(j.pid) ? j.pid : null, startedAt: (j && j.startedAt) || null, beatAt: (j && j.beatAt) || null }
    }
    let beat = Date.parse(j.beatAt || j.startedAt || '')
    return { running: true, stale: Number.isFinite(beat) && Date.now() - beat > staleMs, pid: j.pid, startedAt: j.startedAt || null, beatAt: j.beatAt || null }
}

//appendEvent, 於事件檔追加一列(自動帶 at); 寫入失敗不拋(事件檔不得影響執行本身)
/**
 * 於事件檔追加一列 JSON(自動帶 at 時間戳;寫入失敗不拋錯;fpEvents 非非空字串或 obj 非一般物件者回 false、不寫任何檔)
 *
 * @param {String} fpEvents 輸入事件檔路徑,資料夾不存在時自動建立
 * @param {Object} obj 輸入事件物件(其欄位置於 at 之後,同名欄位會覆蓋 at)
 * @returns {Boolean} 回傳是否寫入成功
 * @example
 * appendEvent('./state/events.jsonl', { event: 'round-done', ok: true })
 * // => true
 */
let appendEvent = (fpEvents, obj) => {
    //輸入之檢測(§5.1: 回傳結果者回失敗之結果; 本函數之結果為 Boolean): fpEvents 非非空字串、obj 非一般物件 ⇒ 回 false、不寫任何檔(原: `appendEvent(path, 'abc')` 把字串展開成 { 0: 'a', 1: 'b', 2: 'c' } 寫入並回 true)
    if (typeof fpEvents !== 'string' || fpEvents === '' || !isPlain(obj)) {
        return false
    }
    try {
        fs.mkdirSync(path.dirname(fpEvents), { recursive: true })
        fs.appendFileSync(fpEvents, `${JSON.stringify({ at: new Date().toISOString(), ...obj })}\n`, 'utf8')
        return true
    }
    catch (err) {
        return false
    }
}

//readEvents, 讀事件檔(無檔或壞列略過)
/**
 * 讀事件檔(無檔、壞列與非一般物件之列〔null、數字、字串、陣列〕略過;fpEvents 非非空字串者同步拋 invalid fpEvents: must be a non-empty string)
 *
 * @param {String} fpEvents 輸入事件檔路徑
 * @returns {Array} 回傳事件物件陣列
 * @example
 * let events = readEvents('./state/events.jsonl')
 */
let readEvents = (fpEvents) => {
    //輸入之檢測(§5.1): fpEvents 須為非空字串(原: 非字串者於讀檔時拋錯而被吞 ⇒ 回 [] ⇒ trailingFailures 為 0 ⇒ 「連續失敗達上限不再自動起跑」之守門靜默放行)
    strReq(fpEvents, 'fpEvents')
    let out = []
    try {
        for (let l of fs.readFileSync(fpEvents, 'utf8').split('\n')) {
            if (!l.trim()) {
                continue
            }
            try {
                let e = JSON.parse(l)
                //非一般物件之列(`null`、數字、字串、陣列)視為壞列略過(原: 照收 ⇒ trailingFailures 讀 `e.event` 時 null 拋原生 TypeError)
                if (isPlain(e)) {
                    out.push(e)
                }
            }
            catch (err) {}
        }
    }
    catch (err) {}
    return out
}

//trailingFailures, 事件檔末尾連續失敗之輪數: 自末尾往前數 event === 'round-done' 且 ok === false(或 'run-failed')者, 遇成功之一輪即止
//  'skipped'、'run-start'、'run-done' 不計也不打斷(未啟動不是失敗)
/**
 * 計算事件檔末尾連續失敗之輪數
 *
 * 自末尾往前數 event 為 'round-done' 且 ok 為 false(或 event 為 'run-failed')者,遇成功之一輪即止。events 非陣列者拋 invalid events: must be an array of event objects;非一般物件之元素略過。
 *
 * @param {Array} events 輸入事件物件陣列(如 readEvents 之回傳)
 * @returns {Number} 回傳連續失敗數
 * @example
 * let n = trailingFailures([{ event: 'round-done', ok: false }, { event: 'round-done', ok: false }])
 * console.log(n)
 * // => 2
 */
let trailingFailures = (events) => {
    //輸入之檢測(§5.1): events 須為陣列(原: 非陣列者 `events.length` 為 undefined ⇒ 迴圈不進入 ⇒ 回 0 ⇒ 「連續失敗達上限即不再自動起跑」之守門靜默放行; null 則原生 TypeError); 非一般物件之元素視為壞列略過
    if (!Array.isArray(events)) {
        throw new Error('invalid events: must be an array of event objects')
    }
    let n = 0
    for (let i = events.length - 1; i >= 0; i--) {
        let e = events[i]
        if (!isPlain(e)) {
            continue
        }
        if (e.event === 'round-done') {
            if (e.ok === false) {
                n++
            }
            else {
                break
            }
        }
        else if (e.event === 'run-failed') {
            n++
        }
    }
    return n
}

//planGateOf, 起一輪之前依乾跑之計畫判斷是否起跑(規則 R-RUN-9; 不含領域用語: 只看數目)
//  一般輪次: 計畫 0 個節點、無閘門拒絕、無待重審之節點、無認領中之節點 ⇒ 不起(reason 'frontier-empty'): 起了只得一個空輪(耗一個輪號、輪末鉤子無可審計而略過做夢、記為失敗輪)
//  只做夢輪次(主控明示要求): 計畫須為 0 個節點且無待重審之節點、無認領中之節點(否則 reason 'dream-only-busy': 有節點可執行、待重審或待回收時起一般輪次即可, 做夢照常)
//  認領中之節點(planned / running): 乾跑之計畫視之為已占用, 而正式之探索於開始時先回收遺留(認領者行程已死)之認領、使其可重新計畫 ⇒ 一般輪次須照起(讓正式之探索回收), 否則遺留之認領永遠不被回收
/**
 * 起一輪之前,依乾跑之計畫判斷是否起跑
 *
 * 一般輪次於「計畫 0 個節點、無閘門拒絕、無待重審之節點、無認領中之節點」時不起(前緣為空);只做夢輪次(明示要求)只在「計畫 0 個節點、無待重審之節點、無認領中之節點」時起。
 * opt 非物件者視為 {}(依跨套件規格 K3),其各數目欄非正整數者視為 0。
 *
 * @param {Object} [opt={}] 輸入設定物件,預設{}
 * @param {Boolean} [opt.dreamOnly=false] 輸入是否為明示要求之只做夢輪次,須嚴格為 true,預設false
 * @param {Number} [opt.nItems=0] 輸入乾跑之計畫之節點數,非正整數視為 0,預設0
 * @param {Number} [opt.nRejected=0] 輸入乾跑時被閘門拒絕之計畫項數,非正整數視為 0,預設0
 * @param {Number} [opt.nReaudit=0] 輸入待重審(上一輪判重做後之新交付)之節點數,非正整數視為 0,預設0
 * @param {Number} [opt.nClaimed=0] 輸入認領中(planned / running)之節點數,非正整數視為 0,預設0
 * @returns {Object} 可起跑回傳 { go: true };不起回傳 { go: false, reason: 'frontier-empty' 或 'dream-only-busy', detail: 說明文字 }
 * @example
 * let g = planGateOf({ nItems: 0 })
 * console.log(g.go, g.reason)
 * // => false frontier-empty
 * console.log(planGateOf({ dreamOnly: true, nItems: 0 }))
 * // => { go: true }
 */
let planGateOf = (opt = {}) => {
    let o = opt && typeof opt === 'object' ? opt : {}
    let cnt = (v) => (Number.isInteger(v) && v > 0 ? v : 0)
    let nItems = cnt(o.nItems)
    let nRejected = cnt(o.nRejected)
    let nReaudit = cnt(o.nReaudit)
    let nClaimed = cnt(o.nClaimed)
    if (o.dreamOnly === true) {
        if (nItems > 0 || nReaudit > 0 || nClaimed > 0) {
            return { go: false, reason: 'dream-only-busy', detail: `不受理只做夢輪次:下一輪之計畫有 ${nItems} 個節點、待重審之節點 ${nReaudit} 個、認領中之節點 ${nClaimed} 個 —— 起一般輪次即可(做夢照常;認領中者由正式之探索先回收);要先讓做夢重新彙整者,先把前緣之提案結案` }
        }
        return { go: true }
    }
    if (nItems === 0 && nRejected === 0 && nReaudit === 0 && nClaimed === 0) {
        return { go: false, reason: 'frontier-empty', detail: '前緣為空:下一輪之計畫為 0 個節點,亦無待重審之節點、認領中之節點' }
    }
    return { go: true }
}

//roundOkOf, 一輪是否算成功(事件 round-done 之 ok; 事件檔末尾之連續失敗計數據此):
//  一般輪次: 有節點執行成功且輪末回調未失敗(原判定)
//  只做夢輪次: 做夢已交付且輪末回調未失敗(本來就沒有節點, 不以節點數判)
/**
 * 判定一輪是否算成功(事件 round-done 之 ok)
 *
 * @param {Object} [opt={}] 輸入設定物件,預設{}
 * @param {Boolean} [opt.dreamOnly=false] 輸入是否為只做夢輪次,預設false
 * @param {Number} [opt.nExecuted=0] 輸入本輪執行成功之節點數,預設0
 * @param {Boolean} [opt.hookFailed=false] 輸入輪末回調是否失敗,預設false
 * @param {Boolean} [opt.delivered=false] 輸入做夢是否已交付(只做夢輪次才看),預設false
 * @returns {Boolean} 回傳該輪是否算成功
 * @example
 * console.log(roundOkOf({ nExecuted: 3, hookFailed: false }))
 * // => true
 * console.log(roundOkOf({ dreamOnly: true, nExecuted: 0, delivered: true }))
 * // => true
 */
let roundOkOf = (opt = {}) => {
    let o = opt && typeof opt === 'object' ? opt : {}
    if (o.hookFailed === true) {
        return false
    }
    if (o.dreamOnly === true) {
        return o.delivered === true
    }
    return Number.isInteger(o.nExecuted) && o.nExecuted > 0
}
