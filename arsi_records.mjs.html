/**
 * 檢核臂目錄之提案(armProposals)
 *
 * 須為陣列,每筆 kind 須在 ARM_KINDS 內、mechanism 不得為空;new 與 layer 須附 title,revise 與 retire 須附 arm。
 *
 * @param {Array} xs 輸入提案陣列
 * @param {String} where 輸入欄位路徑(僅用於問題訊息)
 * @returns {Array} 回傳問題字串陣列,空陣列代表合格
 * @example
 * let es = checkArmProposals([{ kind: 'new', mechanism: 'm' }], 'x')
 * console.log(es)
 * // => [ 'x[0] 之 kind 為 new 時須附 title(名稱)' ]
 */
let checkArmProposals = (xs, where) => {
    if (!isArr(xs)) {
        return [`${where} 須為陣列(沒有提案給空陣列)`]
    }
    let e = []
    xs.forEach((x, i) => {
        if (!isObj(x) || !ARM_KINDS.includes(x.kind)) {
            e.push(`${where}[${i}].kind 須為 ${ARM_KINDS.join(' / ')}`)
        }
        else if (!isStr(x.mechanism) || !x.mechanism.trim()) {
            e.push(`${where}[${i}].mechanism 不得為空(機制之中性描述;revise = 修訂後之定義;retire = 停用之理由)`)
        }
        else if ((x.kind === 'new' || x.kind === 'layer') && (!isStr(x.title) || !x.title.trim())) {
            e.push(`${where}[${i}] 之 kind 為 ${x.kind} 時須附 title(名稱)`)
        }
        else if ((x.kind === 'revise' || x.kind === 'retire') && (!isStr(x.arm) || !x.arm.trim())) {
            e.push(`${where}[${i}] 之 kind 為 ${x.kind} 時須附 arm(既有臂之 id)`)
        }
        else if (['arm', 'layer', 'title', 'boundary', 'evidence'].some((k) => x[k] !== undefined && x[k] !== null && !isStr(x[k])) || (x.sources !== undefined && !strArr(x.sources))) {
            e.push(`${where}[${i}] 須為 { kind, arm?, layer?, title?, mechanism, boundary?, evidence?, sources?[] }(字串欄須為字串)`)
        }
    })
    return e
}

 * @returns {Array} 回傳問題字串陣列,空陣列代表合格;非物件時回傳 ['節點文件不是 JSON 物件']
 * @example
 * let es = checkNodeAgentPart(JSON.parse(text))
 * if (es.length > 0) {
 *     console.log(es)
 * }
 */
let checkNodeAgentPart = (a) => {
    let e = []
    if (!isObj(a)) {
        return ['節點文件不是 JSON 物件']
    }
    if (!isObj(a.plan) || !isStr(a.plan.target) || !a.plan.target.trim() || !isStr(a.plan.approach) || !a.plan.approach.trim()) {
        e.push('plan.target、plan.approach 須為非空字串')
    }
    else if (a.plan.hypotheses !== undefined && !strArr(a.plan.hypotheses)) {
        e.push('plan.hypotheses 須為字串陣列')
    }
    let art = isObj(a.code) ? (a.code.artifact === undefined ? a.code.pick : a.code.artifact) : null //code.pick 為 /1 之舊名
    if (!isObj(a.code) || !isStr(art) || !art.trim()) {
        e.push('code.artifact 須為成品之路徑字串')
    }
    else {
        if (a.code.reused !== undefined && !strArr(a.code.reused)) {
            e.push('code.reused 須為字串陣列')
        }
        if (a.code.files !== undefined && !strArr(a.code.files)) {
            e.push('code.files 須為字串陣列')
        }
        if (a.code.shareable !== undefined && !(isArr(a.code.shareable) && a.code.shareable.every((x) => isObj(x) && isStr(x.file) && isStr(x.what)))) {
            e.push('code.shareable 須為 [{ file, what }] 陣列(沒有可共用者給空陣列)')
        }
        if (a.code.inactiveArms !== undefined && !strArr(a.code.inactiveArms)) {
            e.push('code.inactiveArms 須為字串陣列(指派之臂中, 交付版未作用者之 id;沒有則空陣列)')
        }
    }
    if (!isObj(a.results) || !isObj(a.results.claimed)) {
        e.push('results.claimed 須為物件(自報之最終績效)')
    }
    else if (a.results.experiments !== undefined && !(isArr(a.results.experiments) && a.results.experiments.every((x) => isObj(x) && isStr(x.what) && isStr(x.outcome)))) {
        e.push('results.experiments 須為 [{ what, outcome }] 陣列')
    }
    //outcome(相對其比較對象之判定): 有主張之判定(claimVerdict)者可省 —— 驗證類節點之結論即主張之判定, 其任務書不要求 outcome; 給了仍須在值域內
    let hasVerdict = isObj(a.conclusion) && CLAIM_VERDICTS.includes(a.conclusion.claimVerdict)
    if (!isObj(a.conclusion) || !(OUTCOMES.includes(a.conclusion.outcome) || (hasVerdict && a.conclusion.outcome === undefined)) || !isStr(a.conclusion.summary) || !a.conclusion.summary.trim()) {
        e.push(`conclusion.outcome 須為 ${OUTCOMES.join(' / ')} 之一(有 claimVerdict 者可省), conclusion.summary 須為非空字串`)
    }
    else {
        for (let k of ['findings', 'limitations', 'nextIdeas']) {
            if (a.conclusion[k] !== undefined && !strArr(a.conclusion[k])) {
                e.push(`conclusion.${k} 須為字串陣列`)
            }
        }
        if (a.conclusion.armProposals !== undefined) {
            e.push(...checkArmProposals(a.conclusion.armProposals, 'conclusion.armProposals'))
        }
        if (a.conclusion.claimVerdict !== undefined && a.conclusion.claimVerdict !== null && !CLAIM_VERDICTS.includes(a.conclusion.claimVerdict)) {
            e.push(`conclusion.claimVerdict 須為 ${CLAIM_VERDICTS.join(' / ')}(主張依方向所寫之判定方式是否成立;無待驗之主張者省略此鍵)`)
        }
        if (a.conclusion.claimNote !== undefined && a.conclusion.claimNote !== null && !isStr(a.conclusion.claimNote)) {
            e.push('conclusion.claimNote 須為字串')
        }
    }
    return e
}

//做夢 agent 交付之做夢文件部分(audits / synthesis / guidance / docsUpdated)之檢核; scope = 本輪應審之節點 id
/**
 * 檢核做夢 agent 交付之做夢文件部分(audits、synthesis、guidance、docsUpdated 等)
 *
 * @param {Object} a 輸入做夢 agent 交付之文件物件
 * @param {Array} [scope=[]] 輸入本輪應審之節點 id 陣列,預設[]
 * @returns {Array} 回傳問題字串陣列,空陣列代表合格;非物件時回傳 ['做夢文件不是 JSON 物件']
 * @example
 * let es = checkDreamAgentPart(JSON.parse(text), nodeIds)
 * if (es.length > 0) {
 *     console.log(es)
 * }
 */
let checkDreamAgentPart = (a, scope = []) => {
    let e = []
    if (!isObj(a)) {
        return ['做夢文件不是 JSON 物件']
    }
    if (!isArr(a.audits)) {
        e.push('audits 須為陣列')
    }
    else {
        let seen = new Set()
        for (let x of a.audits) {
            if (!isObj(x) || !isStr(x.nodeId)) {
                e.push('audits 之每項須含 nodeId')
                continue
            }
            seen.add(x.nodeId)
            if (!TRUST.includes(x.trust)) {
                e.push(`audits[${x.nodeId.slice(0, 8)}].trust 須為 ${TRUST.join(' / ')}`)
            }
            if (!ACTIONS.includes(x.action)) {
                e.push(`audits[${x.nodeId.slice(0, 8)}].action 須為 ${ACTIONS.join(' / ')}`)
            }
            if (!strArr(x.evidence || []) || !strArr(x.issues || [])) {
                e.push(`audits[${x.nodeId.slice(0, 8)}].evidence / issues 須為字串陣列`)
            }
            if (x.action === 'rework' && !(isStr(x.reworkBrief) && x.reworkBrief.trim())) {
                e.push(`audits[${x.nodeId.slice(0, 8)}] 之 action 為 rework 時須附 reworkBrief`)
            }
            if (x.action === 'delete' && !(isStr(x.deleteWhy) && x.deleteWhy.trim())) {
                e.push(`audits[${x.nodeId.slice(0, 8)}] 之 action 為 delete 時須附 deleteWhy`)
            }
            if (x.inactiveArms !== undefined && !strArr(x.inactiveArms)) {
                e.push(`audits[${x.nodeId.slice(0, 8)}].inactiveArms 須為字串陣列(該節點指派之臂中, 交付版未作用者之 id;沒有則空陣列)`)
            }
        }
        for (let id of scope) {
            if (!seen.has(id)) {
                e.push(`audits 缺節點 ${id.slice(0, 8)}`)
            }
        }
    }
    if (!isObj(a.synthesis) || !isStr(a.synthesis.summary) || !a.synthesis.summary.trim()) {
        e.push('synthesis.summary 須為非空字串')
    }
    else {
        for (let k of ['worked', 'failed', 'patterns']) {
            if (a.synthesis[k] !== undefined && !strArr(a.synthesis[k])) {
                e.push(`synthesis.${k} 須為字串陣列`)
            }
        }
        let pd = a.synthesis.prevDirections
        if (pd !== undefined && !(isArr(pd) && pd.every((x) => isObj(x) && Number.isInteger(x.round) && strArr(x.arms) && x.arms.length > 0 && DIR_VERDICTS.includes(x.verdict) && isStr(x.note)))) {
            e.push(`synthesis.prevDirections 須為 [{ round: 整數, arms: [臂 id], verdict: ${DIR_VERDICTS.join(' / ')}, note }] 陣列`)
        }
        //plain(選配): 給使用者之白話摘要(字串); 輪末摘要與做夢文件之最前面
        if (a.synthesis.plain !== undefined && !isStr(a.synthesis.plain)) {
            e.push('synthesis.plain 須為字串(給使用者之白話摘要)')
        }
        //整體研判(選配): 有給即查格式(三個判定之值域、各附非空之依據)
        let ov = a.synthesis.overview
        if (ov !== undefined) {
            let okPart = (x, key, dom) => isObj(x) && dom.includes(x[key]) && isStr(x.why) && x.why.trim().length > 0
            if (!isObj(ov)) {
                e.push('synthesis.overview 須為物件')
            }
            else {
                if (!okPart(ov.coverage, 'verdict', OVERVIEW_COVERAGE)) {
                    e.push(`synthesis.overview.coverage 須為 { verdict: ${OVERVIEW_COVERAGE.join(' / ')}, why: 非空字串 }`)
                }
                if (!okPart(ov.progress, 'verdict', OVERVIEW_PROGRESS)) {
                    e.push(`synthesis.overview.progress 須為 { verdict: ${OVERVIEW_PROGRESS.join(' / ')}, why: 非空字串 }`)
                }
                if (!okPart(ov.localOptimum, 'risk', OVERVIEW_RISK)) {
                    e.push(`synthesis.overview.localOptimum 須為 { risk: ${OVERVIEW_RISK.join(' / ')}, why: 非空字串 }`)
                }
                if (ov.blindSpots !== undefined && !strArr(ov.blindSpots)) {
                    e.push('synthesis.overview.blindSpots 須為字串陣列')
                }
                //families(選配): 機制家族之歸類(機器可讀; 下一輪之搜尋地圖據以標示各節點之家族, 日後之做夢沿用或推翻)
                if (ov.families !== undefined && !(isArr(ov.families) && ov.families.every((f) => isObj(f) && isStr(f.name) && f.name.trim() && strArr(f.nodes) && isStr(f.basis)))) {
                    e.push('synthesis.overview.families 須為 [{ name: 家族名, nodes: [節點 id], basis: 歸類之依據 }] 陣列')
                }
                for (let k of ['adjust', 'prevReview']) {
                    if (ov[k] !== undefined && !isStr(ov[k])) {
                        e.push(`synthesis.overview.${k} 須為字串`)
                    }
                }
                //outwardChecked(選配): 本輪已預先核對而皆無內容之向外候選(種數與其檔) —— 無向外之方向時之依據(由領域層決定是否足以免除「須有向外」)
                if (ov.outwardChecked !== undefined && ov.outwardChecked !== null && !(isObj(ov.outwardChecked) && Number.isInteger(ov.outwardChecked.kinds) && ov.outwardChecked.kinds >= 1 && strArr(ov.outwardChecked.files) && ov.outwardChecked.files.length >= 1)) {
                    e.push('synthesis.overview.outwardChecked 須為 { kinds: 正整數(已預先核對而皆無內容之向外候選之種數), files: [檔名](至少 1 個) }')
                }
                //outwardSaturated(選配): 宣告(或續存)「現行資訊範圍內已無可試之向外方向」之依據 —— 無向外之方向時之另一依據(由領域層決定是否足以免除「須有向外」)
                if (ov.outwardSaturated !== undefined && ov.outwardSaturated !== null && !saturatedOf(ov)) {
                    e.push('synthesis.overview.outwardSaturated 須為 { basis: 非空字串(宣告「現行資訊範圍內已無可試之向外方向」之依據:已核之機制與結果,或首次宣告之輪次) }')
                }
            }
        }
    }
    if (!isObj(a.guidance) || !isArr(a.guidance.directions)) {
        e.push('guidance.directions 須為陣列')
    }
    else {
        for (let d of a.guidance.directions) {
            if (!isObj(d) || !strArr(d.arms) || d.arms.length === 0 || !(d.base === null || isStr(d.base)) || !isStr(d.rationale)) {
                e.push('guidance.directions 之每項須為 { arms: [臂 id], base: 節點 id 或 null, rationale, expect? }')
                break
            }
            if (d.type !== undefined && !TASK_TYPES.includes(d.type)) {
                e.push(`guidance.directions 之 type 須為 ${TASK_TYPES.join(' / ')}(省略 = explore)`)
                break
            }
            if (d.type === 'verify' && !(isStr(d.claim) && d.claim.trim())) {
                e.push('guidance.directions 之 type 為 verify 時須附 claim(要檢驗之主張)')
                break
            }
            //newFamily(選配; 布林): 此方向是否押不同於現有候選之機制家族(換資訊源或換機制)—— 以臂 / 層區分之「向外」擋不住「換一個臂、仍是同一家族」
            if (d.newFamily !== undefined && typeof d.newFamily !== 'boolean') {
                e.push('guidance.directions 之 newFamily 須為布林值(此方向是否押新的機制家族)')
                break
            }
        }
        for (let k of ['avoid', 'openQuestions', 'toolRequests']) {
            if (a.guidance[k] !== undefined && !strArr(a.guidance[k])) {
                e.push(`guidance.${k} 須為字串陣列`)
            }
        }
        let im = a.guidance.improvements
        if (im !== undefined) {
            if (!isArr(im)) {
                e.push('guidance.improvements 須為陣列')
            }
            else {
                im.forEach((x, i) => {
                    if (!isObj(x) || !IMPROVE_KINDS.includes(x.kind)) {
                        e.push(`guidance.improvements[${i}].kind 須為 ${IMPROVE_KINDS.join(' / ')}`)
                    }
                    else if (!isStr(x.target) || !x.target.trim() || !isStr(x.what) || !x.what.trim() || (x.evidence !== undefined && !isStr(x.evidence))) {
                        e.push(`guidance.improvements[${i}] 須為 { kind, target(對象), what(內容), evidence?(依據) }, target 與 what 不得為空`)
                    }
                })
            }
        }
        if (a.guidance.armProposals !== undefined) {
            e.push(...checkArmProposals(a.guidance.armProposals, 'guidance.armProposals'))
        }
    }
    if (a.docsUpdated !== undefined && !(isArr(a.docsUpdated) && a.docsUpdated.every((x) => isObj(x) && isStr(x.file) && isStr(x.what)))) {
        e.push('docsUpdated 須為 [{ file, what }] 陣列')
    }
    if (a.expansion !== undefined) {
        if (!isArr(a.expansion)) {
            e.push('expansion 須為陣列')
        }
        else {
            a.expansion.forEach((x, i) => {
                if (!isObj(x) || !EXPAND_DECISIONS.includes(x.decision)) {
                    e.push(`expansion[${i}].decision 須為 ${EXPAND_DECISIONS.join(' / ')}`)
                }
                else if (!isStr(x.item) || !x.item.trim() || (x.sources !== undefined && !strArr(x.sources)) || (x.files !== undefined && !strArr(x.files)) || (x.verify !== undefined && !isStr(x.verify)) || (x.note !== undefined && !isStr(x.note))) {
                    e.push(`expansion[${i}] 須為 { item(項目), sources?[], decision, files?[], verify?, note? }, item 不得為空`)
                }
                else if ((x.decision === 'built' || x.decision === 'updated') && !(isArr(x.files) && x.files.length > 0 && isStr(x.verify) && x.verify.trim())) {
                    e.push(`expansion[${i}] 之 decision 為 ${x.decision} 時須附 files(共用工具區內之檔)與 verify(怎麼驗、看到什麼)`)
                }
            })
        }
    }
    return e
}

//tallyDream, 做夢文件之改進項依類別之筆數與前次方向回顧之判定筆數(逐輪比較用: 各類之筆數下降 = 該層趨於穩定; 方向之判定 = 做夢品質之量測)
//  回 { improvements: { total, loop, dream, node, tool, score, validation, external, unclassified }, prevDirections: { total, confirmed, refuted, inconclusive, flawed, 'not-run' } }
//  unclassified = /1 文件之 toolRequests 筆數, 以及 kind 不在值域者
//  另回 expansion: { total, built, updated, declined, core }(擴建項依處置之筆數; 處置不在值域者只計入 total)
/**
 * 統計做夢文件之改進項、前次方向判定、擴建處置與臂提案之筆數(逐輪比較用)
 *
 * @param {Object} d 輸入做夢文件(可僅含部分欄位)
 * @returns {Object} 回傳物件,含 improvements、prevDirections、expansion、armProposals(各為筆數統計物件)、overview(整體研判之三個判定,無則null)、hasPlain(是否有白話總結)
 * @example
 * let t = tallyDream({ guidance: { improvements: [{ kind: 'loop' }, { kind: 'x' }] } })
 * console.log(t.improvements)
 * // => { total: 2, loop: 1, dream: 0, node: 0, tool: 0, score: 0, validation: 0, external: 0, unclassified: 1 }
 */
let tallyDream = (d) => {
    let g = (d && d.guidance) || {}
    let im = { total: 0, ...Object.fromEntries(IMPROVE_KINDS.map((k) => [k, 0])), unclassified: 0 }
    for (let x of isArr(g.improvements) ? g.improvements : []) {
        im.total++
        if (isObj(x) && IMPROVE_KINDS.includes(x.kind)) {
            im[x.kind]++
        }
        else {
            im.unclassified++
        }
    }
    let legacy = isArr(g.toolRequests) ? g.toolRequests.length : 0
    im.total += legacy
    im.unclassified += legacy
    let pd = { total: 0, ...Object.fromEntries(DIR_VERDICTS.map((k) => [k, 0])) }
    for (let x of isArr(d && d.synthesis && d.synthesis.prevDirections) ? d.synthesis.prevDirections : []) {
        if (isObj(x) && DIR_VERDICTS.includes(x.verdict)) {
            pd.total++
            pd[x.verdict]++
        }
    }
    let ex = { total: 0, ...Object.fromEntries(EXPAND_DECISIONS.map((k) => [k, 0])) }
    for (let x of isArr(d && d.expansion) ? d.expansion : []) {
        ex.total++
        if (isObj(x) && EXPAND_DECISIONS.includes(x.decision)) {
            ex[x.decision]++
        }
    }
    //整體研判之三個判定(無此欄或不在值域 ⇒ null); 逐輪比較用
    let ov = d && d.synthesis && isObj(d.synthesis.overview) ? d.synthesis.overview : null
    let pickOf = (x, key, dom) => (isObj(x) && dom.includes(x[key]) ? x[key] : null)
    let overview = ov ? { coverage: pickOf(ov.coverage, 'verdict', OVERVIEW_COVERAGE), progress: pickOf(ov.progress, 'verdict', OVERVIEW_PROGRESS), localOptimum: pickOf(ov.localOptimum, 'risk', OVERVIEW_RISK), nFamilies: isArr(ov.families) ? ov.families.length : null, outwardChecked: isObj(ov.outwardChecked) && Number.isInteger(ov.outwardChecked.kinds) && ov.outwardChecked.kinds >= 1 ? ov.outwardChecked.kinds : 0, ...(saturatedOf(ov) ? { outwardSaturated: true } : {}) } : null
    let plain = d && d.synthesis && isStr(d.synthesis.plain) ? d.synthesis.plain.trim() : ''
    //臂目錄之提案依類別之筆數(類別不在值域者只計入 total)
    let ap = { total: 0, ...Object.fromEntries(ARM_KINDS.map((k) => [k, 0])) }
    for (let x of isArr(g.armProposals) ? g.armProposals : []) {
        ap.total++
        if (isObj(x) && ARM_KINDS.includes(x.kind)) {
            ap[x.kind]++
        }
    }
    return { improvements: im, prevDirections: pd, expansion: ex, armProposals: ap, overview, hasPlain: plain.length > 0 && ![0xFF08, 0x28].includes(plain.charCodeAt(0)) }
}

let li = (xs) => (isArr(xs) && xs.length ? xs.map((x) => `- ${x}`).join('\n') : '- (無)')

let cell = (x) => {
    if (x === undefined || x === null || x === '') {
        return '—'
    }
    if (typeof x === 'number') {
        return Number.isFinite(x) ? String(Math.round(x * 10000) / 10000) : '—'
    }
    return String(x).replace(/\|/g, '/')
}

/**
 * 由節點文件正本渲染為 Markdown 文字(格式統一,不接受手改)
 *
 * @param {Object} d 輸入節點文件物件(arsi.node/3)
 * @returns {String} 回傳 Markdown 字串
 * @example
 * let md = renderNodeMd(nodeDoc)
 */
let renderNodeMd = (d) => {
    let h = d.harness || {}
    let sc = h.score || {}
    let cl = (d.results && d.results.claimed) || {}
    let au = d.audit
    let out = []
    out.push(`# 節點文件 ${d.nodeId}`)
    out.push('')
    out.push(`- 規格: ${NODE_SCHEMA};輪次 r${String(d.round).padStart(3, '0')};第 ${d.attempt} 次交付;狀態 ${d.status}`)
    out.push(`- 臂: ${((d.spec && d.spec.arms) || []).join(' + ')};起點(base): ${d.spec && d.spec.base ? d.spec.base : '無'}`)
    out.push(`- agent: ${d.agent ? `${d.agent.kind} ${d.agent.model}${d.agent.variant ? ` (${d.agent.variant})` : ''};${d.agent.startedAt} ~ ${d.agent.endedAt}(${d.agent.durationSec}s);${d.agent.ok ? '正常結束' : `異常: ${d.agent.error || ''}`}` : '—'}`)
    out.push('')
    out.push('## 1. 任務(迴圈指派)')
    out.push('')
    out.push(d.task && d.task.goal ? d.task.goal : '—')
    if (d.task && d.task.type) {
        out.push('')
        out.push(`任務類型: **${TASK_LABELS[d.task.type] || d.task.type}**(${d.task.type})${d.task.claim ? `;待驗之主張: ${d.task.claim}` : ''}`)
    }
    //任務書條款之試驗(選配; 由專案之任務指派帶入: task.trial = { id, variant }): 同一輪之節點任務書有兩種版本時, 記本節點拿到哪一種
    if (d.task && d.task.trial && d.task.trial.id) {
        out.push('')
        out.push(`任務書條款之試驗: ${d.task.trial.id} / ${d.task.trial.variant} 組(A = 現行條款、B = 新條款;組別不代表好壞)`)
    }
    if (d.task && isArr(d.task.refs) && d.task.refs.length) {
        out.push('')
        //參考之標識: 節點 id / 做夢輪次 / 檔案路徑(三種 kind 之欄位不同; 皆無則用 path)

    //  並列「判定 worse」與「主張之判定 成立」只生混淆, 文字條款擋不住; 探索類照列
    if (!(d.task && d.task.type === 'verify' && d.conclusion && CLAIM_VERDICTS.includes(d.conclusion.claimVerdict))) {

        out.push('- 證據:')

            out.push(`- 重做要求: ${au.reworkBrief}`)

            out.push(`- 刪除理由: ${au.deleteWhy}`)
        }
        if (isArr(au.inactiveArms)) {
            out.push(`- 未作用之臂(審計所定): ${au.inactiveArms.length ? au.inactiveArms.join('、') : '無'}`)
        }
    }
    else {
        out.push('(尚未審計)')
    }
    //主控補記之未作用之臂(審計之後才發生者: 臂之定義修訂使本節點不再屬該臂, 或此欄設立前已審之節點; 規則 R-REC-20): 以此列為準
    if (isObj(d.control) && isObj(d.control.inactiveArms) && isArr(d.control.inactiveArms.arms)) {

        out.push(`- 未作用之臂(主控補記${typeof c.at === 'string' ? `;${c.at.slice(0, 10)}` : ''};以此為準): ${c.arms.length ? c.arms.join('、') : '無'}${isArr(c.offDef) && c.offDef.length ? `(其中依修訂後之定義不屬該臂、不計入已試: ${c.offDef.join('、')})` : ''}${typeof c.why === 'string' && c.why ? ` —— ${c.why}` : ''}`)
    }
    out.push('')
    return out.join('\n')

//預設之檔案儲存; opt: { nodeDir, dreamDir, trashDir }
/**
 * 建立預設之檔案儲存(每份文件一個 JSON 正本加一個由正本渲染之 Markdown)
 *
 * 未給 nodeDir 或 dreamDir 會拋錯;目錄於寫入時自動建立。
 *
 * @param {Object} opt 輸入設定物件
 * @param {String} opt.nodeDir 輸入節點文件資料夾
 * @param {String} opt.dreamDir 輸入做夢文件資料夾
 * @param {String} [opt.trashDir] 輸入回收夾,removeNodeDoc 需要此設定,未給則呼叫時拋錯
 * @returns {Object} 回傳物件,含 paths(nodeJson、nodeMd、dreamJson、dreamMd 路徑函數)與方法 putNodeDoc(寫入節點文件並渲染)、getNodeDoc(讀取節點文件)、listNodeDocs(列出節點文件,可帶 { round } 篩選)、removeNodeDoc(移入回收夾並寫原因)、putDreamDoc(寫入做夢文件並渲染)、getDreamDoc(依輪次讀取)、listDreamDocs(依輪次升冪列出)
 * @example
 * let records = createFileRecords({ nodeDir: './nodes', dreamDir: './dreams', trashDir: './trash' })
 * records.putDreamDoc({ round: 2, audits: [] })
 * console.log(records.getDreamDoc(2).round)
 * // => 2
 */
let createFileRecords = ({ nodeDir, dreamDir, trashDir }) => {
    if (!nodeDir || !dreamDir) {
        throw new Error('createFileRecords: 須給 nodeDir 與 dreamDir')
    }
    let ensure = () => {
        for (let d of [nodeDir, dreamDir, trashDir].filter(Boolean)) {
            fs.mkdirSync(d, { recursive: true })
        }
    }
    let paths = {
        nodeJson: (id) => path.join(nodeDir, `${id}.json`),
        nodeMd: (id) => path.join(nodeDir, `${id}.md`),
        dreamJson: (r) => path.join(dreamDir, `${roundName(r)}.json`),
        dreamMd: (r) => path.join(dreamDir, `${roundName(r)}.md`),
    }
    let readJson = (fp) => {
        try {
            return JSON.parse(fs.readFileSync(fp, 'utf8'))
        }
        catch (err) {
            return null
        }
    }
    let putNodeDoc = (doc) => {
        ensure()
        let d = { ...doc, schema: NODE_SCHEMA, updatedAt: new Date().toISOString() }
        fs.writeFileSync(paths.nodeJson(d.nodeId), JSON.stringify(d, null, 1), 'utf8')
        fs.writeFileSync(paths.nodeMd(d.nodeId), renderNodeMd(d), 'utf8')
        return d
    }
    let getNodeDoc = (id) => readJson(paths.nodeJson(id))
    let listNodeDocs = ({ round = null } = {}) => {
        ensure()
        return fs.readdirSync(nodeDir).filter((f) => f.endsWith('.json')).map((f) => readJson(path.join(nodeDir, f))).filter((d) => d && (round === null || d.round === round))
    }
    let removeNodeDoc = (id, why) => {
        ensure()
        if (!trashDir) {
            throw new Error('removeNodeDoc: 未設定 trashDir')
        }
        let moved = []
        for (let fp of [paths.nodeJson(id), paths.nodeMd(id)]) {
            if (fs.existsSync(fp)) {
                let to = path.join(trashDir, `${new Date().toISOString().replace(/[:.]/g, '-')}_${path.basename(fp)}`)
                fs.renameSync(fp, to)
                moved.push(to)
            }
        }
        if (moved.length) {
            fs.writeFileSync(path.join(trashDir, `${new Date().toISOString().replace(/[:.]/g, '-')}_${id}.why.txt`), String(why || ''), 'utf8')
        }
        return moved
    }
    let putDreamDoc = (doc) => {
        ensure()
        let d = { ...doc, schema: DREAM_SCHEMA, updatedAt: new Date().toISOString() }
        fs.writeFileSync(paths.dreamJson(d.round), JSON.stringify(d, null, 1), 'utf8')
        fs.writeFileSync(paths.dreamMd(d.round), renderDreamMd(d), 'utf8')
        return d
    }
    let getDreamDoc = (r) => readJson(paths.dreamJson(r))
    let listDreamDocs = () => {
        ensure()
        return fs.readdirSync(dreamDir).filter((f) => /^r\d+\.json$/.test(f)).map((f) => readJson(path.join(dreamDir, f))).filter(Boolean).sort((a, b) => a.round - b.round)
    }
    return { paths, putNodeDoc, getNodeDoc, listNodeDocs, removeNodeDoc, putDreamDoc, getDreamDoc, listDreamDocs }
}
