//guideText, 讀守則檔之條目(略去標題列與以「>」開頭之說明列), 依列截到 cap 位元組; 無檔或無條目回空字串
let guideText = (FD_ROOT, g) => {
    let text = ''
    try {
        text = fs.readFileSync(path.join(FD_ROOT, g.rel), 'utf8')
    }
    catch (err) {
        return ''
    }
    let out = []
    let n = 0
    for (let l of text.split('\n')) {
        if (/^\s*(#|>)/.test(l) || (out.length === 0 && l.trim() === '')) {
            continue
        }
        let b = Buffer.byteLength(l, 'utf8') + 1
        if (n + b > g.cap) {
            break
        }
        out.push(l)
        n += b
    }
    return out.join('\n').trim()
}
