import fs from 'fs'
import path from 'path'
import iseint from 'wsemi/src/iseint.mjs'
import { P, bytesAs, assertIntrinsics } from './intrinsics.mjs'

const { ObjectFreeze, NumberCtor, NumberIsInteger, MathImul, ErrorCtor, TypedArrayPrototypeGetLength, TypedArrayPrototypeGetToStringTag, Int8ArrayCtor, Uint8ArrayCtor, Uint16ArrayCtor, Int32ArrayCtor, Uint32ArrayCtor, Float32ArrayCtor, Float64ArrayCtor } = P


//訂單帳本之載入: 讀 ledgerBuild 產出之二進位欄, 並建「依出場棒排序」之輔助欄(供「某時點以前已結之單」之統計以二分搜尋取得)
//  回傳 L: { meta, n(策略數), nBars, off/ent/ext/pnl(依進場棒升冪之逐單欄), xoff/xExt/xPnlCum/xWinCum(依出場棒升冪之已結單欄; Cum 為策略內前綴和), 靜態欄, fac(因子序列欄; 用到才讀), seriesNames / ser(外部序列之名稱與欄; 欄用到才讀) }
//readBin, 讀檔為定型陣列(規則 R-EV-35 ②): fac / ser 為延遲讀取, 首次讀取發生於挑選函數呼叫 ctx.factor / ctx.series 之期間
//  ⇒ 只用載入時擷取之參照(bytesAs: 擷取之建構子與 set; 不經 ArrayBuffer.prototype.slice 與其 species 建構子); 延遲讀取之兩欄傳入擷取之建構子(不以全域名稱取用)
//  且延遲讀取之前先比對內建物件(lazyCheck): Node 之 fs.readFileSync 本身會經可被改寫之 %TypedArray%.prototype.byteLength 與 Buffer.allocUnsafe 取得讀入用之緩衝區,
//  被改時拒讀(拋 EVAL_INTEGRITY; 未讀入, 下次取用再試)—— 讀檔為同步, 比對通過後至讀完之間不會執行模組之程式
//  其餘各欄於 loadLedger 呼叫當下讀入(命令列於讀帳本前先比對內建物件; 見 src/cli/eval.mjs)
let readBin = (fp, Ctor) => bytesAs(fs.readFileSync(fp), Ctor)

/**
 * 載入訂單帳本資料夾(由 buildLedger 產出之 meta.json 與二進位欄), 並建立依出場棒排序之已結單欄
 * K 線欄 px.f64 有此檔者須恰為 5 × K 棒數個 float64(否則拋 invalid px.f64: must be … 之錯誤, 請重建帳本); 無此檔(舊帳本)者 px 為 null
 * 因子序列欄 fac 為延遲讀取(首次存取 L.fac 才載入; 無檔者為 null); 外部序列欄 ser 同(首次存取 L.ser 才載入; 無檔者為 null), 其名稱 seriesNames 取自 meta.seriesNames(舊帳本無此欄者為空陣列)
 *
 * @param {String} dir 輸入帳本資料夾路徑
 * @returns {Object} 回傳帳本物件L, 含 dir、meta、n(策略數)、nBars、barsPerDay、逐單欄(off、ent、ext、pnl)、靜態欄(mode、tp、sl、nKeys、lv、fitNt、cls、keyOff、keyIdx、keyNames、px、keySym、keyTh、keyOr; px、keySym、keyTh、keyOr 於無檔時為 null)、依出場棒升冪之已結單欄(xoff、xExt、xPnlCum、xWinCum)、延遲讀取之 fac 與外部序列(seriesNames、延遲讀取之 ser)
 * @example
 * let L = loadLedger('./data/ledger/eval')
 */
let loadLedger = (dir) => {
    //入口比對(規則 R-EV-35 ②): 讀帳本之一般寫法(JSON.parse、push、sort、length 等)須在比對之後; 呼叫端之前可能已載入模組
    assertIntrinsics('ledger-load')
    let meta = JSON.parse(fs.readFileSync(path.join(dir, 'meta.json'), 'utf8'))
    //px, K 線欄(開高低收量逐棒五個值; px[i * 5 + 0..4]; 規則 R-EV-36): 有此檔者須恰為 5 × K 棒數個 float64(位元組數 = 5 × 8 × nBars), 否則拋錯請重建帳本
    //  截短之欄使 ctx.market / api.market 於缺尾處回 NaN 而不拋錯; 無此檔(舊帳本)者為 null(取用時拋「帳本無 K 線欄」); 因子欄與外部序列欄為延遲讀取, 其長度於取用時核對(evalCore.mjs 之 factorAt / seriesAt)
    //  核對之位元組數即實際使用之位元組(讀入一次、核對該份、再轉為 Float64Array; 不先 stat 再讀: 兩次之間檔案被重建者核對與使用不同份; 非 8 之倍數者不致由定型陣列建構子拋出不指明欄名之錯誤)
    //  meta.nBars 須為非負整數(小數、數字字串使乘積可通過核對而於其後建評估器時拋不指明欄名之錯誤); 錯誤訊息不帶值與路徑(跨套件規格〈輸入參數之檢測機制〉§6 V19、V20)
    let fpPx = path.join(dir, 'px.f64')
    let px = null
    if (fs.existsSync(fpPx)) {
        let bufPx = fs.readFileSync(fpPx)
        if (!Number.isInteger(meta.nBars) || meta.nBars < 0 || TypedArrayPrototypeGetLength(bufPx) !== 40 * meta.nBars) {
            throw new Error('invalid px.f64: must be 5 * meta.nBars float64 values (請重建帳本)')
        }
        px = bytesAs(bufPx, Float64Array)
    }
    let L = {
        dir,
        meta,
        n: meta.nStrategies,
        nBars: meta.nBars,
        barsPerDay: Math.round(24 / meta.barHours),
        off: readBin(path.join(dir, 'off.u32'), Uint32Array),
        ent: readBin(path.join(dir, 'ent.i32'), Int32Array),
        ext: readBin(path.join(dir, 'ext.i32'), Int32Array),
        pnl: readBin(path.join(dir, 'pnl.f32'), Float32Array),
        mode: readBin(path.join(dir, 'mode.i8'), Int8Array),
        tp: readBin(path.join(dir, 'tp.f32'), Float32Array),
        sl: readBin(path.join(dir, 'sl.f32'), Float32Array),
        nKeys: readBin(path.join(dir, 'nKeys.u8'), Uint8Array),
        lv: readBin(path.join(dir, 'lv.i32'), Int32Array),
        fitNt: readBin(path.join(dir, 'fitNt.i32'), Int32Array),
        cls: readBin(path.join(dir, 'cls.u8'), Uint8Array),
        keyOff: readBin(path.join(dir, 'keyOff.u32'), Uint32Array),
        keyIdx: readBin(path.join(dir, 'keyIdx.u16'), Uint16Array),
        keyNames: meta.keyNames,
        px,
        //進場規則之各條件(與 keyIdx 逐格對齊; 規則 R-EV-29); 舊帳本無此三欄者為 null
        keySym: fs.existsSync(path.join(dir, 'keySym.i8')) ? readBin(path.join(dir, 'keySym.i8'), Int8Array) : null,
        keyTh: fs.existsSync(path.join(dir, 'keyTh.f64')) ? readBin(path.join(dir, 'keyTh.f64'), Float64Array) : null,
        keyOr: fs.existsSync(path.join(dir, 'keyOr.u8')) ? readBin(path.join(dir, 'keyOr.u8'), Uint8Array) : null,
    }
    //fac, 因子序列欄(fac[ki * nBars + i]; 約 60 MB): 用到才讀 —— 未呼叫 ctx.factor 之評估不付載入之時間與記憶體; 舊帳本無此欄者為 null
    let fpFac = path.join(dir, 'fac.f64')
    let fac
    Object.defineProperty(L, 'fac', {
        enumerable: false,
        configurable: true,
        get: () => {
            if (fac === undefined) {
                assertIntrinsics('ledger-read')
                fac = fs.existsSync(fpFac) ? readBin(fpFac, Float64ArrayCtor) : null
            }
            return fac
        },
    })
    //seriesNames / ser, 外部序列(規則 R-EV-34; 選配; ser[si * nBars + i], 缺值 NaN): 名稱取自 meta.seriesNames(舊帳本無此欄者為空陣列), 欄同 fac 用到才讀; 無檔者為 null
    //  存在而型態不合(非字串陣列)者拋錯, 不當成未建入(免得有序列欄而名稱被靜默忽略)
    if (meta.seriesNames !== undefined && meta.seriesNames !== null && !(Array.isArray(meta.seriesNames) && meta.seriesNames.every((x) => typeof x === 'string'))) {
        throw new Error(`帳本 meta.seriesNames 之格式不合(須為名稱字串陣列; 請重建帳本): ${dir}`)
    }
    L.seriesNames = Array.isArray(meta.seriesNames) ? meta.seriesNames : []
    let fpSer = path.join(dir, 'ser.f64')
    let ser
    Object.defineProperty(L, 'ser', {
        enumerable: false,
        configurable: true,
        get: () => {
            if (ser === undefined) {
                assertIntrinsics('ledger-read')
                if (fs.existsSync(fpSer)) {
                    let sz = fs.statSync(fpSer).size
                    if (sz % 8 !== 0) {
                        throw new Error(`帳本之外部序列欄 ser.f64 之位元組數(${sz})非 8 之倍數(請重建帳本): ${fpSer}`)
                    }
                    ser = readBin(fpSer, Float64ArrayCtor)
                }
                else {
                    ser = null
                }
            }
            return ser
        },
    })
    setExitCols(L)
    return L
}

//setExitCols, 由逐單欄(off / ext / pnl)建「依出場棒排序」之已結單欄(xoff / xExt / xPnlCum / xWinCum; Cum 為策略內前綴和)並掛於 L
//  loadLedger 與 memberLedger(半庫帳本; 規則 R-EV-40)共用同一份寫法; 兩者皆於入口比對內建物件之後同步執行(其間不執行模組之程式)
let setExitCols = (L) => {
    let n = L.n
    let xoff = new Uint32Array(n + 1)
    let nClosed = 0
    for (let j = 0; j < L.ext.length; j++) {
        if (L.ext[j] >= 0) {
            nClosed++
        }
    }
    let xExt = new Int32Array(nClosed)
    let xPnlCum = new Float64Array(nClosed)
    let xWinCum = new Int32Array(nClosed)
    let p = 0
    let tmp = []
    for (let s = 0; s < n; s++) {
        xoff[s] = p
        tmp.length = 0
        for (let j = L.off[s]; j < L.off[s + 1]; j++) {
            if (L.ext[j] >= 0) {
                tmp.push(j)
            }
        }
        tmp.sort((a, b) => L.ext[a] - L.ext[b] || a - b)
        let cp = 0
        let cw = 0
        for (let j of tmp) {
            cp += L.pnl[j]
            cw += L.pnl[j] > 0 ? 1 : 0
            xExt[p] = L.ext[j]
            xPnlCum[p] = cp
            xWinCum[p] = cw
            p++
        }
    }
    xoff[n] = p
    L.xoff = xoff
    L.xExt = xExt
    L.xPnlCum = xPnlCum
    L.xWinCum = xWinCum
}

/**
 * 各策略固定之亂數(同分次鍵之基底): 由「策略於全庫之序號」經雜湊而得之無號 32 位元整數, 與策略之任何屬性無關(規則 R-EV-40)
 * 評估器之 ctx.meta.tie(同分次序為正本時)即此值; 半庫帳本(memberLedger)之策略沿用其於全庫之序號(L.srcIdx), 故其值不因重新編號而變
 * 單一來源: 評估器(makeEvaluator)與半庫之切法(memberLedger)皆經此函數, 其值一變, 既有成品之同分次序與選集皆變
 *
 * @param {Integer} idx 輸入策略於全庫之序號(0 以上之整數)
 * @returns {Integer} 回傳無號 32 位元整數
 * @example
 * let t = tieBaseOf(0)
 */
let tieBaseOf = (idx) => {
    let x = MathImul(idx + 0x9E3779B9, 0x85EBCA6B)
    x = MathImul(x ^ (x >>> 13), 0xC2B2AE35)
    return (x ^ (x >>> 16)) >>> 0
}

//memberOf, 策略是否屬於某一半(成員重抽; 規則 R-EV-40): 以其同分次鍵之基底與鹽之雜湊切半 ——
//  ((Math.imul(tie ^ salt, 2654435761) >>> 0) / 4294967296) < 0.5 者為甲半(half 0), 其餘為乙半(half 1); 換鹽 = 換一種切法
let memberOf = (tieBase, salt, half) => (((MathImul(tieBase ^ salt, 2654435761) >>> 0) / 4294967296) < 0.5) === (half === 0)

/**
 * 半庫帳本之 srcIdx(各策略於全庫之序號)是否合格: 型別以內部槽判定為 Int32Array(不看原型鏈; 同 L.ntStatic)、長度 n、全為 0 以上
 * 評估器(makeEvaluator)與半庫帳本(memberLedger)共用
 *
 * @param {*} srcIdx 輸入待檢查之值
 * @param {Integer} n 輸入策略數(L.n)
 * @returns {Boolean} 回傳是否合格
 * @example
 * let ok = isSrcIdx(new Int32Array(3), 3)
 */
let isSrcIdx = (srcIdx, n) => {
    if (TypedArrayPrototypeGetToStringTag(srcIdx) !== 'Int32Array' || TypedArrayPrototypeGetLength(srcIdx) !== n) {
        return false
    }
    for (let s = 0; s < n; s++) {
        if (!(srcIdx[s] >= 0)) {
            return false
        }
    }
    return true
}

//isLedgerShape, memberLedger 之入口檢查: 帳本物件之逐策略欄、逐單欄、進場規則之各條件之型別(內部槽)與長度須相符(讀取之區段不得超出)
//  型別對照 loadLedger 之讀入; keySym / keyTh / keyOr 於舊帳本為 null(皆可); 起訖欄(off、keyOff)自 0 起且不遞減; 逐單與條件欄之長度不得短於其起訖欄之終點
//  帳本為評估器自己之資料(loadLedger 之回傳; 挑選模組與擴充診斷碰不到): 本檢查為防呆(手組或損毀之帳本), 不處理「呼叫端以存取子在檢查後改內建物件」之情形(同 loadLedger / makeEvaluator 對呼叫端之信任)
let isLedgerShape = (L) => {
    if (L === null || typeof L !== 'object' || !NumberIsInteger(L.n) || L.n < 0) {
        return false
    }
    let n = L.n
    let colOk = (a, tag, len) => TypedArrayPrototypeGetToStringTag(a) === tag && TypedArrayPrototypeGetLength(a) === len
    let atLeast = (a, tag, len) => TypedArrayPrototypeGetToStringTag(a) === tag && TypedArrayPrototypeGetLength(a) >= len
    if (!colOk(L.off, 'Uint32Array', n + 1) || !colOk(L.keyOff, 'Uint32Array', n + 1)) {
        return false
    }
    //起訖欄須自 0 起且不遞減(各段之端點才不越界; 損毀之帳本不得靜默產生假資料)
    let mono = (a) => {
        if (a[0] !== 0) {
            return false
        }
        for (let s = 0; s < n; s++) {
            if (!(a[s] <= a[s + 1])) {
                return false
            }
        }
        return true
    }
    if (!mono(L.off) || !mono(L.keyOff)) {
        return false
    }
    let nOrd = L.off[n]
    let nKey = L.keyOff[n]
    return colOk(L.mode, 'Int8Array', n) && colOk(L.tp, 'Float32Array', n) && colOk(L.sl, 'Float32Array', n) && colOk(L.nKeys, 'Uint8Array', n) && colOk(L.lv, 'Int32Array', n) && colOk(L.fitNt, 'Int32Array', n) && colOk(L.cls, 'Uint8Array', 3 * n) &&
        atLeast(L.ent, 'Int32Array', nOrd) && atLeast(L.ext, 'Int32Array', nOrd) && atLeast(L.pnl, 'Float32Array', nOrd) &&
        atLeast(L.keyIdx, 'Uint16Array', nKey) &&
        (L.keySym === null || atLeast(L.keySym, 'Int8Array', nKey)) && (L.keyTh === null || atLeast(L.keyTh, 'Float64Array', nKey)) && (L.keyOr === null || atLeast(L.keyOr, 'Uint8Array', nKey)) &&
        (L.srcIdx === undefined || L.srcIdx === null || isSrcIdx(L.srcIdx, n)) &&
        (L.member === undefined || L.member === null || (typeof L.member === 'object' && NumberIsInteger(L.member.nFull) && L.member.nFull >= n))
}

/**
 * 成員重抽之半庫帳本: 把策略庫依各策略之同分次鍵之基底(tieBaseOf)與鹽之雜湊切成兩半, 回傳只含其中一半、且重新編號之帳本物件(不改動傳入之帳本; 規則 R-EV-40)
 * 半庫帳本之各逐策略欄(逐單欄、靜態欄、進場規則之各條件、依出場棒排序之已結單欄)皆只含該半之策略、序號為半庫內之 0 ~ n-1(保持原先後), 內容與其於全庫時逐位相同(皆為新建之複本);
 * 逐棒之欄(K 線、因子序列、外部序列)與原帳本共用; 另帶 srcIdx(Int32Array; 各策略於全庫之序號)—— 評估器以之產生同分次鍵, 使半庫內各策略之 ctx.meta.tie 與其於全庫時相同
 * 用途: 以同一個挑選模組(不必改寫)在「換一批策略」之半庫上重跑, 檢驗其內容是否依賴特定成員; 訓練期之進場數須於其後另給(L.ntStatic = ntStaticOf(半庫帳本, 訓練期之終點棒序))
 * dir 與 meta 沿用來源帳本(全庫之資訊: meta.nStrategies、nOrders 等為全庫之值; loadIds(dir) 之序號為全庫序號 —— 半庫內第 i 隻之策略檔名為 loadIds(dir)[srcIdx[i]])
 * 入口先比對受保護之內建物件; 複製只用定型陣列之索引存取(不經可被改寫之方法); 參數不合者拋錯(訊息不帶輸入值)
 * salt 與 half 依 wsemi iseint 判定(安全整數之數字或其數字字串, 通過者轉為數字); 十六進位之寫法(0x…)只於命令列解析, 此處只收整數
 *
 * @param {Object} L 輸入由 loadLedger 取得之帳本物件(亦可為另一個半庫帳本: 於其內再切, srcIdx 與 member.nFull 仍指全庫)
 * @param {Integer} salt 輸入切半之鹽(0 ~ 4294967295 之整數; 換鹽 = 換一種切法)
 * @param {Integer} half 輸入取哪一半(0 = 甲半、1 = 乙半)
 * @returns {Object} 回傳半庫帳本物件, 欄位同 loadLedger 之回傳(不含 ntStatic), 另含 srcIdx 與 member({ salt, half, nFull }; 凍結)
 * @example
 * let st = getSettings()
 * let L = loadLedger(st.eval.ledgerDir)
 * //甲半(half = 0); 換鹽 = 換一種切法
 * let La = memberLedger(L, 0x5bd1e995, 0)
 * //訓練期之界線須明給(同 ntStaticOf: 依帳本之 time0 與每棒小時數自設定之訓練期終點換算)
 * let trainEndBar = Math.round((Date.parse(`${st.periods.train.end}Z`) - Date.parse(`${La.meta.time0}Z`)) / (La.meta.barHours * 3600000))
 * La.ntStatic = ntStaticOf(La, trainEndBar)
 * let ev = makeEvaluator(La, { warmupDays: 180, cadenceDays: 7 })
 * console.log(La.n, La.member.half) //半庫之策略數(依鹽而異, 約為全庫之一半)與所取之半
 */
let memberLedger = (L, salt, half) => {
    assertIntrinsics('ledger-load')
    if (!isLedgerShape(L)) {
        throw new ErrorCtor('invalid L: must be a ledger object (see loadLedger and memberLedger)')
    }
    let saltV = iseint(salt) ? NumberCtor(salt) : NaN
    if (!(saltV >= 0 && saltV <= 0xFFFFFFFF)) {
        throw new ErrorCtor('invalid salt: must be an integer from 0 to 4294967295')
    }
    let halfV = iseint(half) ? NumberCtor(half) : NaN
    if (halfV !== 0 && halfV !== 1) {
        throw new ErrorCtor('invalid half: must be 0 or 1')
    }
    saltV = saltV === 0 ? 0 : saltV //-0 正規化
    halfV = halfV === 0 ? 0 : 1
    let n0 = L.n
    let src0 = L.srcIdx === undefined || L.srcIdx === null ? null : L.srcIdx
    //成員: 依各策略於全庫之序號(已是半庫者取其 srcIdx)之同分次鍵之基底切半
    let sel = new Uint8ArrayCtor(n0)
    let n = 0
    for (let s = 0; s < n0; s++) {
        if (memberOf(tieBaseOf(src0 === null ? s : src0[s]), saltV, halfV)) {
            sel[s] = 1
            n++
        }
    }
    let keep = new Int32ArrayCtor(n)
    for (let s = 0, i = 0; s < n0; s++) {
        if (sel[s] === 1) {
            keep[i++] = s
        }
    }
    //gather, 逐策略 k 值之欄(第 s 隻策略之值在 [s * k, s * k + k); 只以索引讀寫定型陣列): 一般之靜態欄 k = 1; cls(各因子類別之條件數)k = 3(ledgerBuild.mjs 之 cls[si * 3 + 類別])
    let gather = (a, Ctor, k = 1) => {
        let out = new Ctor(n * k)
        for (let i = 0; i < n; i++) {
            for (let j = 0; j < k; j++) {
                out[i * k + j] = a[keep[i] * k + j]
            }
        }
        return out
    }
    //gatherRanges, 以起訖欄(長度 n0 + 1)分段之欄: 回新之起訖欄與各欄(cols 之元素為 [欄, 建構子]; 欄為 null 者照舊為 null); 段內之先後不變
    let gatherRanges = (offs, cols) => {
        let off = new Uint32ArrayCtor(n + 1)
        let tot = 0
        for (let i = 0; i < n; i++) {
            off[i] = tot
            tot += offs[keep[i] + 1] - offs[keep[i]]
        }
        off[n] = tot
        let outs = []
        for (let c = 0; c < cols.length; c++) {
            outs[c] = cols[c][0] === null ? null : new cols[c][1](tot)
        }
        for (let i = 0; i < n; i++) {
            let a0 = offs[keep[i]]
            let len = offs[keep[i] + 1] - a0
            let p = off[i]
            for (let c = 0; c < cols.length; c++) {
                let src = cols[c][0]
                if (src !== null) {
                    let dst = outs[c]
                    for (let j = 0; j < len; j++) {
                        dst[p + j] = src[a0 + j]
                    }
                }
            }
        }
        return { off, outs }
    }
    let ord = gatherRanges(L.off, [[L.ent, Int32ArrayCtor], [L.ext, Int32ArrayCtor], [L.pnl, Float32ArrayCtor]])
    let key = gatherRanges(L.keyOff, [[L.keyIdx, Uint16ArrayCtor], [L.keySym, Int8ArrayCtor], [L.keyTh, Float64ArrayCtor], [L.keyOr, Uint8ArrayCtor]])
    let srcIdx = new Int32ArrayCtor(n)
    for (let i = 0; i < n; i++) {
        srcIdx[i] = src0 === null ? keep[i] : src0[keep[i]]
    }
    let M = {
        dir: L.dir,
        meta: L.meta,
        n,
        nBars: L.nBars,
        barsPerDay: L.barsPerDay,
        off: ord.off,
        ent: ord.outs[0],
        ext: ord.outs[1],
        pnl: ord.outs[2],
        mode: gather(L.mode, Int8ArrayCtor),
        tp: gather(L.tp, Float32ArrayCtor),
        sl: gather(L.sl, Float32ArrayCtor),
        nKeys: gather(L.nKeys, Uint8ArrayCtor),
        lv: gather(L.lv, Int32ArrayCtor),
        fitNt: gather(L.fitNt, Int32ArrayCtor),
        cls: gather(L.cls, Uint8ArrayCtor, 3),
        keyOff: key.off,
        keyIdx: key.outs[0],
        keyNames: L.keyNames,
        px: L.px,
        keySym: key.outs[1],
        keyTh: key.outs[2],
        keyOr: key.outs[3],
        seriesNames: L.seriesNames,
        srcIdx,
        member: ObjectFreeze({ salt: saltV, half: halfV, nFull: L.member ? L.member.nFull : n0 }),
    }
    //逐棒之欄與原帳本共用(延遲讀取者仍於首次取用時才由原帳本載入, 其載入前之內建物件比對亦在原帳本內)
    Object.defineProperty(M, 'fac', { enumerable: false, configurable: true, get: () => L.fac })
    Object.defineProperty(M, 'ser', { enumerable: false, configurable: true, get: () => L.ser })
    setExitCols(M)
    return M
}

/**
 * 讀取帳本資料夾內之 ids.json, 取得策略序號對應之策略檔名陣列
 *
 * @param {String} dir 輸入帳本資料夾路徑
 * @returns {Array} 回傳策略檔名陣列, 以策略序號為索引
 * @example
 * let ids = loadIds('./data/ledger/eval')
 */
let loadIds = (dir) => {
    assertIntrinsics('ledger-load')
    return JSON.parse(fs.readFileSync(path.join(dir, 'ids.json'), 'utf8'))
}
