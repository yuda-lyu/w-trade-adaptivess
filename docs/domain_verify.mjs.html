//callsOf, 自一列操作紀錄(JSON 事件)取出工具呼叫 [{ tool(小寫之工具名), input }](可能多個); 兩種格式:
//  ① opencode --format json: part.tool 與 part.state.input  ② claude --output-format stream-json: message.content[*](type 為 tool_use)之 name 與 input
let callsOf = (j) => {
    let out = []
    let a = j && j.part && j.part.state && j.part.state.input
    if (a && typeof a === 'object') {
        //status / error: 該次工具呼叫之結果(opencode 之紀錄才有; 供分辨「被權限拒絕而未執行」者)
        out.push({ tool: String((j.part && j.part.tool) || '').toLowerCase(), input: a, status: typeof j.part.state.status === 'string' ? j.part.state.status : null, error: typeof j.part.state.error === 'string' ? j.part.state.error : null })
    }
    let cs = j && j.message && Array.isArray(j.message.content) ? j.message.content : []
    for (let c of cs) {
        if (c && c.type === 'tool_use' && c.input && typeof c.input === 'object') {
            out.push({ tool: String(c.name || '').toLowerCase(), input: c.input })
        }
    }
    return out
}

//deniedHalt, 操作紀錄是否結束於「被權限拒絕」之工具呼叫(規則 R-RUN-7): opencode 於 ask 被自動拒絕後可能結束整個工作階段(離開碼 0),
//  節點因此未交付成品; 派工設定已帶 continue_loop_on_deny, 本函式供失敗訊息標明原因(該設定失效時之警示)
//  判法: 最後一步(最後一個 step_start 之後)內有被拒之工具呼叫 —— 同一步並行發出數個呼叫時, 被拒者不一定是最後一筆;
//    紀錄無 step_start 者(其他格式)只看最後一筆工具呼叫。被拒後續做而以文字收尾者, 最後一步無工具呼叫 ⇒ null
//  回 null | { tool, target }(target = 該呼叫之路徑或指令, 前 120 字元)
let deniedHalt = (logFile) => {
    let cur = []
    let sawStep = false
    try {
        for (let line of fs.readFileSync(logFile, 'utf8').split('\n')) {
            if (!line.startsWith('{')) {
                continue
            }
            let j
            try {
                j = JSON.parse(line)
            }
            catch (err) {
                continue
            }
            if (j.type === 'step_start' || (j.part && j.part.type === 'step-start')) {
                sawStep = true
                cur = []
            }
            for (let c of callsOf(j)) {
                cur.push(c)
            }
        }
    }
    catch (err) {
        return null
    }
    let last = (sawStep ? cur : cur.slice(-1)).find((c) => c.status === 'error' && typeof c.error === 'string' && /rejected permission|permission (was |is )?denied|not permitted/i.test(c.error))
    if (!last) {
        return null
    }
    let inp = last.input || {}
    return { tool: last.tool, target: String(inp.filePath || inp.file_path || inp.path || inp.command || '').replace(/\s+/g, ' ').slice(0, 120) }
}

/**

let guardExtraOf = (st) => {
    let out = []

        }
    }
    return out
}

//stripSearchPatterns, 去掉指令內搜尋類指令(grep / egrep / fgrep / rg / ag / ack / findstr / git grep)之「搜尋樣式」引數; 其餘(含被搜尋之路徑)原樣保留
//  做夢 agent 會以禁區字樣為搜尋目標自檢(例: grep -rlE "g_backtest|ARSI_HARNESS" rsi/toolbox/): 樣式不是存取, 被搜尋之路徑仍照查
//  (例: 做夢交付前以 grep 自檢共用工具區與自己的腳本, 曾被記為觸及禁區 —— 誤報)
//  只認「該段指令以搜尋指令開頭」者(經 xargs、env 前綴等間接呼叫者不去, 寧可多報)
let SEARCH_CMD = /^(?:(?:\S*[\\/])?(?:grep|egrep|fgrep|rg|ag|ack|findstr)(?:\.exe)?|git(?:\s+-C\s+\S+)?\s+grep)(?=\s|$)/i
let SEARCH_OPT_VAL = /^(-[ABCmdDfgt]|--(?:max-count|after-context|before-context|context|include|exclude|exclude-dir|glob|type|file))$/
let stripSearchPatterns = (cmd) => {
    //引號內之內容先以佔位符保護(樣式內之 |、;、& 不是指令分隔)
    let quoted = []
    let masked = String(cmd).replace(/"(?:[^"\\]|\\.)*"|'[^']*'/g, (m) => {
        quoted.push(m)
        return `\u0001${quoted.length - 1}\u0002`
    })
    let segs = masked.split(/(&&|\|\||[;|\n])/).map((seg) => {
        let lead = /^\s*/.exec(seg)[0]
        let body = seg.slice(lead.length)
        let m0 = SEARCH_CMD.exec(body)
        if (!m0) {
            return seg
        }
        let isFindstr = /findstr/i.test(m0[0])
        let toks = body.slice(m0[0].length).split(/(\s+)/)
        let removed = false
        let skipNext = false
        for (let i = 0; i < toks.length; i++) {
            let t = toks[i]
            if (!t || /^\s+$/.test(t)) {
                continue
            }
            if (skipNext) {
                skipNext = false
                continue
            }
            if (/^(-e|--regexp)$/.test(t)) {
                //其後第一個引數即樣式
                for (let k = i + 1; k < toks.length; k++) {
                    if (toks[k] && !/^\s+$/.test(toks[k])) {
                        toks[k] = ' '
                        break
                    }
                }
                removed = true
                continue
            }
            if (/^--regexp=/.test(t)) {
                toks[i] = '--regexp='
                removed = true
                continue
            }
            if (isFindstr && /^\/[A-Za-z]+(:.*)?$/.test(t)) {
                if (/^\/C:/i.test(t)) {
                    toks[i] = '/C:'
                    removed = true
                }
                continue
            }
            if (SEARCH_OPT_VAL.test(t)) {
                skipNext = true
                continue
            }
            if (t.startsWith('-')) {
                continue
            }
            //第一個非選項引數 = 樣式(已由 -e / --regexp / /C: 給樣式者, 其餘皆為路徑)
            if (!removed) {
                toks[i] = ' '
                removed = true
            }
            break
        }
        return lead + m0[0] + toks.join('')
    })
    return segs.join('').replace(/\u0001(\d+)\u0002/g, (_, i) => quoted[Number(i)]) // eslint-disable-line no-control-regex
}

 * @param {String|null} [opt.logFile=null] 輸入操作紀錄路徑,預設null

 * @param {String|null} [opt.root=null] 輸入專案根路徑,與 writeRoots 並用時查寫出範圍
 * @param {Array<String>|null} [opt.writeRoots=null] 輸入允許寫入之範圍路徑陣列,預設null
 * @param {String|null} [opt.workRoot=null] 輸入各節點工作夾之根路徑,給定時查存取其他節點工作夾
 * @param {Array<String>} [opt.allowWork=[]] 輸入允許讀取之節點 id 陣列,預設[]
 * @param {String} [opt.fileHits='issue'] 輸入程式檔含禁區字樣之記法('issue' 列為問題、'note' 另列待核對),預設'issue'
 * @param {Boolean} [opt.nodeRules=false] 輸入是否套用節點專用規則,預設false
 * @param {Boolean|null} [opt.planRule=null] 輸入是否查「規劃先於評估」,預設null(同 nodeRules)
 * @returns {Object} 回傳 { issues, note, nEvals, fileNotes, planAfterEval }
 * @example
 * let r = scanTrace({ workDir: './rsi/work/node1', attempt: 1, privateRoot: './private', nodeRules: true })
 */
let scanTrace = ({ workDir, attempt, logFile = null, privateRoot, root = null, writeRoots = null, workRoot = null, allowWork = [], fileHits = 'issue', nodeRules = false, planRule = null }) => {
    let planOn = planRule === null ? nodeRules : planRule === true
    let pats = forbiddenPats(privateRoot, nodeRules)
    //工作夾之根(相對於專案根; 例如 rsi/work)後面不接節點 id 者 = 列出全部節點之工作夾
    let reListRoot = null
    if (nodeRules && workRoot) {
        let segs = path.relative(root || path.dirname(path.dirname(workRoot)), workRoot).split(/[\\/]+/).filter((x) => x && x !== '.').map((x) => x.replace(/[.*+?^${}()|[\]\\]/g, '\\$&'))
        if (segs.length) {
            reListRoot = new RegExp(`(^|[\\s"'=(\\\\/])${segs.join('[\\\\/]+')}[\\\\/]*(?=$|[\\s"'|;&)*])`)
        }
    }
    let hits = new Map()
    let fileMap = new Map()
    let test = (s, where, to = hits) => {
        for (let [re, label] of pats) {
            if (re.test(s)) {
                let k = `${label}(${where})`
                if (!to.has(k)) {
                    to.set(k, String(s).replace(/\s+/g, ' ').slice(0, 160))
                }
            }
        }
    }
    let note = null
    let nEvals = null
    //規劃先於評估(nodeRules): 第幾個工具呼叫首次寫入節點文件、第幾個首次以完整模式評估自己的實驗檔(起點之複本 base_pick.mjs、參考節點成品之複本 ref_<id 前 8 碼>.mjs 與範例不算: 重現既有成品不是選版)
    let firstDoc = null
    let firstFull = null
    let fpLog = logFile || path.join(workDir, `agent_a${attempt}.log`)
    if (fs.existsSync(fpLog)) {
        let nEv = 0
        nEvals = 0
        for (let line of fs.readFileSync(fpLog, 'utf8').split('\n')) {
            if (!line.startsWith('{')) {
                continue
            }
            let j
            try {
                j = JSON.parse(line)
            }
            catch (err) {
                continue
            }
            for (let { tool, input: inp, status, error } of callsOf(j)) {
                nEv++
                //被權限拒絕之工具呼叫: 未執行(例: 寫檔工具以打錯之絕對路徑呼叫而被拒); 其寫檔目標另標「企圖」, 不與實際寫出者同列
                let denied = status === 'error' && typeof error === 'string' && /rejected permission|permission (was |is )?denied|not permitted/i.test(error)
                let fpTarget = inp.filePath || inp.file_path || inp.path
                let isWriteTool = /write|edit|patch|create/.test(tool)
                let isSearchTool = /grep|search/i.test(tool)
                for (let [k, v] of Object.entries(inp)) {
                    //寫檔工具之 content 是檔案內容(其程式檔另由下方掃), 不當成指令或路徑
                    if (typeof v !== 'string' || CONTENT_KEYS.includes(k)) {
                        continue
                    }
                    //做夢(非節點規則): 搜尋樣式與指令之說明文字不是存取(以禁區字樣為搜尋目標之自檢); 被搜尋之路徑照查。
                    //  節點搜尋禁區字樣屬探查, 照列
                    if (!nodeRules) {
                        if (k === 'description' || (isSearchTool && /^(pattern|query|regex)$/i.test(k))) {
                            continue
                        }
                        if (k === 'command') {
                            test(stripSearchPatterns(v), '指令或路徑')
                            continue
                        }
                    }
                    test(v, '指令或路徑')
                    if (nodeRules && NODE_CMD_KEYS.includes(k)) {
                        for (let label of nodeCmdHits(k, v)) {
                            let kk = `${label}(指令或路徑)`
                            if (!hits.has(kk)) {
                                hits.set(kk, String(v).replace(/\s+/g, ' ').slice(0, 160))
                            }
                        }
                    }
                }
                //評估次數: 指令內每出現一次「node … g_eval.mjs … --pick」算一次(以 shell 迴圈批次評估者會低估; 以 grep / cat 提到檔名者不算)
                if (typeof inp.command === 'string') {
                    nEvals += (inp.command.match(/node\s[^|;&]*g_eval\.mjs[^|;&]*--pick\s/g) || []).length
                }
                //規劃先於評估: 節點文件之首次寫入(寫檔 / 改檔工具, 或指令以重導向、tee、writeFileSync 寫 node.json; 以 cp 複製範本不算)與第一次完整評估
                if (planOn) {
                    if (firstDoc === null) {
                        let byTool = isWriteTool && typeof fpTarget === 'string' && /(^|[\\/])node\.json$/i.test(fpTarget)
                        let byCmd = typeof inp.command === 'string' && /(>>?\s*["']?[^\s"'|;&<>]*node\.json|\btee\s+(-a\s+)?["']?[^\s"'|;&<>]*node\.json|writeFileSync\([^)]*node\.json)/i.test(inp.command)
                        if (byTool || byCmd) {
                            firstDoc = nEv
                        }
                    }
                    if (firstFull === null && typeof inp.command === 'string') {
                        for (let m of inp.command.matchAll(/node\s[^|;&]*g_eval\.mjs[^|;&]*/g)) {
                            let pk = /--pick[=\s]+("[^"]+"|'[^']+'|[^\s|;&)]+)/.exec(m[0])
                            if (/--mode[=\s]+full\b/.test(m[0]) && pk && !/(^|[\\/])(base_pick|ref_[0-9a-f]{8})\.mjs["']?$/i.test(pk[1]) && !/(^|[\\/"'])examples[\\/]/i.test(pk[1])) {
                                firstFull = nEv
                                break
                            }
                        }
                    }
                }
                //存取其他節點之工作夾(併行節點互看會使結果不獨立; 只可讀任務書列為參考者)
                if (workRoot) {
                    let own = path.basename(workDir)
                    let base = path.basename(workRoot).replace(/[.*+?^${}()|[\]\\]/g, '\\$&')
                    //寫入文件(非程式檔)之內容是文字, 不是存取; 寫入程式檔之內容照查(腳本內讀他人工作夾者)
                    let docWrite = isWriteTool && typeof fpTarget === 'string' && !CODE_EXT.test(fpTarget)
                    let codeWrite = isWriteTool && typeof fpTarget === 'string' && CODE_EXT.test(fpTarget)
                    for (let [k, v0] of Object.entries(inp)) {
                        if (typeof v0 !== 'string' || (docWrite && CONTENT_KEYS.includes(k))) {
                            continue
                        }
                        //寫入程式檔之內容先去註解(與下方逐檔掃描同一作法): 共用片段之檔頭註解常含「出處 rsi/work/<某工作夾>/…」,
                        //  照方向複製片段原文者不是存取(此類曾被誤報); 非註解之程式碼照查
                        let v = codeWrite && CONTENT_KEYS.includes(k) ? stripComments(v0) : v0
                        for (let m of v.matchAll(new RegExp(`${base}[\\\\/]+([A-Za-z0-9_]+)`, 'g'))) {
                            let id = m[1]
                            if (id !== own && !allowWork.includes(id)) {
                                let key = `存取其他節點之工作夾(${id.slice(0, 12)})`
                                if (!hits.has(key)) {
                                    hits.set(key, v.replace(/\s+/g, ' ').slice(0, 160))
                                }
                            }
                        }
                        if (reListRoot && reListRoot.test(v)) {
                            let key = '存取其他節點之工作夾(列出工作夾之根)'
                            if (!hits.has(key)) {
                                hits.set(key, v.replace(/\s+/g, ' ').slice(0, 160))
                            }
                        }
                    }
                }
                //寫到允許範圍之外
                if (root && writeRoots) {
                    let targets = []
                    let fp = inp.filePath || inp.file_path || inp.path
                    if (/write|edit|patch|create/.test(tool) && typeof fp === 'string') {
                        targets.push(fp)
                    }
                    if (typeof inp.command === 'string') {
                        for (let m of inp.command.matchAll(/--out[=\s]+("[^"]+"|'[^']+'|[^\s|;&)]+)/g)) {
                            //純數字不是路徑(agent 自寫之腳本可能以 --out 為數量參數, 例如「--out 2500」)
                            if (!/^["']?\d+(\.\d+)?["']?$/.test(m[1])) {
                                targets.push(m[1])
                            }
                        }
                        //輸出重導向: 先去掉 here-doc 之內文(只留其首列; 內文之 > 不是重導向 —— 例: 以 cat >> 檔 << 'EOF' 追加文件時之誤報),
                        //  再去掉引號內之內容(node -e "…" 等內嵌程式之 =>、> 不是重導向), 再找 > 或 >>
                        let noDoc = inp.command.replace(/(<<-?\s*(['"]?)([A-Za-z_][A-Za-z0-9_]*)\2[^\n]*)\n[\s\S]*?\n[ \t]*\3[ \t]*(?=\n|$)/g, '$1')
                        let bare = noDoc.replace(/"(?:[^"\\]|\\.)*"|'[^']*'/g, ' Q ')
                        for (let m of bare.matchAll(/(?:^|[^0-9&>=-])>>?\s*([^\s|;&)<>=]+)/g)) {
                            if (!/^&|^\/dev\/null$|^NUL$|^Q$/i.test(m[1])) {
                                //相對路徑之重導向以其前最近一個 cd 之目錄為準(cd <夾> && … > 檔 —— 例: 於自己之工作夾內如此寫檔而被誤報);
                                //  該目錄無法靜態判定者(引號內、含 shell 變數)照舊以專案根為準
                                let cds = [...bare.slice(0, m.index).matchAll(/(?:^|&&|;|\n)\s*cd\s+([^\s;&|]+)/g)]
                                let cd = cds.length ? cds[cds.length - 1][1] : null
                                let relative = !/^([a-zA-Z]:[\\/]|[\\/]|~)/.test(m[1])
                                targets.push(cd && cd !== 'Q' && !/\$|`/.test(cd) && relative ? path.resolve(absOf(root, cd), m[1]) : m[1])
                            }
                        }
                    }
                    for (let t of targets) {
                        if (/\$|`/.test(t)) {
                            continue //含 shell 變數之路徑無法靜態判定
                        }
                        let abs = absOf(root, t)
                        let ok = writeRoots.some((w) => {
                            let r = path.relative(w, abs); return r === '' || (!r.startsWith('..') && !path.isAbsolute(r))
                        })
                        if (!ok) {
                            //被權限拒絕者未寫出任何檔: 仍列出(供審計), 但標為企圖(例: 寫檔工具以打錯專案名之絕對路徑呼叫而被拒)
                            let key = `${denied ? '寫到工作夾外之企圖(被拒,未寫出)' : '寫到工作夾外'}(${path.relative(root, abs).replace(/\\/g, '/')})`
                            if (!hits.has(key)) {
                                hits.set(key, String(inp.command || fp).replace(/\s+/g, ' ').slice(0, 160))
                            }
                        }
                    }
                }
            }
        }
        if (nEv === 0) {
            note = '操作紀錄無可解析之工具呼叫(未掃指令)'
            nEvals = null
        }
    }
    else {
        note = '無操作紀錄檔'
    }
    for (let f of fs.readdirSync(workDir)) {
        let fp = path.join(workDir, f)
        if (!fs.statSync(fp).isFile() || !/\.(mjs|js|cjs|ts|py|sh|ps1|bat|cmd)$/i.test(f)) {
            continue
        }
        if (fs.statSync(fp).size > 2 * 1024 * 1024) {
            continue
        }
        test(stripComments(fs.readFileSync(fp, 'utf8')), `檔 ${f}`, fileHits === 'note' ? fileMap : hits)
    }
    let issues = [...hits.entries()].map(([k, s]) => (/^(寫到工作夾外|存取其他節點之工作夾)/.test(k) ? `${k}: ${s}` : `觸及禁區: ${k}: ${s}`))
    //規劃於評估後才寫: 兩個時點皆由操作紀錄看得到才判(看不到節點文件之寫入者不判, 以免誤報)
    let planAfterEval = planOn && firstDoc !== null && firstFull !== null && firstDoc > firstFull ? { docCall: firstDoc, evalCall: firstFull } : null
    if (planAfterEval) {
        issues.push(`規劃於評估後才寫: 節點文件(node.json)首次寫入為第 ${firstDoc} 個工具呼叫, 第一次完整評估為第 ${firstFull} 個(任務書 §6「先寫規劃再評估」)`)
    }
    return { issues, note, nEvals, fileNotes: [...fileMap.keys()], planAfterEval }
}

//evalLines, 工作夾之 .evals.log 之各列(原文); 無此檔回 null
//  until(選配; ISO 時刻): 只取 at ≤ until 之列 —— 節點之 agent 結束之後才寫入之列不是該節點之操作
//    (做夢或主控日後直接評估該工作夾內之模組時, 評估器照樣記於此檔); 讀不出 at 之列照留
let evalLines = (workDir, until = null) => {
    let fp = path.join(workDir, '.evals.log')
    if (!fs.existsSync(fp)) {
        return null
    }
    let ls = fs.readFileSync(fp, 'utf8').split('\n')
    if (typeof until !== 'string' || !until) {
        return ls
    }
    return ls.filter((l) => {
        let m = /"at":"([^"]+)"/.exec(l)
        return !m || m[1] <= until
    })
}

//evalsLogged, 評估器自己記下之評估次數(工作夾之 .evals.log; 每次 g_eval.mjs 執行一列); 無此檔回 null
let evalsLogged = (workDir, until = null) => {
    let ls = evalLines(workDir, until)
    if (!ls) {
        return null
    }
    return ls.filter((l) => l.trim().startsWith('{')).length
}

//evalsDistinct, 完整評估之次數與其相異版本數(工作夾之 .evals.log; 只計成功之 full 列); 無此檔回 null
//  nFull = 完整評估之次數; nFullSel = 其相異 selHash 數(相異之選集序列; 只差 tag、只加 --ref 或同一內容重跑者不算新版本);
//  nFullMod = 其相異模組雜湊數(相異之檔案內容; 舊紀錄無 h 者不計)
//  「試得越多, 其中最佳者越可能只是運氣」以相異 selHash 數衡量; 總次數照記(evalsLogged)
let evalsDistinct = (workDir, until = null) => {
    let ls = evalLines(workDir, until)
    if (!ls) {
        return null
    }
    let nFull = 0
    let sel = new Set()
    let mod = new Set()
    for (let l of ls) {
        if (!l.trim().startsWith('{')) {
            continue
        }
        let j
        try {
            j = JSON.parse(l)
        }
        catch (err) {
            continue
        }
        //只計評估期(回測期)之完整評估; --period train 之診斷不是版本之評估
        if (j.ok !== true || j.mode !== 'full' || (j.period && j.period !== 'backtest')) {
            continue
        }
        nFull++
        if (typeof j.selHash === 'string') {
            sel.add(j.selHash)
        }
        if (typeof j.h === 'string') {
            mod.add(j.h)
        }
    }
    return { nFull, nFullSel: sel.size, nFullMod: mod.size }
}

//noLogTried, .evals.log 內標記 noLogIgnored 之列數(非迴圈之呼叫者帶 --no-log 而被照記者)
let noLogTried = (workDir, until = null) => {
    let ls = evalLines(workDir, until)
    if (!ls) {
        return 0
    }
    return ls.filter((l) => l.includes('"noLogIgnored":true')).length
}

//lostVersions, 已以完整模式評估過、而工作夾內現已無任何檔之內容與之相同之版本(評估後覆寫了該檔 ⇒ 該次實驗無檔可重現; 任務書「一實驗一檔」)
//  依 .evals.log 各列之 h(模組內容之雜湊; g_eval.mjs 所記)與工作夾內現有程式檔之雜湊比對; 回 [{ pick, n }](依檔名彙整; 舊紀錄無 h 者略過)
let lostVersions = (workDir, until = null) => {
    let ls = evalLines(workDir, until)
    if (!ls) {
        return []
    }
    let now = new Set()
    for (let f of fs.readdirSync(workDir)) {
        let p = path.join(workDir, f)
        if (/\.(mjs|js|cjs)$/i.test(f) && fs.statSync(p).isFile()) {
            now.add(crypto.createHash('sha1').update(fs.readFileSync(p)).digest('hex').slice(0, 12))
        }
    }
    let lost = new Map()
    for (let l of ls) {
        if (!l.trim().startsWith('{')) {
            continue
        }
        let j
        try {
            j = JSON.parse(l)
        }
        catch (err) {
            continue
        }
        if (j.ok === true && j.mode === 'full' && !(j.period && j.period !== 'backtest') && typeof j.h === 'string' && !now.has(j.h)) {
            if (!lost.has(j.pick)) {
                lost.set(j.pick, new Set())
            }
            lost.get(j.pick).add(j.h)
        }
    }
    return [...lost.entries()].map(([pick, hs]) => ({ pick, n: hs.size }))
}

//summaryOf, 節點文件之重點數字(與 agent 自報 results.claimed 同鍵者逐項比對); 全部為評估期(回測期)之數字; trainScore = 不含檢定旗標之判讀(agent 自跑看得到的那一份)
/**
 * 取得節點文件之重點數字(與 agent 自報 results.claimed 同鍵者逐項比對;皆為回測期之數字)
 *
 * @param {Object} m 輸入評估之 metrics,讀取 op.eqYr、op.monthlyOrders、op.peakHold、null、selHash
 * @param {Object} trainScore 輸入不含檢定旗標之判讀,讀取 S、ladder
 * @returns {Array<Object>} 回傳陣列,每項 { key, label, value, tol? },key 為 S、ladder、eqYrMean、eqYrMin、monthlyOrders、peakHold、nullP、selHash
 * @example
 * let rows = summaryOf(metrics, { S: 1.2, ladder: 2 })
 */
let summaryOf = (m, trainScore) => [
    { key: 'S', label: 'S(回測期)', value: trainScore.S },
    { key: 'ladder', label: '階梯(回測期)', value: trainScore.ladder, tol: 0 },
    { key: 'eqYrMean', label: '等效年化 七相位平均(%;回測期)', value: m.op.eqYr.mean },
    { key: 'eqYrMin', label: '等效年化 最差相位(%;回測期)', value: m.op.eqYr.min },
    { key: 'monthlyOrders', label: '月均單', value: m.op.monthlyOrders.mean },
    { key: 'peakHold', label: '最大同時持倉', value: m.op.peakHold.max, tol: 0 },
    { key: 'nullP', label: '同劑量虛無 p', value: m.null ? m.null.p : null, tol: 0.0001 },
    { key: 'selHash', label: 'selHash(選集與下單序列之指紋)', value: m.selHash },
]

//notesOf, 節點文件之補充說明列(領域指標; 全部為評估期 = 回測期之數字)
//  seedS(選配): 迴圈換亂數種子重放所得之 S(主種子以外者)
//  train(選配): 迴圈以訓練期之診斷重放所得之選集組成 { longRatio, tags }(只記組成, 不記績效 —— 訓練期之績效不是目標)
//下單流量偏一側之機械判讀(規則 R-REC-15)
//  shortShareOf(metrics): 做空單 ÷ 全部下單(各相位平均之單數; 無下單或無此欄 ⇒ null)
//  sideFlowIssue(回測期之值, 訓練期診斷之值, cfg): 兩期皆 ≥ 全體隨機基準之 hi 倍(或皆 ≤ lo 倍)⇒ 回問題字串, 否則 null
//    基準 = 中性隨機 40 隻(templates/neutral_random40.mjs)之同一比例: 回測期約 0.16、訓練期(quick 診斷)約 0.22; 可由 settings.score.sideFlow 覆寫
//    例: 「做多取甲規則、做空取乙規則」之分側組合, 席次之做多比例兩期皆正常(固定側別之檢查不列), 但兩期之下單皆大半為做空單 ——
//    兩側之活躍度不對稱使下單集中於做空側, 等於把回測期之方向寫進模組; S、階梯與同側同活躍度虛無都照樣保留它, 故另設此檢查(不否決, 由做夢審計)
let SIDE_FLOW = { baseShort: { backtest: 0.16, train: 0.22 }, hi: 2.5, lo: 0.4 }
let shortShareOf = (m) => {
    let b = m && m.op && m.op.bySide
    if (!b || !b.long || !b.short || typeof b.long.orders !== 'number' || typeof b.short.orders !== 'number' || b.long.orders + b.short.orders <= 0) {
        return null
    }
    return Math.round((b.short.orders / (b.long.orders + b.short.orders)) * 1000) / 1000
}
let sideFlowIssue = (bt, tr, cfg = null) => {
    if (typeof bt !== 'number' || typeof tr !== 'number') {
        return null
    }
    let c = { ...SIDE_FLOW, ...(cfg || {}), baseShort: { ...SIDE_FLOW.baseShort, ...((cfg && cfg.baseShort) || {}) } }
    let up = bt >= c.baseShort.backtest * c.hi && tr >= c.baseShort.train * c.hi
    let dn = bt <= c.baseShort.backtest * c.lo && tr <= c.baseShort.train * c.lo
    if (!up && !dn) {
        return null
    }
    return `下單流量偏一側: 做空單之比例 訓練期 ${tr} / 回測期 ${bt}(全體隨機約 ${c.baseShort.train} / ${c.baseShort.backtest};兩期皆${up ? `≥ ${c.hi} 倍` : `≤ ${c.lo} 倍`} ⇒ 下單偏${up ? '做空' : '做多'}側屬組成,回測期之方向可能經由兩側活躍度之不對稱寫進模組;由審計判定)`
}

/**
 * 產生節點文件之補充說明列(領域指標;皆為回測期之數字)
 *

 * @param {Array<Number>|null} [seedS=null] 輸入換亂數種子重放所得之 S 陣列,預設null
 * @param {Object|null} [train=null] 輸入訓練期診斷之選集組成 { longRatio, tags },預設null
 * @param {Array<Object>|null} [replays=null] 輸入各次重放之 { S, comp, recent },預設null
 * @param {Boolean} [half=false] 輸入是否為半庫版,預設false
 * @returns {Array<String>} 回傳說明列字串陣列
 * @example
 * let lines = notesOf(metrics, [1.1, 1.3], null, null, false)
 */
let notesOf = (m, seedS = null, train = null, replays = null, half = false) => {
    let f1 = (x) => (typeof x === 'number' ? x.toFixed(1) : '—')
    let out = []
    if (m.period && m.period.evalFrom) {
        out.push(`評估期間(回測期): ${String(m.period.evalFrom).slice(0, 10)} 起至 ${String(m.period.endTime).slice(0, 10)};首個計入之決策 ${String(m.period.firstEvalDecision || '').slice(0, 10)};各相位決策 ${m.op.nDecisions.join(' / ')} 次`)
    }

        }

    }
    //擴充診斷(共用工具區 metrics/; 由做夢擴建; 每次完整評估自動算出): 每個一列, 取其自述之一行說明
    for (let [nm, r] of Object.entries(m.ext || {})) {
        out.push(`擴充診斷 ${nm}: ${r && r.error ? `未算(${r.error})` : (r && typeof r.text === 'string' ? r.text.slice(0, 600) : JSON.stringify(r).slice(0, 300))}`)
    }
    if (m.op.bySide) {
        let sd = (x) => `${f1(x.orders)} 單、損益 ${x.pnl}、勝率 ${f1(x.winRate * 100)}%`
        out.push(`損益歸因(各相位平均;損益 = 每單 1 單位之損益和,合計 ${m.op.pnl}): 做多 ${sd(m.op.bySide.long)};做空 ${sd(m.op.bySide.short)}${Array.isArray(m.op.byGroup) && m.op.byGroup.length > 1 ? `;依子帳: ${m.op.byGroup.map((g) => `子帳 ${g.group}(每次決策 ${f1(g.seats)} 席)${sd(g)}`).join(' | ')}` : ''}${m.sel && Array.isArray(m.sel.tagPerf) && m.sel.tagPerf.length ? `;依 tag(前 6 種): ${m.sel.tagPerf.slice(0, 6).map((t) => `${t.tag} ${t.decisions} 次決策、${f1(t.orders)} 單、損益 ${t.pnl}`).join(' | ')}` : ''}`)
    }
    if (Array.isArray(m.op.thirds) && m.op.thirds.length) {
        out.push(`評估期間三等分(各段自空倉起算;看績效是否集中於某一段): ${m.op.thirds.map((g) => `${String(g.from).slice(0, 10)} ~ ${String(g.to).slice(0, 10)} 均 ${f1(g.eqYr.mean)}%(最差相位 ${f1(g.eqYr.min)}${typeof g.nullMean === 'number' ? `;隨機挑 ${f1(g.nullMean)}` : ''})`).join(' | ')}`)
    }
    if (train && typeof train.longRatio === 'number' && m.sel && typeof m.sel.longRatio === 'number') {
        let tg = (xs) => (Array.isArray(xs) && xs.length ? xs.slice(0, 4).map((t) => `${t.tag} ${t.share}`).join('、') : '無')

        let sb = shortShareOf(m)
        if (typeof train.shortShare === 'number' && sb !== null) {
            out.push(`下單之做空比例(做空單 ÷ 全部下單): 訓練期(診斷)${train.shortShare} / 回測期 ${sb};全體隨機約 ${SIDE_FLOW.baseShort.train} / ${SIDE_FLOW.baseShort.backtest} —— 席次之做多比例正常而下單偏在一側者,是兩側活躍度不對稱所致之組成;兩期皆遠高(或遠低)於隨機者,回測期之方向可能經此寫進模組`)
        }
    }
    if (m.fresh) {
        out.push(`新鮮度(七相位均 %): ${Object.keys(m.fresh).map((H) => `${H}d ${f1(m.fresh[H].eqYr.mean)}`).join(' | ')}`)
    }
    if (m.sel) {
        out.push(`選集: 大小均 ${f1(m.sel.sizeMean)}(${m.sel.sizeMin} ~ ${m.sel.sizeMax});換手 Jaccard ${m.sel.turnoverJaccard};執行規則 ${m.hasAccept ? `有(下單率 ${f1(m.op.acceptRate * 100)}%)` : '無'};勝率 ${f1(m.op.winRate * 100)}%`)
        if (typeof m.sel.sigPerPick === 'number') {
            //選集組成: 分數受活躍度與多空組成之影響很大, 故與同劑量虛無(全體隨機)並列, 一眼可見成品是否只是換了活躍度或多空比例
            out.push(`選集組成(括號內為同劑量虛無 = 全體隨機): 每隻每個決策視窗之訊號數 ${m.sel.sigPerPick}${m.null && typeof m.null.sigPerPick === 'number' ? `(${m.null.sigPerPick})` : ''};做多比例 ${m.sel.longRatio}${m.null && typeof m.null.longRatio === 'number' ? `(${m.null.longRatio})` : ''};平均因子數 ${m.sel.nKeysMean};平均止盈 / 止損 ${m.sel.tpMean} / ${m.sel.slMean}`)
        }
        if (m.sel.orders) {
            //下單集中度: 選集很大而下單率低時, 實際下單者可能集中在少數策略 / 因子組合 / 進場棒
            out.push(`下單集中度(各相位平均): 實際下單之不同策略 ${f1(m.sel.orders.strats)} 隻、不同因子組合 ${f1(m.sel.orders.combos)} 種;下單最多之前 1% 進場棒占全部下單 ${f1(m.sel.orders.barTop1Share * 100)}%;單一進場棒最多 ${m.sel.orders.perBarMax} 單`)
        }
    }
    return out
}

//datedIn, 一段程式(已去註解)內第一個落在某期間之年份內之日期字樣(YYYY-MM 或 YYYY-MM-DD; 分隔可為 - / .); 無則 null
//  period = { start, end }(時間字串): 只看年份範圍 —— 粗篩「依評估期間內之日期寫死行為」; 訓練期之日期(例如求解窗終點)不列
let datedIn = (src, period) => {
    let y0 = Number(String(period.start).slice(0, 4))
    let y1 = Number(String(period.end).slice(0, 4))
    let m0 = Number(String(period.start).slice(5, 7))
    let years = []
    for (let y = y0; y <= y1; y++) {
        years.push(y)
    }
    let re = new RegExp(`(${years.join('|')})[-/.](0[1-9]|1[0-2])(?:[-/.](?:[0-2]\\d|3[01]))?`, 'g')
    for (let m of String(src).matchAll(re)) {
        //起始年之內, 早於期間起始月者屬期間之前(例: 回測期自 2025-10 起, 2025-04 為訓練期內之日期)
        if (Number(m[1]) === y0 && Number(m[2]) < m0) {
            continue
        }
        return m[0]
    }
    return null
}

        //  第一次重放之選集與 S 皆與主重放相同者(未用亂數、同分不影響入選、持倉上限未因先後而擋單)不再重放
        //  每次重放另附算組成對照(32 抽)與近期活躍度之組成對照(擴充診斷 nullRecentAct)之超額(規則 R-EV-24):
        //    只列 S 時看不出「內容」在重放下是否仍在; 單抽之組成對照不足以判內容(全庫隨機之兩組抽取 +1.3 / +11.6), 兩次重放即現成之第二、三抽
        let seedS = []
        let replays = []
        let isHalf = !!(out.pickMeta && out.pickMeta.params && (out.pickMeta.params.half === 0 || out.pickMeta.params.half === 1))
        {
            let nRep = Number.isInteger(st.eval.replicaSeeds) ? st.eval.replicaSeeds : 0
            let seed0 = st.eval.seed === undefined ? 1 : st.eval.seed
            let nDraw = Number.isInteger(st.eval.replicaNullDraws) ? st.eval.replicaNullDraws : 32
            for (let i = 1; i <= nRep; i++) {
                let fpS = path.join(workDir, `harness_seed${i}.json`)
                try {
                    fs.rmSync(fpS, { force: true })
                }
                catch (err) {}
                let rs = await runNodeProc([path.join(FD_ROOT, 'g_eval.mjs'), '--pick', fpPick, '--mode', 'full', '--null', String(nDraw), '--ext-only', 'nullRecentAct', '--seed', String(seed0 + i), '--tie-salt', String(i), '--out', fpS, '--no-log'], { cwd: FD_ROOT, timeoutMs: (st.eval.timeoutSec || 900) * 1000, env: { ...process.env, ARSI_HARNESS: '1' } })
                let o = readJson(fpS)
                if (rs.timeout || !o || o.ok !== true || !o.score) {
                    issues.push(`換亂數種子與同分次序之重放失敗(第 ${i} 次)`)
                    break
                }
                if (i === 1 && o.metrics.selHash === out.metrics.selHash && out.score && o.score.S === out.score.S) {
                    break //結果不依賴亂數種子與同分次序
                }
                seedS.push(o.score.S)
                let nr = o.metrics.ext && o.metrics.ext.nullRecentAct && o.metrics.ext.nullRecentAct.side
                replays.push({ S: o.score.S, comp: o.metrics.nullSideAct && typeof o.metrics.nullSideAct.excess === 'number' ? o.metrics.nullSideAct.excess : null, recent: nr && typeof nr.excess === 'number' ? nr.excess : null })
            }
            //主分數為正而重放後為零或負: 分數依賴亂數抽取或同分次序(不否決, 由做夢審計)
            //  半庫版(成品自述 meta.params.half 為 0 或 1; 共用工具區之 *_half.mjs 樣板: 以 ctx.meta.tie 之雜湊切半)換同分次序即換一種切法 ⇒ 不列「分數不穩」,
            //  改於判讀加註(判準由成品自述, 由審計核對 —— 與擴充診斷之 notContent 同一作法)
            let sMain = out.score && typeof out.score.S === 'number' ? out.score.S : null
            if (!isHalf && seedS.length && sMain !== null && sMain > 0 && Math.min(...seedS) <= 0) {
                issues.push(`分數不穩: 主 S ${sMain},換亂數種子與同分次序後為 ${seedS.join(' / ')}(由正轉負;含隨機抽取或主鍵大量同分 ⇒ 入選者實際由亂數或 ctx.meta.tie 決定,由審計判定)`)
            }
        }
        //固定側別之機械判讀: 另以訓練期之診斷重放一次(quick、不算虛無), 只取選集組成(做多比例、tag 之決策比例)
        //  回測期若為單邊行情, 固定只挑一側者即可得高分而毫無自適應; 訓練期與回測期之做多比例皆 < 0.1 或皆 > 0.9 者列為問題(不否決, 由做夢審計)
        let train = null
        {
            let fpT = path.join(workDir, 'harness_train.json')
            try {
                fs.rmSync(fpT, { force: true })
            }
            catch (err) {}
            let rt = await runNodeProc([path.join(FD_ROOT, 'g_eval.mjs'), '--pick', fpPick, '--mode', 'quick', '--period', 'train', '--null', '0', '--out', fpT, '--no-log'], { cwd: FD_ROOT, timeoutMs: (st.eval.timeoutSec || 900) * 1000, env: { ...process.env, ARSI_HARNESS: '1' } })
            let o = readJson(fpT)
            if (!rt.timeout && o && o.ok === true && o.metrics && o.metrics.sel && typeof o.metrics.sel.longRatio === 'number') {
                train = { longRatio: o.metrics.sel.longRatio, tags: (o.metrics.sel.tags || []).slice(0, 4), ...(shortShareOf(o.metrics) === null ? {} : { shortShare: shortShareOf(o.metrics) }) }
                let a = train.longRatio
                let b = out.metrics.sel.longRatio
                if (out.metrics.sel.sizeMax > 0 && ((a < 0.1 && b < 0.1) || (a > 0.9 && b > 0.9))) {
                    issues.push(`固定側別: 選集之做多比例 訓練期 ${a} / 回測期 ${b}(兩期皆只挑${a < 0.1 ? '做空' : '做多'}側;回測期之行情方向可能被寫進模組,由審計判定)`)
                }
                //下單流量偏一側(規則 R-REC-15): 席次之做多比例正常, 仍可經由兩側活躍度之不對稱使下單集中在一側
                let sf = sideFlowIssue(shortShareOf(out.metrics), train.shortShare, st.score && st.score.sideFlow)
                if (sf) {
                    issues.push(sf)
                }
            }
        }
        //寫到工作夾之外之檔(常見: 路徑少一個分隔符 ⇒ <工作區>/<節點 id><檔名>): 移入工作夾之 _stray/ 並列為問題
        {
            let wr = path.dirname(workDir)
            let strays = []
            try {
                strays = fs.readdirSync(wr).filter((f) => f.startsWith(nodeId) && f !== path.basename(workDir) && fs.statSync(path.join(wr, f)).isFile())
            }
            catch (err) {}
            if (strays.length) {
                let moved = []
                for (let f of strays) {
                    try {
                        fs.mkdirSync(path.join(workDir, '_stray'), { recursive: true })
                        fs.renameSync(path.join(wr, f), path.join(workDir, '_stray', f.slice(nodeId.length) || f))
                        moved.push(f.slice(nodeId.length) || f)
                    }
                    catch (err) {}
                }
                issues.push(`有 ${strays.length} 個檔寫在工作夾之外(${strays.slice(0, 4).join('、')}${strays.length > 4 ? ' …' : ''};路徑少了分隔符)${moved.length ? `,已移入工作夾之 _stray/` : ''}`)
            }
        }
        //可讀之他節點工作夾: 起點(spec.base)與任務書列為參考之節點(由呼叫端經 allowWork 給入)
        let tr = scanTrace({ workDir, attempt, privateRoot: backtest.cfg.privateRoot, root: FD_ROOT, writeRoots: [workDir], workRoot: st.records.workDir, allowWork: [...(spec.base ? [spec.base] : []), ...(typeof refsOf === 'function' ? refsOf(spec) : [])], nodeRules: true })
        issues.push(...tr.issues)
        if (nNoLog > 0) {
            issues.push(`以 --no-log 執行評估 ${nNoLog} 次(該旗標只供迴圈重跑; 已照記入評估次數)`)
        }
        let lost = lostVersions(workDir, evalsUntil)
        if (lost.length) {
            issues.push(`已完整評估而現無檔可重現之版本 ${lost.reduce((s, x) => s + x.n, 0)} 個(${lost.map((x) => `${x.pick} × ${x.n}`).join('、')}): 評估後覆寫了該檔(一實驗一檔)`)
        }
        //評估次數: 以評估器之紀錄為準; 無紀錄檔(舊交付)才用操作紀錄數得者
        let nEvals = nLogged !== null ? nLogged : tr.nEvals
        //完整評估之相異版本數(迴圈之重跑帶 --no-log 不記, 故此處讀到的只有 agent 自己的評估)
        let nDist = evalsDistinct(workDir, evalsUntil)
        let layer = layerOfSpec(spec)
        let score = scoreOf(out.metrics, { limits: st.eval.limits, score: st.score, layer })
        let trainScore = score
        //檢定旗標(「回測期旗標」機制; 隨附之參考設定明設 settings.backtest.enabled = false(未設時 backtestCfg 視為啟用): 不啟用時評估期即回測期, 節點自己看得到數字;
        //  日後若由主控以保留期或他幣作檢定, 可沿用此接口, 見 sgud/proc/process.md §5)
        let validation = null
        if (keepValidation && typeof keepValidation.flag === 'string' && backtest.cfg.enabled) {
            validation = { name: 'backtest', flag: keepValidation.flag, label: gradeLabel(keepValidation.flag), at: keepValidation.at || null, error: null }
            score = scoreOf(out.metrics, { limits: st.eval.limits, score: st.score, layer, evidence: { backtest: keepValidation.flag } })
        }
        else if (backtest.cfg.enabled && score.ladder >= backtest.cfg.minLadder && !score.flags.vetoed) {
            let b = await backtest.run({ nodeId, attempt, workDir })
            validation = { name: 'backtest', flag: b.flag, label: b.flag ? gradeLabel(b.flag) : null, at: b.at, error: b.error }
            if (b.flag) {
                score = scoreOf(out.metrics, { limits: st.eval.limits, score: st.score, layer, evidence: { backtest: b.flag } })
            }
        }
        return {
            ok: true,
            raw: { metrics: out.metrics, selHash: out.metrics.selHash, seedS, replays, ...(train ? { train } : {}) },
            score,
            summary: summaryOf(out.metrics, trainScore),
            notes: notesOf(out.metrics, seedS, train, replays, isHalf),
            fingerprint: out.metrics.selHash,
            issues,
            checks: { ...(nEvals === null ? {} : { nEvals }), ...(nDist ? { ...nDist, nEvalsNote: `其中完整評估 ${nDist.nFull} 次、相異之選集(selHash)${nDist.nFullSel} 個;只差 tag、只加 --ref 或同一內容重跑者不算新版本` } : {}), ...(tr.note ? { traceNote: tr.note } : {}) },
            validation,
            evalJson: rel(fpOut),
        }
    }

//擴充診斷自帶之看板值與迴圈提示(規則 R-REC-21): 共用工具區 metrics/*.mjs 之回傳物件可另帶
//  board(看板之值; 字串或有限數字, 取前 32 字元)、boardLabel(該值之欄名; 取前 32 字元, 預設 = 診斷之檔名)、flag(迴圈檢查之提示; 非空字串, 取前 160 字元; 不否決)
//  ⇒ 做夢所建之診斷可自行上看板、列入迴圈檢查, 不必每次由主控改迴圈; 診斷出錯(error)或未帶者略過; 直線與換行改為空白(看板為表格)
//  notContent(規則 R-REC-22): 非空字串(取前 160 字元)= 診斷自述「此成品之組成對照不是挑選內容之量」及其理由
//    (例: 成品之執行規則非中性且本身帶來大增益 ⇒ 組成對照共用該規則, 其超額含執行層之增益)
//    ⇒ 搜尋地圖不把該成品計入「有內容」與量尺之「挑選內容」, 譜系與看板於組成對照旁註明; 該診斷之數值另以 board 上看板
/**
 * 取出擴充診斷自帶之看板值與迴圈提示(來自 metrics.ext 各診斷之 board、boardLabel、flag、notContent、subset;字串去除直線與換行)
 *
 * @param {Object} m 輸入評估之 metrics,讀取 m.ext
 * @returns {Object} 回傳 { cells, flags, notContent, subsets }:cells 每項 { name, label, value },flags 與 notContent 每項 { name, text },subsets 每項 { name, label }
 * @example
 * let s = extSurfaceOf({ ext: { demo: { board: 3, flag: '提示' } } })
 */
let extSurfaceOf = (m) => {
    let ext = m && m.ext && typeof m.ext === 'object' && !Array.isArray(m.ext) ? m.ext : {}
    let clean = (v, n) => String(v).replace(/[|\r\n]+/g, ' ').replace(/\s+/g, ' ').trim().slice(0, n)
    //  subset(規則 R-REC-27): 非空字串(取前 32 字元)= 診斷自述「此成品屬於某個子集」之名稱(例: 「時效 ≥ 15」)
    //    ⇒ 搜尋地圖 §1 於現行量尺旁, 另列各子集內「有內容且可信」之最高者(並列、不取代現行量尺); 條件不成立時回 null 或不帶; 子集之定義由診斷自述, 迴圈不寫死哪一個診斷
    let cells = []
    let flags = []
    let notContent = []
    let subsets = []
    for (let name of Object.keys(ext).sort()) {
        let o = ext[name]
        if (!o || typeof o !== 'object' || Array.isArray(o) || o.error) {
            continue
        }
        if (typeof o.subset === 'string' && o.subset.trim()) {
            subsets.push({ name, label: clean(o.subset, 32) })
        }
        if ((typeof o.board === 'string' && o.board.trim()) || (typeof o.board === 'number' && Number.isFinite(o.board))) {
            cells.push({ name, label: typeof o.boardLabel === 'string' && o.boardLabel.trim() ? clean(o.boardLabel, 32) : name, value: clean(o.board, 32) })
        }
        if (typeof o.flag === 'string' && o.flag.trim()) {
            flags.push({ name, text: clean(o.flag, 160) })
        }
        if (typeof o.notContent === 'string' && o.notContent.trim()) {
            notContent.push({ name, text: clean(o.notContent, 160) })
        }
    }
    return { cells, flags, notContent, subsets }
}
